# PP trade expeditions (docs/trade_expeditions.md): the trade companies founded by trade expeditions stay trade
# companies. EU5 1.4 keeps a subject type only while the overlord has an advance that unlocks it (vanilla: only
# trade_companies_advance, Age of Reformation); without it the engine turns the company into a plain vassal at its next
# validity check, and a landless vassal disappears (old expeditions branch, tested in game 2026-10-08). The ocean-going
# advances of the Age of Discovery unlock it as well.
TRY_INJECT:open_sea_exploration = {
	unlock_subject_type = trade_company
}

TRY_INJECT:explorer_commisions_advance = {
	unlock_subject_type = trade_company
}

TRY_INJECT:por_carreira_da_india = {
	unlock_subject_type = trade_company
}
