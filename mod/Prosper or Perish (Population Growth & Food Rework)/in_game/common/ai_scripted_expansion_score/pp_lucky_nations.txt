# Lucky nations (docs/country_finetunes.md): England, France, Muscovy, Brandenburg, the Ottomans, Portugal, Castile
# (and their successors) get their first century through without throwing themselves at stronger enemies.

# A target whose army, or whose overlord's army, is clearly larger counts a quarter.
pp_lucky_avoid_stronger = {
	attacker_potential = {
		OR = {
			tag = ENG
			tag = GBR
			tag = FRA
			tag = MOS
			tag = RUS
			tag = BRA
			tag = PRU
			tag = TUR
			tag = POR
			tag = CAS
			tag = SPA
		}
	}
	target_trigger = {
		OR = {
			army_size > {
				value = scope:attacker.army_size
				multiply = 1.25
			}
			overlord ?= {
				army_size > {
					value = scope:attacker.army_size
					multiply = 1.25
				}
			}
		}
	}
	multiplier = {
		add = -0.75
	}
}

# France consolidates its own kingdom first: every independent country ruling from the France region (Bourbon,
# Armagnac, Brittany, Foix ...); a vassal such as England's Aquitaine would mean war with its overlord.
pp_france_realm_focus = {
	attacker_potential = {
		tag = FRA
	}
	target_trigger = {
		is_subject = no
		capital ?= { region = region:france_region }
	}
	score = {
		add = 10
	}
}

# Castile/Spain plans no war on a country ruling from France before 1450 (run d6dc966a: Castile took Armagnac although
# its French land desire was -1; the war's own target province is always taken).
pp_castile_not_into_france = {
	attacker_potential = {
		OR = {
			tag = CAS
			tag = SPA
		}
	}
	target_trigger = {
		capital ?= { region = region:france_region }
	}
	never_attack = {
		current_year < 1450
	}
}

# Portugal looks across the strait once Morocco is the weaker side (the strength rule above), from 1400 (Ceuta 1415).
pp_portugal_morocco_focus = {
	attacker_potential = {
		tag = POR
		current_year >= 1400
	}
	target_trigger = {
		NOT = { is_subject_of = scope:attacker }
		capital ?= { region = region:maghreb_region }
	}
	score = {
		add = 20
	}
}

# Brandenburg prefers the small countries of its own marches: Brandenburg, Pomerania, Mecklenburg.
pp_brandenburg_marches_focus = {
	attacker_potential = {
		OR = {
			tag = BRA
			tag = PRU
		}
	}
	target_trigger = {
		NOT = { is_subject_of = scope:attacker }
		capital ?= {
			OR = {
				area = area:brandenburg_area
				area = area:pomerania_area
				area = area:mecklenburg_area
			}
		}
	}
	score = {
		add = 30
	}
}
