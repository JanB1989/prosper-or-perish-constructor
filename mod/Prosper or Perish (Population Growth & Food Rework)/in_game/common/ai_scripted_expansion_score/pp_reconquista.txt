# Reconquista aim (docs/country_finetunes.md).

# Castile/Spain and Portugal never plan a war on each other, the whole game (Jan, 2026-10-11).
pp_iberian_peace = {
	attacker_potential = {
		OR = {
			tag = CAS
			tag = SPA
			tag = POR
		}
	}
	target_trigger = {
		OR = {
			tag = CAS
			tag = SPA
			tag = POR
		}
	}
	never_attack = {
		always = yes
	}
}

# Castile/Spain prefers Muslim Iberia, and the Maghreb once it has the Maghreb preference.
pp_reconquista_focus = {
	attacker_potential = {
		OR = {
			tag = CAS
			tag = SPA
		}
	}
	target_trigger = {
		religion.group = religion_group:muslim
		capital ?= {
			OR = {
				region = region:iberia_region
				AND = {
					region = region:maghreb_region
					scope:attacker = { has_area_preference = pp_castile_conquer_maghreb }
				}
			}
		}
	}
	score = {
		add = 30
	}
}
