# Kiev west, Hungary east, Muscovy into the steppe (docs/country_finetunes.md): those countries are war targets even
# where they are not land neighbours.
pp_kiev_west_targets = {
	attacker_potential = {
		tag = KIE
		months_since_war >= 36
	}
	candidate_list = {
		every_country = {
			limit = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= { region = region:carpathia_region }
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 30
	}
}
