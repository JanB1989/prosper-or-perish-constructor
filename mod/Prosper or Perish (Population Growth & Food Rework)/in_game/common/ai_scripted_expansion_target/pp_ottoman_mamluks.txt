# Ottoman aim (docs/country_finetunes.md): once Constantinople is theirs and no rival rules from Anatolia, the countries
# ruling Egypt and the Levant (the Mamluk sultanate in 1516-17) are war targets from 1490, even where not neighbours.
pp_ottoman_mamluk_targets = {
	attacker_potential = {
		tag = TUR
		months_since_war >= 36
		current_date >= 1490.1.1
		owns = location:constantinople
		NOT = {
			any_country = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= { region = region:anatolia_region }
			}
		}
	}
	candidate_list = {
		every_country = {
			limit = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= {
					OR = {
						region = region:egypt_region
						region = region:crescent_region
					}
				}
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 30
	}
}
