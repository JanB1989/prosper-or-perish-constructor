# Reconquista aim (docs/country_finetunes.md): Castile/Spain looks at Muslim Iberia, then across the strait,
# even where the target is not a land neighbour.
pp_reconquista_targets = {
	attacker_potential = {
		OR = {
			tag = CAS
			tag = SPA
		}
		months_since_war >= 36
	}
	candidate_list = {
		every_country = {
			limit = {
				religion.group = religion_group:muslim
				capital ?= {
					OR = {
						region = region:iberia_region
						AND = {
							region = region:maghreb_region
							root = { has_area_preference = pp_castile_conquer_maghreb }
						}
					}
				}
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 30
	}
}

# The war for Granada: with the Reconquista casus belli (pp_reconquista.2), every Muslim country holding Iberian land is
# a target whatever the antagonism, after 24 months of peace.
pp_reconquista_granada_war = {
	attacker_potential = {
		OR = {
			tag = CAS
			tag = SPA
		}
		months_since_war >= 24
	}
	candidate_list = {
		every_country = {
			limit = {
				religion.group = religion_group:muslim
				NOT = { is_subject_of = root }
				any_owned_location = { region = region:iberia_region }
			}
			add_to_list = source
		}
	}
	casus_belli = casus_belli:cb_pp_reconquista
	ignore_antagonism = { always = yes }
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 60
	}
}
