# PP conquest preferences (docs/country_finetunes.md).

# Castile/Spain, once no Muslim country rules from Iberia (events/pp_reconquista_events.txt): on to the Maghreb.
pp_castile_conquer_maghreb = {
	preference_type = conquest
	modifier = 10.0

	region = maghreb_region
}

# Castile/Spain and Portugal never conquer each other (Jan, 2026-10-11), for the whole game. -1 does nothing: the engine
# stores 1 + modifier and reads 0 as "no preference" (the conquest tooltip showed no preference line at -1, and Castile
# took Portugal in run cf982ee6); -1000 makes the desire negative. Each covers the other's heartland and every area
# where the other or its subjects hold land.
pp_portugal_no_castile_conquest = {
	preference_type = conquest
	modifier = -1000

	continent = europe
	continent = africa
	continent = asia
	continent = america
	continent = oceania
	allow = {
		OR = {
			this = area:galicia_area
			this = area:leon_area
			this = area:castile_area
			this = area:extremadura_area
			this = area:toledo_area
			this = area:andalusia_area
			this = area:murcia_area
			any_location_in_area = {
				owner ?= {
					OR = {
						tag = CAS
						tag = SPA
						any_overlord_or_above = {
							OR = {
								tag = CAS
								tag = SPA
							}
						}
					}
				}
			}
		}
	}
}

# England: instead of vanilla's england_conquer_france (x15 on all of France, removed at start), only the old Angevin
# lands, and only an area where England or its subjects still hold land (the Hundred Years' War front).
pp_england_french_claims = {
	preference_type = conquest
	modifier = 4

	area = gascony_area
	area = guyenne_area
	area = poitou_area
	area = normandy_area
	area = picardy_area
	allow = {
		country_exists = c:ENG
		any_location_in_area = {
			is_ownable = yes
			owner ?= {
				OR = {
					this = c:ENG
					is_subject_or_below_of = c:ENG
				}
			}
		}
	}
}

# Castile leaves France to the French (its land goes to Iberia and the Maghreb). The desire must go below zero: the
# peace AI keeps every location that scores above zero (vanilla's -0.9 still let Castile take land), and -1 is read as
# no preference (see pp_portugal_no_castile_conquest).
pp_castile_no_france_conquest = {
	preference_type = conquest
	modifier = -2

	region = france_region
}

# France takes no Iberian land before 1450 (France and Castile both thrive; France's land comes from its own realm);

pp_france_spare_iberia = {

	modifier = -2
