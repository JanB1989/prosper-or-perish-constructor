# PP conquest preferences (docs/country_finetunes.md). Castile also gets vanilla's castile_no_portugal_conquest at start.

# Castile/Spain, once no Muslim country rules from Iberia (events/pp_reconquista_events.txt): on to the Maghreb.
pp_castile_conquer_maghreb = {
	preference_type = conquest
	modifier = 10.0

	region = maghreb_region
}

# Portugal's mirror of castile_no_portugal_conquest: leave the Castilian heartland alone.
pp_portugal_no_castile_conquest = {
	preference_type = conquest
	modifier = -0.9

	area = galicia_area
	area = leon_area
	area = castile_area
	area = extremadura_area
	area = toledo_area
	area = andalusia_area
	area = murcia_area
}

# England: instead of vanilla's england_conquer_france (x15 on all of France, removed at start), only the old Angevin
# lands, and only an area where England or its subjects still hold land (the Hundred Years' War front).
pp_england_french_claims = {
	preference_type = conquest
	modifier = 4

	area = gascony_area
	area = guyenne_area
	area = poitou_area
	area = normandy_area
	area = picardy_area
	allow = {
		country_exists = c:ENG
		any_location_in_area = {
			is_ownable = yes
			owner ?= {
				OR = {
					this = c:ENG
					is_subject_or_below_of = c:ENG
				}
			}
		}
	}
}

# Castile leaves France to the French (its land goes to Iberia and the Maghreb). -1 zeroes the desire: the peace AI

	preference_type = conquest
	modifier = -1

	region = france_region
}
