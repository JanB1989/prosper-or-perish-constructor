# Castle and Factory: the walled trading station on a harbour the expedition's country holds (Elmina, Fort Jesus,
# Fort St George, Fort Zeelandia). A fort (fort level 3, between a castle and a bastion; indestructible, because forts
# are what the AI demolishes when it saves money) and a trading house that gives the owner merchant capacity and
# presence on the coast.
pp_te_factory = {
	audio_tier = 3
	is_special = yes
	is_indestructible = yes
	max_levels = 1
	pop_type = burghers
	category = trade_category

	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	country_potential = {
		always = no
	}
	allow = {
		always = no
	}

	build_time = trade_post_build_time

	unique_production_methods = {
		pp_te_factory_maintenance = {
			icon_type = building_type
			icon = por_feitoria
			masonry = 0.5
			naval_supplies = 0.3
			category = building_maintenance
		}
	}

	modifier = {
		local_merchant_capacity = 1
		merchant_power_from_building = 2
		local_maritime_presence = 0.5
		local_defensive = 0.5
		local_garrison_size = 1
	}

	raw_modifier = {
		fort_level = 3
		propagating_zone_of_control = yes
		local_monthly_development = 0.01
		local_market_access = 0.1
	}
}
