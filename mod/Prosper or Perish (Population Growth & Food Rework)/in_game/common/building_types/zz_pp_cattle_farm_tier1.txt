ai_construct_weight = {
    value = 0
    if = {
        limit = { exists = market }
        add = {
            value = "market.market_price(goods:livestock)"
            divide = "default_price(goods:livestock)"
            multiply = { value = 1 add = modifier:local_livestock_output_modifier subtract = pp_stored_food_staple_line }
            subtract = 1
            multiply = 400
        }
    }
    # a province short of food wants farms first
    add = { value = 14 subtract = pp_location_stored_months divide = 14 min = 0 max = 1 multiply = 120 }
    # spread out: every farm level already here makes the next one less wanted
    subtract = { value = modifier:local_pp_farm_levels max = 30 multiply = 4 }
    # upgrade where the previous tier stands ([age_food] upgrade_weight): the engine scores a replacement with a
    # near-zero upgrade utility, so the weight carries it
    if = { limit = { has_building = building_type:cattle_farm } add = 200 }
    divide = { value = scope:owner.monthly_income_total add = 10 }
    # the AI builds up food in its capital province (pp_capital_food_weight, flat)
    add = pp_capital_food_weight
}
