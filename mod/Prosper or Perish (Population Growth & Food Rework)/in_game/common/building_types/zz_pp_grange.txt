# AI (2026-10-03): a new Grange only where it will pay and the food really piles up. Both terms are flat (vetoes):
# the store must stand a month above the Grange's break-even at today's victuals price (63.53 - 13.85 x price, the
# profit line at slot 2, never below 12 months; with the month: 23 at 3.0, 16 at 3.5, 13 from 3.75 up), and the
# province must gain food each month (production + consumption, which is negative). The second keeps the AI from
# building Granges everywhere in the first years, when every store starts far above its rest point and is falling.
# tests/test_store_lever.py ties the numbers.
ai_construct_weight = {
  value = 0
  if = {
    limit = { exists = market }
    subtract = { value = 64.53 subtract = { value = "market.market_price(goods:victuals)" multiply = 13.85 max = 51.53 } subtract = pp_location_stored_months min = 0 max = 1 multiply = 10000 }
  }
  if = { limit = { NOT = { exists = market } } subtract = 10000 }
  if = { limit = { pp_location_province_food_balance <= 0 } subtract = 10000 }
}

# slot 2: Surplus Sales, the store leg and the AI gate method. Store lever ([stored_food] in constructor.toml):
# Surplus Sales output +13 % per stored year above 12 months (Full Stores, which stops growing at 24 months),
# -300 % per year below (zero at 8 months), and -500 % per staffed Tavern level in the location. The gate margin
# 4.0 x output modifier / 3.83 reaches the AI threshold 1.05 at 12.5 months stored, so the AI weight's price line
# (above) decides where a new Grange goes.
# Per level at floor prices, f = years above 12 months: profit = 0.6 p - 0.08 + 4.0 x (1 + 0.13 f) - 6.153
# (2026-10-03 evening: the line was 37.88 - 5.29 p, which left stores of 14-19 months in dear-victuals markets
