tavern = {
is_foreign = no
# hefty: every further level costs another full base price (1.33 x the 0.75 [building_scaling] multiplier = 1.00)
increase_per_level_cost = 1.00
max_levels = tavern_max_level
pop_type = nobles
employment_size = 0.001
price = pp_tavern_price
category = consumer_goods_category
custom_tags = { pp_food_security_priority pp_food_distribution_priority }
icon = tavern
# 2026-10-01: the urban food answer only: towns and larger that are province capitals (one site per province).
# Rural provinces answer hunger with farms and Cookshops, which pull staple goods off the market.
rural_settlement = no
town = yes
city = yes
megalopolis = yes
location_potential = {
  is_province_capital = yes
}
# AI (2026-10-03): the Tavern is the fallback for a town whose province cannot feed itself; farms and Cookshops come
# first. Only a store near empty makes a Tavern level worth wanting. The first part is divided by the owner's
# income like the farm weight; the last two are flat, because the AI scored Taverns in hungry towns from about 25
# to millions and income-scaled terms vanish next to that (Mini World run 3, 2026-10-03: 68 -> 691 levels by 1502,
# most added in short famines, two to twelve levels at once, and kept idle once the store refilled):
# no new level once the store holds 6 months (fading in from 4), and every standing level costs 100, so a second
# or third level needs a deep shortage.
ai_construct_weight = {
  value = -60
  add = { value = 6 subtract = pp_location_stored_months divide = 6 min = 0 max = 1 multiply = 120 }
  divide = { value = scope:owner.monthly_income_total add = 10 }
  subtract = { value = pp_location_stored_months subtract = 4 divide = 2 min = 0 max = 1 multiply = 10000 }
  subtract = { value = "location_building_level(building_type:tavern)" multiply = 100 }
}
ai_forbid_shutdown = yes
important_for_AI = yes
can_close = yes
forbidden_for_estates = no
AI_ignore_available_worker_flag = yes
build_time = 365
construction_demand = victuals_trade_construction

# The Tavern serves bought-in victuals to the province: 0.8 victuals -> 24 Province Food per staffed level
# (1 victual = 30 food). The food is the Serve method's Province Food, so it follows the victuals actually bought:
# a Tavern only feeds its province as far as Granges and Victualling Yards supply the market.
# Store lever (2026-10-02, [stored_food] in constructor.toml): Province Food output runs from +75 % at an empty
# store over 0 at 12 months to -75 % at 24, so the Tavern earns 24 x 0.10 x (1 + that) and stops where the recipe
# stops paying. Per level at floor prices (dummy goods and labour 1 gold, Province Food 0.10), s = years short of
# 12 months: profit = 2.40 x (1 + 0.75 s) - 0.8 p - 0.08 - 0.91
# -> at victuals p = 2.7 it fills the store to 7 months, at 2.0 to 10.7, at 3.5 to 2.7, above 4.0 never.
# The AI builds a new Tavern where this margin reaches the threshold: below about 6 months at p = 2.7.
# No Scarcity Premium slot and no droop any more: the Serve method is the gate.
unique_production_methods = {
  pp_tavern_serve_victuals = {
    icon_type = building_type
    icon = tavern
    produced = local_food
    victuals = 0.8
    manual_labor = 0.08
    offset = 0.91
    output = 24.0
    category = building_maintenance
  }
}

raw_modifier = {
	local_population_capacity = 0.001
}

modifier = {
  # a working Tavern means the province has no surplus to sell: each staffed level takes the Surplus Sales of a
  # Grange in the same location away, so a Grange and a Tavern never pass the same food round in a circle
  # (they could in a country with high production efficiency, where both would pay at the same store)
  local_province_food_sales_output_modifier = -5.0
  # inns quartered and fed marching troops
  local_supply_limit_modifier = 0.01
}
}
