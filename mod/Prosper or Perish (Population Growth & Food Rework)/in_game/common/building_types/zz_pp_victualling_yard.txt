# Harbour sites only (pp_victuals_site_triggers.txt, generated from config/victuals_logistics.json); everywhere
# else the Grange packs the province store for the overland markets.
location_potential = {
  pp_victualling_yard_site = yes
}

# The harbour Yard is the merchants' victualling house: it packs a little of the province store and ships in
# grain from the wider market, so it needs far less of the local food than the Grange and makes twice the victuals.
# Per level: 20 food + 5.83 grain (12 food each in the farms' Provisioning) = 90 food for 3.0 victuals, which a
# Tavern turns back into 90 food: food-neutral, where the Grange loses a quarter (60 food for 1.5 victuals).
# slot 0: the province's own food, packed (0.67 victuals from the 20 food); labour or goods, whichever is cheaper.
# At default goods prices Hand Packing wins below labour 1.3, the Salting House from 1.3, Ship's Stores from 1.6.
unique_production_methods = {
  pp_victualling_yard_pack_provisions = {
    produced = victuals
    manual_labor_cost = 0.6
    output = 0.67
    category = building_maintenance
  }
  pp_victualling_yard_salting_house = {
    salt = 0.078
    produced = victuals
    manual_labor_cost = 0.3
    output = 0.67
    category = building_maintenance
  }
  pp_victualling_yard_ship_stores = {
    naval_supplies = 0.12
    tar = 0.16
    produced = victuals
    manual_labor_cost = 0.12
    output = 0.67
    category = building_maintenance
  }
}
# slot 1: grain shipments, the long-distance grain the harbour receives (already packed, so no packing slot).
unique_production_methods = {
  pp_victualling_yard_grain_shipment = {
    wheat = 5.83
    produced = victuals
    output = 2.33
    category = building_maintenance

  }
  pp_victualling_yard_millet_shipment = {
    millet = 5.83
    produced = victuals
    output = 2.33
    category = building_maintenance
  }
}
# slot 2: surplus sales, the storage leg (the farms' Surplus Sales good) at a third of the Grange's weight per level,
# matching its 20 food: output modifier = -1 + 8 per stored year = 8y, the offset input is the fixed cost. Per level
# at victuals 3.0, grain 1.0, labour 1.3 (Hand Packing): -2.7 gold at 0 months, 0 at 12, +2.6 at 24 (Grange: -26..+6).
unique_production_methods = {
  pp_victualling_yard_surplus_sales = {
    produced = province_food_sales
    offset = 5.09
    output = 0.33
    category = building_maintenance
  }
}

  local_province_food_sales_output_modifier = -0.067
