# The Victualling Yard packs the lasting part of a full province store into victuals for the goods market:
# each staffed level takes 60 food (the modifier below) and packs 1.5 victuals loose, 1:40 against the Tavern's
# 1:30, so food that goes out and comes back always loses a quarter. Packing goods lift it to at most 1:30.
# slot 0: packing the store. A real, positive victuals output (production efficiency scales it).
unique_production_methods = {
  pp_victualling_yard_pack_provisions = {
    produced = victuals
    manual_labor_cost = 0.2
    output = 1.5
    category = building_maintenance
  }
  # TEST (2026-09-26): coastal-only, more efficient alternative. A method's potential/allow run in country scope;
  # the location is scope:target (pattern from the fortress granary methods), so the coast gate sits in allow.
  pp_victualling_yard_ship_stores = {
    allow = { scope:target ?= { is_coastal = yes } }
    naval_supplies = 0.3
    produced = victuals
    manual_labor_cost = 0.2
    output = 2.25
    category = building_maintenance
  }
}
# slot 1: surplus sales, the storage leg (the farms' Surplus Sales good): output modifier = country constant -1
# + 8 per stored year = 8y; the 30.52 offset input is the fixed cost. Per level at floor prices (dummy goods and
# labour 1 gold, y = stored years, capped at 2, loose stores): profit = 1.5p - 30.72 + 2.0 x 8y
# -> at p = 2.7: -10.7 at 12 months, 0 at 20, +5.3 at 24. The Tavern stops at 12 months, so the two keep a
# 12-20 month dead band that only closes at about +125 % production efficiency.
unique_production_methods = {
  pp_victualling_yard_surplus_sales = {
    produced = province_food_sales
    offset = 30.52
    output = 2.0
    category = building_maintenance
  }
}
# slot 2: packing goods. Each breaks even at default prices; they only pay where the packing good is cheap.
