# Harbour sites (pp_victuals_site_triggers.txt, generated from config/victuals_logistics.json) and, since
# 2026-10-03, every market centre: an inland market had no buyer that turned its grain into victuals, so dear
# victuals never reached its farms (Mini World run 5, Prague and Nuremberg). Inland the cap stays small: 2 levels
# with river or sea access, 1 without, a little more with development (victualling_yard_max_level).
location_potential = {
  OR = {
    pp_victualling_yard_site = yes
    is_market_center = yes
  }
}

# The harbour Yard is the merchants' victualling house: it ships in grain, pulses and salt fish from the wider
# market and packs them into victuals. Since 2026-10-03 it takes no food from its own province (the great harbours
# are short of food themselves, and a Yard draining a hungry port made no sense): the Provisions slot that shipped
# 8 store food per level (River Barges, Merchantmen, Armed Convoy) is gone, so the Yard turns market staples into
# victuals and nothing else. Per level: 2.33 staples (12 food each in the farms' Provisioning) -> 0.932 victuals,
# which a Tavern turns back into 28 food: food-neutral, it only makes the food denser for trade. The raw_modifier's
# -5 % victuals per level keeps a big Yard from paying at base prices.
# slot 0: food shipments, the long-distance staples the harbour receives (already packed, so no packing slot): grain,
# dried pulses (ship's pease) and salt fish (market fish is already cured, the fisheries buy the salt). Every staple
# costs 1 gold and feeds 12 food in Provisioning, so all shipments share the grain numbers and stay food-neutral.
# Rice and maize need an advance, the same split as their farms: the countries of the rice and maize lands (the farms'
# native triggers) have it from the start, everyone else gets it with Rice / Maize Cultivation.

# No storage leg (2026-09-26; no store food at all since 2026-10-03): the great harbours live on imported grain and
# are short of food by nature, so the Yard's profit follows only the victuals price against the staple's. Per level
# at base prices (victuals 3, grain 1): about +0.3 gold at level 1, falling by the -5 % per level to 0 around level
# 3; at victuals 3.5 about +0.8 at level 1 and 0 near level 7. The Market Sales gate opens only above victuals 3.15
# (level 1), so the AI adds Yards when victuals are dear.

# per Yard level, staffed or not: every level ships a little less (-5 % victuals), so a big Yard never makes food and
# grows with diminishing returns (its free-building-level cost was removed 2026-09-30: supported levels come only from
# location attributes and rank)
raw_modifier = {
	local_population_capacity = 0.12
  local_victuals_output_modifier = -0.05
}

modifier = {
  local_burghers_estate_power = 0.05
  # yards provisioned and refitted the fleets
  local_repair_speed = 0.01
  local_sailors = 0.002   # engine cap: 0.0125 sailors per employed (0.2) = 0.0025, three decimals
}
}
