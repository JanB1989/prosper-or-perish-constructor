price = pp_victuals_market_import_price

icon = victuals_market_import
rural_settlement = yes

can_close = yes
forbidden_for_estates = no
AI_ignore_available_worker_flag = yes
build_time = 365
construction_demand = victuals_market_construction

# slot 0: provisioning. At floor prices (dummy goods = 1 gold, 20 % of default; labour 1 gold) it costs
# 3 victuals x p + 0.3 labour and earns 0.25 offset x 20 (country constant +19) = 5.0 gold, so this slot
# alone loses money whenever the victuals price p > 1.57: the Victualler cannot run on the price alone.
# The x20 constant keeps production efficiency small on this income (+50 % efficiency = +2.5 %).
unique_production_methods = {
  pp_province_food_from_market = {
    produced = offset
    victuals = 3.0
    manual_labor_cost = 0.3
    output = 0.25
    category = building_maintenance
  }
}
# slot 1: storage price. 0.8 province_food_purchase whose output modifier runs from 16x at an empty stockpile
# (12.8 gold) to zero at the 24-month cap (+8 more while starving), minus a fixed 3 offset input (3 gold,
# inputs ignore production efficiency).
# Per level (y = stored years, capped at 2): profit = 5.0 - 3p - 0.3 + 0.8 x (16 - 8y) - 3
# -> at p = 2.7: +6.4 gold when empty, 0 at 12 months, -6.4 at the 24-month cap.
unique_production_methods = {
  pp_province_storage_price = {
    produced = province_food_purchase
    offset = 3.0
    output = 0.8
    category = building_maintenance
  }
}

raw_modifier = {
	local_population_capacity = 0.001
}

modifier = {
  local_monthly_food = 90.0
  # droop: each staffed level lowers the storage leg a little, so staffing settles instead of flipping
  local_province_food_purchase_output_modifier = -0.3
}
}
