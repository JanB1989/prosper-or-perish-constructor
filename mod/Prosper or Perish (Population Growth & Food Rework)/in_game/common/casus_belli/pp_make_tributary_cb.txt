# Vanilla cb_make_tributary (casus_belli/make_tributary_cb.txt) with one PP change (docs/country_finetunes.md): the
# Ottomans never get it against the Mamluks, so they take Egypt instead of making it a tributary. Re-check after a
# game update.
#root is country
#scope:target is the target

REPLACE:cb_make_tributary = {
	years = 15
	create_visible = {
		OR = {
			government_type = government_type:steppe_horde
			scope:target = { government_type = government_type:tribe }
			scope:target = { government_type = government_type:steppe_horde }
			modifier:allow_tributary_subject = yes
		}
	}

	declare_enabled = {
		scope:target = {
			num_locations > 0
		}
		OR = {
			government_type = government_type:steppe_horde
			scope:target = { government_type = government_type:tribe }
			scope:target = { government_type = government_type:steppe_horde }
			modifier:allow_tributary_subject = yes
		}
		# PP: the Ottomans conquer the Mamluks, they do not make them a tributary.
		NOT = {
			AND = {
				has_or_had_tag = TUR
				scope:target = { has_or_had_tag = MAM }
			}
		}
	}

	create_enabled = {
		country_rank_level >= scope:target.country_rank_level
		# PP: the Ottomans conquer the Mamluks, they do not make them a tributary.
		NOT = {
			AND = {
				has_or_had_tag = TUR
				scope:target = { has_or_had_tag = MAM }
			}
		}
		scope:target = {
			num_locations > 0
			subject_type_is_not_locked = yes
			can_make_subject_of = {
				target = root
				type = subject_type:tributary
				ignore_war_limitation = yes	#casus_belli:cb_make_tributary disables all forms of subjugation
			}
		}
		peace_treaty_war_score_cost = {
			peace_treaty = peace_treaty:force_tributary
			loser = scope:target
			value <= 100
		}
		trigger_if = {
			limit = {
				exists = international_organization:japanese_shogunate
				scope:target = { is_member_of_international_organization = international_organization:japanese_shogunate }
				NOT = { is_member_of_international_organization = international_organization:japanese_shogunate }
			}
			scope:target = { is_leader_of_international_organization = international_organization:japanese_shogunate }
		}
	}

	ai_will_do = {
		add = {
			desc = "base value"
			value = 20
		}
		if = {
			limit = {
				scope:target = {
					any_province = {
						area = {
							root = {
								"conquer_area_preference(prev)" > 1
							}
						}
					}
				}
			}
			subtract = {
				desc = "Would rather have the land directly"
				value = 100
			}
		}
		if = {
			limit = {
				modifier:allow_tributary_subject = no
				NOT = { government_type = government_type:steppe_horde }
			}
			subtract = {
				desc = "Is not intended to make tributaries really"
				subtract = 100
			}
		}
	}

	war_goal_type = take_capital_tributary
}
