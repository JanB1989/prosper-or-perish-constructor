# Prosper or Perish: vanilla surrender_civil_war plus an end at maximum war exhaustion (pp_civil_war_exhausted in
# scripted_triggers/pp_civil_war_triggers.txt): the exhausted side may surrender regardless of size, and the AI does.
surrender_civil_war = {
	type = diplomacy
	block_when_at_war = no
	category = CATEGORY_FRIENDLY_ACTIONS
	# SpawnMessage_CivilWarEnd covers the civil-war end; the generic action popups would
	# render after annex_country swaps actor/recipient names via TakeFlagNameTag and show
	# the wrong country (CAESAR-65032).
	show_message = no
	show_message_to_target = no

	potential = {
		scope:actor = {
			in_civil_war = yes
			NOT = { has_variable = will_not_surrender }
		}
	}

	allow = {
		scope:actor = {
			"war_with_country(civil_war_opponent)" = {
				war_length >= 12
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		interaction_source_list = {
			scope:actor = {
				civil_war_opponent = {
					add_to_list = source
				}
			}
		}
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			this = scope:actor.civil_war_opponent
		}
        enabled = {
			OR = {
				total_population > {
					value = scope:actor.total_population
					multiply = 2
				}
				custom_tooltip = {
					text = pp_civil_war_exhausted_tt
					scope:actor = { pp_civil_war_exhausted = yes }
				}
			}
		}
	}

	ai_will_do = {
		#just surrender if it's hopeless
		add = {
			if = {
				limit = {
					scope:actor = {
						total_population < {
							value = scope:recipient.total_population
							add = total_population
							multiply = 0.1
						}
					}
				}
				value = 1000
			}
		}

		add = {
			if = {
				limit = {
					scope:actor = {
						any_current_war = {
							scope:actor = {
								war_enthusiasm = { war = prev value <= 0 }
							}
							OR = {
								attacker_leader = scope:recipient
								defender_leader = scope:recipient
							}
						}
					}
				}
				value = 1000
			}
		}

		# Prosper or Perish: surrender at maximum war exhaustion
		add = {
			if = {
				limit = {
					scope:actor = { pp_civil_war_exhausted = yes }
				}
				value = 1000
			}
		}
	}

	effect = {
        scope:recipient = {
			annex_country = {
				country = scope:actor
				reason = CivilWar
			}
		}
	}
}
