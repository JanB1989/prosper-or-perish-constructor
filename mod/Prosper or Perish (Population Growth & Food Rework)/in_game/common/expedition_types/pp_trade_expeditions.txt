# PP trade expeditions (docs/trade_expeditions.md). Three expensive, gated voyages any colonial power can send (AI and
# player alike) for a station on the coasts of Africa, India and the Far East. Each picks the best free historical
# site when it sails; on arrival the host decides whether the station is bought, licensed, fought for or refused
# (scripted_effects/pp_trade_expedition_effects.txt, pp_te_arrive_effect). Outcome events: events/
# pp_trade_expedition_events.txt.

# 1. Castle on the African Coast (Arguin 1445, Elmina 1482, Christiansborg 1661, Fort Jesus 1593).
pp_te_africa = {
	unique = yes
	repeatable = yes
	travel_speed = 1
	dynamic_first_waypoint = yes
	origin = none
	travel_mode = sea
	category = commercial
	show_end_message = no

	potential = {
		has_advance = colonies
		has_ports = yes
		is_subject = no
	}

	can_start = {
		current_age_or_later = { age = age_3_discovery }
		pp_te_colonial_power = yes
		NOT = { capital ?= { continent = continent:africa } }
		pp_te_years_since_africa >= 15
		OR = {
			NOT = { has_variable = pp_te_n_africa }
			var:pp_te_n_africa < 2.5
		}
		gold >= pp_te_reserve_africa
		pp_te_free_to_sail = yes
		pp_te_any_africa_site = yes
	}

	leader = { pp_te_leader = yes }

	ai_leader_source_list = {
		every_character = {
			limit = { pp_te_leader = yes }
			add_to_list = source
		}
	}

	on_start = {
		pp_te_start_effect = { KIND = africa }
	}

	on_arrive_to_new_location = {
		discover_area = scope:location.area
	}

	on_end = {
		pp_te_log_effect = { STAGE = end KIND = africa }
		pp_te_arrive_effect = { KIND = africa CEDE = 1 WAR = 25 LICENCE = 60 }
	}

	on_fail = {
		pp_te_log_effect = { STAGE = fail KIND = africa }
		pp_te_fail_effect = { KIND = africa }
	}

	ai_chance_to_check = { value = pp_te_ai_check_chance }

	utility = { value = pp_te_utility_base }

	leader_utility = {
		value = 1
		if = {
			limit = { is_explorer = yes }
			add = 5
		}
		if = {
			limit = { is_admiral = yes }
			add = 3
		}
	}
}
