# PP copy of vanilla create_province_subject (2026-10-12, docs/peace_deal_rulebook.md "Enclaves").
# Vanilla never lets the AI use this action (ai_tick = never). PP lets an AI nation that is broken into 4+ pieces on
# its home continent release a small landlocked enclave as a tributary or vassal (the whole enclave as one subject):
# land not connected to the capital by land or strait, no location of it on the sea coast (river channels and lakes do
# not count), at most about 6 locations (a bigger piece is a second part of the country). A nation with fewer pieces keeps its exclaves.
# Footholds reached by sea keep their owner. Not for HRE members (Imperial gore stays), subjects, or countries at war
# (a war may still reconnect the land).
# The player's use of the action is unchanged.
create_province_subject = {
	type = owncountry


	allow = {
		scope:actor = {
			or = {
				is_subject = no
				subject_type = {
					allow_subject_creation = yes
				}
			}
		}
	}

	ai_prerequisite = {
		is_subject = no
		at_war = no
		num_locations > 1
		trigger_if = {
			limit = { exists = international_organization:hre }
			NOT = { is_member_of_international_organization = international_organization:hre }
		}
	}

	select_trigger = {
		looking_for_a = province
		interaction_source_list = {
			scope:actor = {
				every_province = {
					add_to_list = source
				}
			}
		}
		# AI: only nations broken into 4+ pieces on their home continent, and only provinces whose land of the actor is a
		# small enclave without sea coast there. One flood from the capital, then at most 3 more floods to count the pieces.
		ai_interaction_source_list = {
			scope:actor = {
				if = {
					limit = { exists = capital }
					capital = {
						add_to_temporary_list = pp_connected_to_capital
						add_to_temporary_list = pp_counted
						every_connected_location = { add_to_temporary_list = pp_connected_to_capital }
						continent = { save_scope_as = pp_home_continent }
					}
					set_local_variable = { name = pp_pieces value = 1 }
					while = {
						limit = {
							local_var:pp_pieces < 4
							any_owned_location = {
								NOT = { is_in_list = pp_connected_to_capital }
								NOT = { is_in_list = pp_counted }
								continent ?= { this = scope:pp_home_continent }
							}
						}
						random_owned_location = {
							limit = {
								NOT = { is_in_list = pp_connected_to_capital }
								NOT = { is_in_list = pp_counted }
								continent ?= { this = scope:pp_home_continent }
							}
							add_to_temporary_list = pp_counted
							every_connected_location = { add_to_temporary_list = pp_counted }
						}
						change_local_variable = { name = pp_pieces add = 1 }
					}
					if = {
						limit = { local_var:pp_pieces > 3.5 }
						every_province = {
							limit = {
								any_location_in_province = {
									owner = scope:actor
									continent ?= { this = scope:pp_home_continent }
								}
								NOT = {
									any_location_in_province = {
										owner = scope:actor
										OR = {
											is_in_list = pp_connected_to_capital
											# sea footholds stay; river channels and lakes do not count as sea
											pp_is_sea_coast = yes
											any_connected_location = { pp_is_sea_coast = yes }
											# a bigger landlocked piece is a second part of the country, not a splinter
											any_connected_location = { count >= 6 }
										}
									}
								}
							}
							add_to_list = source
						}
					}
				}
			}
		}
		target_flag = target_province
		name = "create_province_subject_select_province"
		none_available_msg_key = "create_province_subject_no_province"
		column = {
			data = name
		}
		visible = {
		}
	}

	select_trigger = {
		looking_for_a = subject_type
		target_flag = subject_type
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		name = "create_building_subject_select_subject_type"
		none_available_msg_key = "create_building_subject_no_subject_type"
		column = {
			data = subject_type_attribute_column
		}
		column = { data = subject_type_icon }
		visible = {
			trigger_if = { # PP: the AI releases enclaves as tributaries where it may have them, else as vassals
				limit = { scope:actor = { is_ai = yes } }
				OR = {
					root = subject_type:vassal
					AND = {
						root = subject_type:tributary
						# checked here first: vanilla's tributary check errors on a situation missing from the save
						scope:actor = {
							OR = {
								government_type = government_type:steppe_horde
								government_type = government_type:tribe
								modifier:allow_tributary_subject = yes
							}
						}
					}
				}
			}
			subject_can_be_created_by = {
				country = scope:actor
				where = scope:target_province
			}
			scope:target_province = {
				NOT = {
					any_location_in_province = { integration_level = colonized }
				}
			}
		}
		enabled = {
			trigger_if = {
				limit = {
					root != subject_type:colonial_nation
				}
				scope:actor = {
					custom_tooltip = {
						text = create_province_subject_union_tt
						trigger_if = {
							limit = { exists = union }
							union ?= {
								OR = {
									NOT = { has_active_senior_partner = yes }
									country_is_senior_partner = { country = scope:actor }
								}
							}
						}
					}
				}
			}
		}
		map_color = {
			if = {
				limit = {
					is_land = yes
					province ?= scope:target_province
				}
				value = define:NMapColors|MAP_COLOR_HIGH
			}
		}
		only_color_selectable = no
	}

	select_trigger = {
		show_if = {
			scope:subject_type ?= {
				has_overlords_ruler = no
			}
		}
		looking_for_a = character
		source = actor
		target_flag = ruler
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		allow_null = yes #so we let the game create a random ruler
		name = "create_building_subject_select_ruler"
		none_available_msg_key = "create_building_subject_no_ruler"
		column = {
			data = character_info
			widget = character_info_for_select_ruler
		}
		visible = {
			scope:actor = { is_ai = no } # PP: the AI lets the game create a random ruler
			can_become_a_ruler = yes
			scope:actor = {
				trigger_if = {
					limit = {
						exists = ruler
					}
					ruler != root
				}
			}
		}
		enabled = {
			NOT = { modifier:is_head_of_cabinet = yes }
			is_eligible_heir = scope:actor #the new subject is going to inherit all its settings from the actor, so one rule for all
		}
		map_color = {
			if = {
				limit = {
					province = scope:target_province
				}
				value = define:NMapColors|MAP_COLOR_HIGH
			}
		}
	}

	effect = {
		# PP: the AI releases the whole landlocked enclave as one subject. Note its piece before the release (another
		# target of the same tick from the same piece finds no land of the actor left in its province and is skipped).
		if = {
			limit = {
				scope:actor = { is_ai = yes }
				exists = scope:target_province
			}
			scope:target_province = {
				random_location_in_province = {
					limit = { owner = scope:actor }
					add_to_temporary_list = pp_enclave_piece
					every_connected_location = { add_to_temporary_list = pp_enclave_piece }
				}
			}
		}
		if = {
			limit = {
				exists = scope:target_province
				exists = scope:ruler
				scope:target_province = { any_location_in_province = { owner = scope:actor } }
			}

			scope:target_province = {
				create_location_country_from_province = {
					ruler = scope:ruler
					subject_type = scope:subject_type

					capital = {
						province = {
							save_scope_as = new_province
						}
					}
					add_truce_with = {
						target = scope:actor
						mutual = yes
					}
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:target_province
				scope:target_province = { any_location_in_province = { owner = scope:actor } }
			}

			if = {
				limit = {
					scope:subject_type = {
						has_overlords_ruler = yes
					}
				}
				#using overlord's ruler
				scope:actor = {
					ruler = {
						save_scope_as = ruler
					}
				}
			}

			scope:target_province = {
				create_location_country_from_province = {
					ruler = scope:ruler
					subject_type = scope:subject_type

					capital = {
						province = {
							save_scope_as = new_province
						}
					}

					save_scope_as = new_country

					add_truce_with = {
						target = scope:actor
						mutual = yes
					}
				}
			}

			scope:new_country = {
				ruler = {
					save_scope_as = ruler
				}
			}
			# PP: the rest of the enclave goes to the same subject
			if = {
				limit = { scope:actor = { is_ai = yes } }
				every_in_list = {
					list = pp_enclave_piece
					limit = { owner = scope:actor }
					change_location_owner = scope:new_country
				}
			}
		}
	}

	# PP: vanilla had ai_tick = never and ai_will_do = 0. The province list above already holds only small landlocked
	# enclaves. A march cannot be created from a province (no creation_visible), a fiefdom is meant for annexation, so
	# the AI picks a tributary where its government allows one, else a vassal. The AI annexes only subjects it borders
	# with enough control (AI_ANNEX_SUBJECT_BORDERING_CONTROL_NEEDED), so an enclave vassal stays unless the overlord
	# reconnects to it.
	ai_tick = monthly
	ai_tick_frequency = 12
	maximum_targets_in_one_tick = 5
	disallowed_duplicates_of_targets_for_ai = { target_province }
	ai_will_do = {
		value = 0
		if = {
			limit = { scope:subject_type ?= subject_type:tributary }
			add = 100
		}
		else_if = {
			limit = { scope:subject_type ?= subject_type:vassal }
			add = 50
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
}
