# PP copy of vanilla create_province_subject (2026-10-12, docs/peace_deal_rulebook.md "Enclaves").
# Vanilla never lets the AI use this action (ai_tick = never). PP lets an AI country release a province as a march
# (or a tributary where only that is possible) when all of its land in that province is a landlocked enclave: not
# connected to the capital by land or strait, and no location of that enclave on a coast or navigable river.
# Footholds reached by sea keep their owner. Not for HRE members (Imperial gore stays), subjects, or countries at war
# (a war may still reconnect the land). Marches and tributaries cannot be annexed, so the land does not come back.
# The player's use of the action is unchanged.
create_province_subject = {
	type = owncountry


	allow = {
		scope:actor = {
			or = {
				is_subject = no
				subject_type = {
					allow_subject_creation = yes
				}
			}
		}
	}

	ai_prerequisite = {
		is_subject = no
		at_war = no
		num_locations > 1
		trigger_if = {
			limit = { exists = international_organization:hre }
			NOT = { is_member_of_international_organization = international_organization:hre }
		}
	}

	select_trigger = {
		looking_for_a = province
		interaction_source_list = {
			scope:actor = {
				every_province = {
					add_to_list = source
				}
			}
		}
		# AI: only provinces whose land of the actor is a landlocked enclave. One flood from the capital per check.
		ai_interaction_source_list = {
			scope:actor = {
				capital ?= {
					add_to_temporary_list = pp_connected_to_capital
					every_connected_location = {
						add_to_temporary_list = pp_connected_to_capital
					}
				}
				every_province = {
					limit = {
						NOT = {
							any_location_in_province = {
								owner = scope:actor
								OR = {
									is_in_list = pp_connected_to_capital
									is_coastal = yes
									any_connected_location = { is_coastal = yes }
								}
							}
						}
					}
					add_to_list = source
				}
			}
		}
		target_flag = target_province
		name = "create_province_subject_select_province"
		none_available_msg_key = "create_province_subject_no_province"
		column = {
			data = name
		}
		visible = {
		}
	}

	select_trigger = {
		looking_for_a = subject_type
		target_flag = subject_type
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		name = "create_building_subject_select_subject_type"
		none_available_msg_key = "create_building_subject_no_subject_type"
		column = {
			data = subject_type_attribute_column
		}
		column = { data = subject_type_icon }
		visible = {
			subject_can_be_created_by = {
				country = scope:actor
				where = scope:target_province
			}
			scope:target_province = {
				NOT = {
					any_location_in_province = { integration_level = colonized }
				}
			}
		}
		enabled = {
			trigger_if = {
				limit = {
					root != subject_type:colonial_nation
				}
				scope:actor = {
					custom_tooltip = {
						text = create_province_subject_union_tt
						trigger_if = {
							limit = { exists = union }
							union ?= {
								OR = {
									NOT = { has_active_senior_partner = yes }
									country_is_senior_partner = { country = scope:actor }
								}
							}
						}
					}
				}
			}
		}
		map_color = {
			if = {
				limit = {
					is_land = yes
					province ?= scope:target_province
				}
				value = define:NMapColors|MAP_COLOR_HIGH
			}
		}
		only_color_selectable = no
	}

	select_trigger = {
		show_if = {
			scope:subject_type ?= {
				has_overlords_ruler = no
			}
		}
		looking_for_a = character
		source = actor
		target_flag = ruler
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		allow_null = yes #so we let the game create a random ruler
		name = "create_building_subject_select_ruler"
		none_available_msg_key = "create_building_subject_no_ruler"
		column = {
			data = character_info
			widget = character_info_for_select_ruler
		}
		visible = {
			scope:actor = { is_ai = no } # PP: the AI lets the game create a random ruler
			can_become_a_ruler = yes
			scope:actor = {
				trigger_if = {
					limit = {
						exists = ruler
					}
					ruler != root
				}
			}
		}
		enabled = {
			NOT = { modifier:is_head_of_cabinet = yes }
			is_eligible_heir = scope:actor #the new subject is going to inherit all its settings from the actor, so one rule for all
		}
		map_color = {
			if = {
				limit = {
					province = scope:target_province
				}
				value = define:NMapColors|MAP_COLOR_HIGH
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target_province
				exists = scope:ruler
			}

			scope:target_province = {
				create_location_country_from_province = {
					ruler = scope:ruler
					subject_type = scope:subject_type

					capital = {
						province = {
							save_scope_as = new_province
						}
					}
					add_truce_with = {
						target = scope:actor
						mutual = yes
					}
				}
			}
		}
		else_if = {
			limit = {
				exists = scope:target_province
			}

			if = {
				limit = {
					scope:subject_type = {
						has_overlords_ruler = yes
					}
				}
				#using overlord's ruler
				scope:actor = {
					ruler = {
						save_scope_as = ruler
					}
				}
			}

			scope:target_province = {
				create_location_country_from_province = {
					ruler = scope:ruler
					subject_type = scope:subject_type

					capital = {
						province = {
							save_scope_as = new_province
						}
					}

					save_scope_as = new_country

					add_truce_with = {
						target = scope:actor
						mutual = yes
					}
				}
			}

			scope:new_country = {
				ruler = {
					save_scope_as = ruler
				}
			}
		}
	}

	# PP: vanilla had ai_tick = never and ai_will_do = 0. The province list above already holds only landlocked
	# enclaves; the AI picks a march, or a tributary where a march is not possible. Never a vassal or fiefdom:
	# those can be annexed and the enclave would come back.
	ai_tick = monthly
	ai_tick_frequency = 12
	maximum_targets_in_one_tick = 5
	disallowed_duplicates_of_targets_for_ai = { target_province }
	ai_will_do = {
		value = 0
		if = {
			limit = { scope:subject_type ?= subject_type:march }
			add = 100
		}
		else_if = {
			limit = { scope:subject_type ?= subject_type:tributary }
			add = 50
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
}
