# PP copy of vanilla create_province_subject (2026-10-12, docs/peace_deal_rulebook.md "Enclaves").
# Vanilla never lets the AI use this action (ai_tick = never). PP lets every AI nation let go, once a year, each piece
# of its land it can never govern (pp_enclave_kept_location): cut off from the capital by land and strait, max_control
# under 0.5 everywhere. Kept: footholds reached by sea that reach outside the capital's region (river channels and
# lakes are not sea), colonies being settled, pieces held by an enemy, HRE members; no release during a civil war or by a rebel state (pp_enclave_release_paused). A piece that borders one of the
# nation's subjects joins that subject; any other becomes a new subject (a tributary where the government allows one,
# else a vassal; a samanta in India). The target is a province, but the AI releases by location
# (create_country_from_location), since a piece often shares its province with kept land of the nation. Subjects and
# union juniors cannot create subjects (pp_cannot_create_subjects); their pieces go through
# pp_release_enclave_independence. The player's action is unchanged.
create_province_subject = {
	type = owncountry


	allow = {
		scope:actor = {
			or = {
				is_subject = no
				subject_type = {
					allow_subject_creation = yes
				}
			}
		}
	}

	ai_prerequisite = {
		pp_cannot_create_subjects = no
		pp_enclave_release_paused = no
		num_locations > 1
		trigger_if = {
			limit = { exists = international_organization:hre }
			NOT = { is_member_of_international_organization = international_organization:hre }
		}
	}

	select_trigger = {
		looking_for_a = province
		interaction_source_list = {
			scope:actor = {
				every_province = {
					add_to_list = source
				}
			}
		}
		# AI: only provinces that hold land of a piece the actor lets go (pp_enclave_kept_location).
		ai_interaction_source_list = {
			scope:actor = { pp_enclave_list_provinces = yes }
		}
		target_flag = target_province
		name = "create_province_subject_select_province"
		none_available_msg_key = "create_province_subject_no_province"
		column = {
			data = name
		}
		visible = {
		}
	}

	select_trigger = {
		looking_for_a = subject_type
		target_flag = subject_type
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		name = "create_building_subject_select_subject_type"
		none_available_msg_key = "create_building_subject_no_subject_type"
		column = {
			data = subject_type_attribute_column
		}
		column = { data = subject_type_icon }
		visible = {
			# PP: the AI releases a piece by location (create_country_from_location), so the engine's province check
			# does not apply to it. It would also block every province: this list is cached from the first one, and a
			# province that also holds the capital fails it. The AI's type follows its government, as in
			# pp_enclave_release_piece: a tributary where allowed, else a vassal (a samanta where samanta_advance hides
			# the vassal).
			trigger_if = {
				limit = { scope:actor = { is_ai = yes } }
				OR = {
					AND = {
						root = subject_type:tributary
						scope:actor = { pp_can_hold_tributary = yes }
					}
					AND = {
						root = subject_type:vassal
						scope:actor = {
							pp_can_hold_tributary = no
							NOT = { has_advance = samanta_advance }
						}
					}
					AND = {
						root = subject_type:samanta
						scope:actor = {
							pp_can_hold_tributary = no
							has_advance = samanta_advance
						}
					}
				}
			}
			trigger_else = {
				subject_can_be_created_by = {
					country = scope:actor
					where = scope:target_province
				}
				scope:target_province = {
					NOT = {
						any_location_in_province = { integration_level = colonized }
					}
				}
			}
		}
		enabled = {
			trigger_if = {
				limit = {
					root != subject_type:colonial_nation
				}
				scope:actor = {
					custom_tooltip = {
						text = create_province_subject_union_tt
						trigger_if = {
							limit = { exists = union }
							union ?= {
								OR = {
									NOT = { has_active_senior_partner = yes }
									country_is_senior_partner = { country = scope:actor }
								}
							}
						}
					}
				}
			}
		}
		map_color = {
			if = {
				limit = {
					is_land = yes
					province ?= scope:target_province
				}
				value = define:NMapColors|MAP_COLOR_HIGH
			}
		}
		only_color_selectable = no
	}

	select_trigger = {
		show_if = {
			scope:subject_type ?= {
				has_overlords_ruler = no
			}
		}
		looking_for_a = character
		source = actor
		target_flag = ruler
		cache_targets = yes #this list of targets doesn't rely on the previous target so we only need to do it once
		allow_null = yes #so we let the game create a random ruler
		name = "create_building_subject_select_ruler"
		none_available_msg_key = "create_building_subject_no_ruler"
		column = {
			data = character_info
			widget = character_info_for_select_ruler
		}
		visible = {
			scope:actor = { is_ai = no } # PP: the AI lets the game create a random ruler
			can_become_a_ruler = yes
			scope:actor = {
				trigger_if = {
					limit = {
						exists = ruler
					}
					ruler != root
				}
			}
		}
		enabled = {
			NOT = { modifier:is_head_of_cabinet = yes }
			is_eligible_heir = scope:actor #the new subject is going to inherit all its settings from the actor, so one rule for all
		}
		map_color = {
			if = {
				limit = {
					province = scope:target_province
				}
				value = define:NMapColors|MAP_COLOR_HIGH
			}
		}
	}

	effect = {
		# PP: the AI lets one whole piece in the target province go (pp_enclave_release_piece); another target of the same
		# tick from the same piece finds nothing left and is skipped.
		if = {
			limit = {
				scope:actor = { is_ai = yes }
				exists = scope:target_province
			}
			scope:actor = { save_scope_as = pp_enclave_overlord }
			scope:target_province = { pp_enclave_pick_piece = yes }
			scope:actor = { pp_enclave_release_piece = yes }
		}
		else = {
			if = {
				limit = {
					exists = scope:target_province
					exists = scope:ruler
					scope:target_province = { any_location_in_province = { owner = scope:actor } }
				}

				scope:target_province = {
					create_location_country_from_province = {
						ruler = scope:ruler
						subject_type = scope:subject_type

						capital = {
							province = {
								save_scope_as = new_province
							}
						}
						add_truce_with = {
							target = scope:actor
							mutual = yes
						}
					}
				}
			}
			else_if = {
				limit = {
					exists = scope:target_province
					scope:target_province = { any_location_in_province = { owner = scope:actor } }
				}

				if = {
					limit = {
						scope:subject_type = {
							has_overlords_ruler = yes
						}
					}
					#using overlord's ruler
					scope:actor = {
						ruler = {
							save_scope_as = ruler
						}
					}
				}

				scope:target_province = {
					create_location_country_from_province = {
						ruler = scope:ruler
						subject_type = scope:subject_type

						capital = {
							province = {
								save_scope_as = new_province
							}
						}

						save_scope_as = new_country

						add_truce_with = {
							target = scope:actor
							mutual = yes
						}
					}
				}

				scope:new_country = {
					ruler = {
						save_scope_as = ruler
					}
				}
			}
		}
	}

	# PP: vanilla had ai_tick = never and ai_will_do = 0. The province list above holds only pieces the nation lets go.
	ai_tick = monthly
	ai_tick_frequency = 12
	maximum_targets_in_one_tick = 5
	disallowed_duplicates_of_targets_for_ai = { target_province }
	ai_will_do = {
		add = 100
	}

	automation_tick = never
	automation_tick_frequency = 12
}
