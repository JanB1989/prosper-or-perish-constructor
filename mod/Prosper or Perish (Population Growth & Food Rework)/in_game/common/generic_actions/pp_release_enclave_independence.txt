# PP (2026-10-12, docs/peace_deal_rulebook.md "Enclaves"): a subject or union junior cannot create subjects
# (pp_cannot_create_subjects), so its pieces go through this action, by the same rule as create_province_subject
# (pp_enclave_kept_location: cut off from the capital, max_control under 0.5 everywhere). A subject's piece joins a
# subject of its overlord that it borders, else becomes a new subject of that overlord; a union junior's piece becomes
# independent. AI only; not offered to players.
pp_release_enclave_independence = {
	type = owncountry
	show_in_gui_list = no
	show_message = no

	potential = {
		scope:actor = {
			is_ai = yes
			pp_cannot_create_subjects = yes
		}
	}

	allow = {
	}

	ai_prerequisite = {
		pp_cannot_create_subjects = yes
		pp_enclave_release_paused = no
		num_locations > 1
		trigger_if = {
			limit = { exists = international_organization:hre }
			NOT = { is_member_of_international_organization = international_organization:hre }
		}
	}

	select_trigger = {
		looking_for_a = province
		interaction_source_list = {
			scope:actor = { pp_enclave_list_provinces = yes }
		}
		target_flag = target_province
		name = "create_province_subject_select_province"
		none_available_msg_key = "create_province_subject_no_province"
		column = {
			data = name
		}
		visible = {
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target_province }
			if = {
				limit = { scope:actor = { is_subject = yes } }
				scope:actor = { overlord = { save_scope_as = pp_enclave_overlord } }
			}
			scope:target_province = { pp_enclave_release_piece = yes }
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 12
	maximum_targets_in_one_tick = 5
	disallowed_duplicates_of_targets_for_ai = { target_province }
	ai_will_do = {
		add = 100
	}

	automation_tick = never
	automation_tick_frequency = 12
}
