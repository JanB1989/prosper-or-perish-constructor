# Castile/Spain and Portugal never fight each other (docs/country_finetunes.md, Jan 2026-10-11): neither joins a war as
# an ally when the other leads the side it would fight. AI only: a player may join.
pp_iberian_peace_war_rule = {
	join_war_disabled_trigger = {
		is_ai = yes
		OR = {
			AND = {
				OR = {
					tag = CAS
					tag = SPA
				}
				scope:second_leader ?= { tag = POR }
			}
			AND = {
				tag = POR
				scope:second_leader ?= {
					OR = {
						tag = CAS
						tag = SPA
					}
				}
			}
		}
	}
}
