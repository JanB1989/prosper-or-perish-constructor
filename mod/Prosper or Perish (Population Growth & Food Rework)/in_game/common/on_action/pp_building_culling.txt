pp_raise_owned_zero_rgo_max_workers = {
	effect = {
		every_owned_location = {
			limit = {
				exists = raw_material
				rgo_level >= 0
				NOT = { rgo_level >= 1 }
				OR = {
					"unemployed_pops_of_pop_type_in_location(pop_type:laborers)" > 0
					"unemployed_pops_of_pop_type_in_location(pop_type:slaves)" > 0
				}
			}
			change_max_raw_material_workers = 1
		}
	}
}

# One random closed (is_opened = no) building level per owned location, yearly, AI countries only.
# Import victuals markets are never culled.
pp_yearly_cull_one_closed_building = {
	effect = {
		if = {
			limit = {
				is_ai = yes
			}
			every_owned_location = {
				limit = {
					any_buildings_in_location = {
						is_opened = no
						owner = root
						building_can_be_destroyed_by = root
						NOT = { building_type = building_type:victuals_market_import }
					}
				}
				random_buildings_in_location = {
					limit = {
						is_opened = no
						owner = root
						building_can_be_destroyed_by = root
						NOT = { building_type = building_type:victuals_market_import }
					}
					change_building_level = -1
				}
			}
		}
	}
}

	}
}
