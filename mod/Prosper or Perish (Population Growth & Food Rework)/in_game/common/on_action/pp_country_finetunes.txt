# Country finetunes at game start (docs/country_finetunes.md): one-off starting gold, starting opinions, conquest
# preferences.
on_game_start = {
	on_actions = {
		pp_country_finetune_start_gold
		pp_country_finetune_opinions
		pp_country_finetune_area_preferences
		delay = { days = 1 }
		pp_colonizer_area_preferences
	}
}

# Vanilla's setup_area_preferences re-adds avoid_african_colonization at every new age; the day's delay runs this
# after it.
on_new_age_global = {
	on_actions = {
		delay = { days = 1 }
		pp_colonizer_area_preferences
	}
}

# Colonizers daisy-chain down Africa (area_preferences/pp_colonizer_preferences.txt): vanilla's avoid-all-Africa
# (which also covers the sea lanes) becomes PP's land-only list without the stations; each nation gets its historical
# sea route and island/Cape stations one age at a time. Europe's empty land stays off limits to them until the Age of
# Revolutions.
pp_colonizer_area_preferences = {
	effect = {
		every_country = {
			limit = {
				is_ai = yes
				OR = {
					tag = POR
					tag = CAS
					tag = SPA
					tag = ENG
					tag = GBR
					tag = FRA
					tag = HOL
					tag = NED
				}
			}
			if = {
				limit = { has_area_preference = avoid_african_colonization }
				remove_area_preference = avoid_african_colonization
			}
			pp_add_area_preference_once = { preference = pp_colonizer_avoid_africa }
			if = {
				limit = {
					NOT = { current_age_or_later = { age = age_6_revolutions } }
				}
				pp_add_area_preference_once = { preference = pp_colonizer_avoid_europe }
			}
			else_if = {
				limit = { has_area_preference = pp_colonizer_avoid_europe }
				remove_area_preference = pp_colonizer_avoid_europe
			}
			# Portugal: Madeira and the Azores, then Cape Verde, St Helena and the Cape route to India.
			if = {
				limit = {
					tag = POR
					current_age_or_later = { age = age_2_renaissance }
				}
				pp_add_area_preference_once = { preference = pp_route_west_africa }
				pp_add_area_preference_once = { preference = pp_station_madeira_azores }
			}
			if = {
				limit = {
					tag = POR
					current_age_or_later = { age = age_3_discovery }
				}
				pp_add_area_preference_once = { preference = pp_route_cape_to_india }
				pp_add_area_preference_once = { preference = pp_station_cape_verde }
				pp_add_area_preference_once = { preference = pp_station_st_helena }
			}
			# Castile/Spain: the Canaries only (its way to Asia runs over the Americas and the Pacific).
			if = {
				limit = {
					OR = {
						tag = CAS
						tag = SPA
					}
					current_age_or_later = { age = age_2_renaissance }
				}
				pp_add_area_preference_once = { preference = pp_station_canaries }
			}
			# England, France, Holland: the sea route to India from the Age of Reformation.
			if = {
				limit = {
					NOR = {
						tag = POR
						tag = CAS
						tag = SPA
					}
					current_age_or_later = { age = age_4_reformation }
				}
				pp_add_area_preference_once = { preference = pp_route_west_africa }
				pp_add_area_preference_once = { preference = pp_route_cape_to_india }
			}
			# England/Great Britain: St Helena, then the Cape.
			if = {
				limit = {
					OR = {
						tag = ENG
						tag = GBR
					}
					current_age_or_later = { age = age_4_reformation }
				}
				pp_add_area_preference_once = { preference = pp_station_st_helena }
			}
			if = {
				limit = {
					OR = {
						tag = ENG
						tag = GBR
					}
					current_age_or_later = { age = age_6_revolutions }
				}
				pp_add_area_preference_once = { preference = pp_station_cape }
			}
			# France: the Mascarenes.
			if = {
				limit = {
					tag = FRA
					current_age_or_later = { age = age_5_absolutism }
				}
				pp_add_area_preference_once = { preference = pp_station_mascarenes }
			}
			# Holland/the Netherlands: the Cape and Mauritius.
			if = {
				limit = {
					OR = {
						tag = HOL
						tag = NED
					}
					current_age_or_later = { age = age_5_absolutism }
				}
				pp_add_area_preference_once = { preference = pp_station_cape }
				pp_add_area_preference_once = { preference = pp_station_mascarenes }
			}
		}
	}
}

# Castile/Spain <-> Portugal: never conquer each other, the whole game (pp_conquest_preferences.txt; vanilla's Rio Salado
# event adds castile_no_portugal_conquest, which PP replaces with the same -1000); never plan a war on each other
# (ai_scripted_expansion_score pp_reconquista.txt). Muscovy: the Russian region (pp_gathering_russian_lands.txt).
pp_country_finetune_area_preferences = {
	effect = {
		c:POR ?= {
			add_area_preference = pp_portugal_no_castile_conquest
		}
		c:MOS ?= {
			add_area_preference = pp_muscovy_gather_russian_lands
			add_area_preference = pp_muscovy_conquer_steppe
		}
		# Kiev and Hungary face each other (ai_scripted_expansion_score/target pp_kiev_hungary.txt).
		c:KIE ?= {
			add_area_preference = pp_kiev_go_west
		}
		c:HUN ?= {
			add_area_preference = pp_hungary_go_east
		}
		# England: the Angevin lands where it still holds land, not all of France (pp_conquest_preferences.txt).
		c:ENG ?= {
			if = {
				limit = { has_area_preference = england_conquer_france }
				remove_area_preference = england_conquer_france
			}
			add_area_preference = pp_england_french_claims
			# vanilla's Ireland preference from the start (vanilla: from 1450)
			if = {
				limit = { NOT = { has_area_preference = england_conquer_ireland } }
				add_area_preference = england_conquer_ireland
			}
		}
		# France: its own kingdom first.
		c:FRA ?= {
			add_area_preference = pp_france_consolidate_realm
			add_area_preference = pp_france_spare_iberia
		}
		# Castile: no French land, no Portuguese land.
		c:CAS ?= {
			add_area_preference = pp_castile_no_france_conquest
			add_area_preference = pp_castile_spare_portugal
		}
		# Brandenburg: its marches and the Prussian lands (pp_lucky_nations.txt).
		c:BRA ?= {
			add_area_preference = pp_brandenburg_marches
			add_area_preference = pp_brandenburg_prussia
		}
		# Serbia and Bulgaria look north to Hungary (pp_balkans_north.txt).
		c:SER ?= {
			add_area_preference = pp_balkans_go_north
		}
		c:BUL ?= {
			add_area_preference = pp_balkans_go_north
		}
	}
}

# Castile <-> Portugal: Iberian Concord (+200, 100 years) replaces vanilla's decaying +50 good relations.
pp_country_finetune_opinions = {
	effect = {
		if = {
			limit = {
				exists = c:CAS
				exists = c:POR
			}
			c:CAS = {
				if = {
					limit = { has_opinion = { target = c:POR modifier = opinion_good_relations } }
					remove_opinion = { target = c:POR modifier = opinion_good_relations }
				}
				add_opinion = { target = c:POR modifier = pp_opinion_iberian_concord }
			}
			c:POR = {
				if = {
					limit = { has_opinion = { target = c:CAS modifier = opinion_good_relations } }
					remove_opinion = { target = c:CAS modifier = opinion_good_relations }
				}
				add_opinion = { target = c:CAS modifier = pp_opinion_iberian_concord }
			}
		}
	}
}

# Castile/Spain and Portugal never conquer each other: a location an AI winner takes from the other in a peace deal goes
# straight back to the loser. The peace AI ignores conquest preferences for border land (it floors every location's
# desire), so this is the only hard stop.
on_took_location_in_peace_treaty = {
	on_actions = {
		pp_iberian_return_peace_location
	}
}

pp_iberian_return_peace_location = {
	trigger = {
		scope:winner = { is_ai = yes }
		OR = {
			AND = {
				scope:winner = {
					OR = {
						tag = CAS
						tag = SPA
					}
				}
				scope:loser = { tag = POR }
			}
			AND = {
				scope:winner = { tag = POR }
				scope:loser = {
					OR = {
						tag = CAS
						tag = SPA
					}
				}
			}
		}
	}
	effect = {
		scope:location = {
			change_location_owner = scope:loser
		}
	}
}

pp_country_finetune_start_gold = {
	effect = {
		if = {
			limit = { exists = c:TUR }
			c:TUR = { add_gold = 1500 }
		}
		if = {
			limit = { exists = c:BYZ }
			c:BYZ = { add_gold = -1000 }
		}
		if = {
			limit = { exists = c:MOS }
			c:MOS = { add_gold = 500 }
		}
	}
}
