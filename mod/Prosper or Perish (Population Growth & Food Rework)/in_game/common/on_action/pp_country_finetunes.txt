# Country finetunes at game start (docs/country_finetunes.md): one-off starting gold, conquest preferences.
on_game_start = {
	on_actions = {
		pp_country_finetune_start_gold
		pp_country_finetune_area_preferences
	}
}

# Castile <-> Portugal: no conquest of each other (vanilla grants Castile's only after Rio Salado); the
# Reconquista aim is in ai_scripted_expansion_score/target pp_reconquista.txt. Muscovy: the Russian region
# (pp_gathering_russian_lands.txt).
pp_country_finetune_area_preferences = {
	effect = {
		c:CAS ?= {
			if = {
				limit = { NOT = { has_area_preference = castile_no_portugal_conquest } }
				add_area_preference = castile_no_portugal_conquest
			}
		}
		c:POR ?= {
			add_area_preference = pp_portugal_no_castile_conquest
		}
		c:MOS ?= {
			add_area_preference = pp_muscovy_gather_russian_lands
		}
	}
}

pp_country_finetune_start_gold = {
	effect = {
		if = {
			limit = { exists = c:TUR }
			c:TUR = { add_gold = 1500 }
		}
		if = {
			limit = { exists = c:BYZ }
			c:BYZ = { add_gold = -1000 }
		}
		if = {
			limit = { exists = c:FRA }
			c:FRA = { add_gold = 500 }
		}
	}
}
