# Establishment (know-how), branch establishment-know-how, 2026-10-06 (Jan).
pp_establishment_game_start = {
	effect = {
		every_country = {
			add_country_modifier = {
				modifier = pp_establishment_start
				months = 2
				mode = replace
			}
		}
	}
}

# Young craft buildings are protected from the cleanup for their first 8 years: a new workshop runs at a small share of
# its output until it reaches the establishment threshold, earns almost nothing and often lays off its workers; without
# the grace the cleanup destroyed it before it could learn (test run 078a901e: 2,543 of 6,550 new workshops gone within
# 10 years, the AI rebuilding them; baseline 25).
on_building_built = {
	on_actions = {
		pp_establishment_mark_young
	}
}

# root = the new building
pp_establishment_mark_young = {
	trigger = {
		pp_is_establishment_craft = yes
	}
	effect = {
		save_scope_as = pp_est_new
		location = {
			add_to_variable_list = { name = pp_est_young target = scope:pp_est_new.building_type years = 8 }
		}
	}
}

# Cleanup (Jan): a craft building of the ramping lines that is closed or has no workers on two yearly checks in a row
# is destroyed (not in its first 8 years). The first idle check notes its type in the location list pp_est_idle (kept 2 years); a check that finds
# it working again removes the note. Buildings are collected first and destroyed after the location's building loop.
yearly_country_pulse = {
	on_actions = {
		pp_establishment_cleanup
	}
}

pp_establishment_cleanup = {
	effect = {
		every_owned_location = {
			limit = {
				OR = {
					location_rank ?= location_rank:town
					location_rank ?= location_rank:city
					location_rank ?= location_rank:megalopolis
				}
			}
			pp_establishment_cleanup_location = yes
		}
	}
}
