# Prosper or Perish - Game Start Building Setup
# Game-start hooks (starting buildings come from the setup files).
on_game_start = {
	on_actions = {
		pp_mod_welcome_situation_game_start
		pp_apply_rgo_static_modifiers
		pp_raise_zero_rgo_max_workers
		pp_precalculate_building_capacities
		pp_game_start_effect
		pp_stored_food_game_start
	}
}

# Stored Food from day one (pp_stored_food.txt refreshes it each month through the country pulse).
pp_stored_food_game_start = {
	effect = {
		every_country = {
			every_province = {
				pp_refresh_stored_food = yes
			}
		}
	}
}

# Ensure every raw-material location has at least one max RGO worker before any startup reductions.
pp_raise_zero_rgo_max_workers = {
	effect = {
		every_location_in_the_world = {
			limit = {
				exists = raw_material
				NOT = { max_rgo_workers >= 1 }
			}
			change_max_raw_material_workers = {
				value = max_rgo_workers
				multiply = -1
				add = 1
			}
		}
	}
}

# Flags the welcome situation to spawn on the next monthly situation check (once per campaign).
pp_mod_welcome_situation_game_start = {
	effect = {
		if = {
			limit = {
				situation:pp_mod_welcome_situation = {
					situation_is_active = no
				}
			}
			set_global_variable = {
				name = pp_mod_welcome_situation_pending
				value = yes
			}
		}
	}
}

# +20% local output for each location's RGO, plus focused local flavor effects (see pp_rgo_static_bonuses.txt).
# Mid-campaign raw-material changes reuse the same location-scope refresh effect via on_raw_material_changed.
pp_apply_rgo_static_modifiers = {
	effect = {
		every_location_in_the_world = {
			limit = { exists = raw_material }
			pp_refresh_rgo_static_bonus = yes
			pp_refresh_fruit_orchard_eligibility = yes
		}
	}
}

pp_precalculate_building_capacities = {
	effect = {
		every_location_in_the_world = {
			set_variable = { name = pp_fish_base_capacity value = 0 }
			set_variable = { name = pp_forest_base_capacity value = 0 }
			set_variable = { name = pp_fruit_orchard_eligible value = 0 }
		}
		every_location_in_the_world = {
			limit = { is_ownable = yes }
			set_variable = {
				name = pp_fish_base_capacity
				value = pp_fish_base_capacity_value
			}
			set_variable = {
				name = pp_forest_base_capacity
				value = pp_forest_base_capacity_value
			}
			pp_refresh_fruit_orchard_eligibility = yes
		}
	}
}

pp_game_start_effect = {
	effect = {
		# Startup building construction is compiler-owned by pp_food_building_startup.
	}
}
