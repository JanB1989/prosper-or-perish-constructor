# Prosper or Perish - Game Start Building Setup
# Adds cookeries to towns and cities based on population and development thresholds.
# pp_apply_adjacent_to_lake_modifier
on_game_start = {
	on_actions = {
		pp_mod_welcome_situation_game_start
		pp_apply_rgo_static_modifiers
		pp_raise_zero_rgo_max_workers
		# pp_reduce_specific_rgo
		pp_precalculate_building_capacities
		pp_cull_setup_estate_buildings
		pp_remove_invalid_fiber_and_stud_farms
		pp_game_start_effect

		pp_load_base_population_baselines_on_start
		# pp_remove_invalid_buildings # we leave this off as it somehow causes crashes?

# Ensure every raw-material location has at least one max RGO worker before any startup reductions.
pp_raise_zero_rgo_max_workers = {
	effect = {
		every_location_in_the_world = {
			limit = {
				exists = raw_material
				NOT = { max_rgo_workers >= 1 }
			}
			change_max_raw_material_workers = {
				value = max_rgo_workers
				multiply = -1
				add = 1
			}
		}
	}
}

# Game-start tweak: lower max RGO workers per raw good. Branch on max_rgo_workers so the cut cannot push a level-1 RGO to 0.
pp_reduce_specific_rgo = {
	effect = {
		every_location_in_the_world = {
			limit = { exists = raw_material }
			# wheat — level >= 4: -2; level == 3: -1 (nested if/else_if so the chain cannot bind to another good's `if`)
			if = {
				limit = { raw_material = goods:wheat }
				if = {
					limit = { max_rgo_workers >= 4 }
					change_max_raw_material_workers = -2
				}
				else_if = {
					limit = { max_rgo_workers >= 3 }
					change_max_raw_material_workers = -1
				}
			}
			# legumes — apply when level >= 3; reduce max by 1
			if = {
				limit = {
					raw_material = goods:legumes
					max_rgo_workers >= 3
				}
				change_max_raw_material_workers = -1
			}
			# livestock — same nested pattern as wheat
			if = {
				limit = { raw_material = goods:livestock }
				if = {
					limit = { max_rgo_workers >= 4 }
					change_max_raw_material_workers = -2
				}
				else_if = {
					limit = { max_rgo_workers >= 3 }
					change_max_raw_material_workers = -1
				}
			}
			# rice — apply when level >= 2; reduce max by 1
			if = {
				limit = {
					raw_material = goods:rice
					max_rgo_workers >= 2
				}
				change_max_raw_material_workers = -1
			}
			# fish — apply when level >= 2; reduce max by 1
			if = {
				limit = {
					raw_material = goods:fish
					max_rgo_workers >= 2
				}
				change_max_raw_material_workers = -1
			}
			# wool — apply when level >= 2; reduce max by 1
			if = {
				limit = {
					raw_material = goods:wool
					max_rgo_workers >= 2
				}
				change_max_raw_material_workers = -1
			}
			# wild game — apply when level >= 2; reduce max by 1
			if = {
				limit = {
					raw_material = goods:wild_game
					max_rgo_workers >= 2
				}
				change_max_raw_material_workers = -1
			}
			if = {
				limit = {
					raw_material = goods:millet
					max_rgo_workers >= 3
				}
				change_max_raw_material_workers = -1
			}
			# cotton — apply when level >= 3; reduce max by 1
			if = {
				limit = {
					raw_material = goods:cotton
					max_rgo_workers >= 3
				}
				change_max_raw_material_workers = -1
			}
		}
	}
}

# Flags the welcome situation to spawn on the next monthly situation check (once per campaign).
# Do not re-set pending while the situation is already active (e.g. loading a save mid year-one).
pp_mod_welcome_situation_game_start = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = pp_mod_welcome_situation_completed }
				situation:pp_mod_welcome_situation = {
					situation_is_active = no
				}
			}
			set_global_variable = {
				name = pp_mod_welcome_situation_pending
				value = yes
			}
		}
	}
}

# Apply is_adjacent_to_lake modifier to locations that border lakes (engine does not auto-apply this like river_flowing_through/coastal)
pp_apply_adjacent_to_lake_modifier = {
	effect = {
		every_location_in_the_world = {
			limit = { is_adjacent_to_lake = yes }
			add_location_modifier = {
				modifier = is_adjacent_to_lake
				months = -1
				mode = replace
			}
		}
		every_location_in_the_world = {
			limit = {
				is_adjacent_to_lake = no
				has_location_modifier = is_adjacent_to_lake
			}
			remove_location_modifier = is_adjacent_to_lake
		}
	}
}
# +20% local output for each location's RGO, plus focused local flavor effects (see pp_rgo_static_bonuses.txt).
# Mid-campaign raw-material changes reuse the same location-scope refresh effect via on_raw_material_changed.
pp_apply_rgo_static_modifiers = {
	effect = {
		every_location_in_the_world = {
			limit = { exists = raw_material }
			pp_refresh_rgo_static_bonus = yes
			pp_refresh_fruit_orchard_eligibility = yes
		}
	}
}

pp_precalculate_building_capacities = {
	effect = {
		every_location_in_the_world = {
			set_variable = { name = pp_fish_base_capacity value = 0 }
			set_variable = { name = pp_forest_base_capacity value = 0 }
			set_variable = { name = pp_fruit_orchard_eligible value = 0 }
		}
		every_location_in_the_world = {
			limit = { is_ownable = yes }
			set_variable = {
				name = pp_fish_base_capacity
				value = pp_fish_base_capacity_value
			}
			set_variable = {
				name = pp_forest_base_capacity
				value = pp_forest_base_capacity_value
			}
			pp_refresh_fruit_orchard_eligibility = yes
		}
	}
}

# Population-delta map modes: load the frozen population of the reference campaign 'base'
# (pp_population_baseline_base_data.txt, generated by scripts/generate_population_baseline_data.py),
# so every campaign paints "change since base" and two campaigns stay comparable.
# Guarded by a global flag so it runs once per campaign. To re-baseline a running campaign:
#   event pp_population_baseline_debug.1   -> snapshot the CURRENT population instead
#   event pp_population_baseline_debug.2   -> reload the 'base' baseline
pp_load_base_population_baselines_on_start = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = pp_pop_baseline_snapshot_taken } }
			set_global_variable = {
				name = pp_pop_baseline_snapshot_taken
				value = yes
			}
			pp_load_base_population_baselines = yes
		}
	}
}
