REPLACE:guild_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	add = {
		desc = "BUILDING_LEVEL_DEVELOPMENT"
		value = development
		multiply = 0.4
	}
	add = {
		desc = "BUILDING_LEVEL_POPULATION"
		value = population
		multiply = 0.2
	}
	if = {
		limit = {
			location_rank ?= location_rank:city
		}
		add = {
			desc = "BUILDING_LEVEL_IS_CITY"
			value = 20
		}
	}
	if = {
		limit = {
			location_rank ?= location_rank:megalopolis
		}
		add = {
			desc = "BUILDING_LEVEL_IS_MEGAPOLIS"
			value = 40
		}
	}
	if = {
		limit = {
			location_rank ?= location_rank:rural_settlement
			has_variable = extra_levels_clothmaker_variable
		}
		add = {
			desc = "[ShowMissionTaskName('mission_cloth_makers')]"
			value = 2
		}
	}
	if = {
		limit = { market_access < 0.75 }
		multiply = {
			desc = "BUILDING_LEVEL_LOW_MARKET_ACCESS_PENALTY"
			value = market_access
			add = 0.25
		}
	}
	min = 1
}

		multiply = 0.1

			value = 1
		}
	}

		desc = "BUILDING_LEVEL_HAS_RIVER"

	}
	add = {
		desc = "BUILDING_LEVEL_FROM_OWNER_MODIFIERS"
		add = owner.modifier:irrigant_cap_level
	}
}

REPLACE:polders_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 0
	}
	if = {
		limit = {

				has_variable = ned_buffed_polders_flag
			}
		}
		add = {
			desc = "BUILDING_LEVEL_DEVELOPMENT"
			value = development
			multiply = 0.25
		}
	}
	else = {
		add = {
			desc = "BUILDING_LEVEL_DEVELOPMENT"
			value = development
			multiply = 0.2
		}
	}
}
