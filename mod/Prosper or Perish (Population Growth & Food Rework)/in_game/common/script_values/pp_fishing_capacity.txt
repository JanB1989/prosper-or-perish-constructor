fish_gross_capacity = {
	add = {
		desc = "BUILDING_LEVEL_BASE_FISHING"
		if = {
			limit = { has_variable = pp_fish_base_capacity }
			value = var:pp_fish_base_capacity
		}
		else = { value = pp_fish_base_capacity_value }
	}
	add = {
		desc = "BUILDING_LEVEL_RGO_SIZE_FISHING"
		if = {
			limit = { has_variable = pp_fish_base_capacity }
			value = var:pp_fish_base_capacity
		}
		else = { value = pp_fish_base_capacity_value }
		add = modifier:fish_capacity
		multiply = max_rgo_workers
		multiply = 0.030
	}
	add = {
		desc = "BUILDING_LEVEL_FISH_CAPACITY_IMPROVEMENTS"
		value = modifier:fish_capacity
	}
}
