# Prosper or Perish - values the food buildings' caps and AI weights read (2026-10-03).

# Location scope: the months of food the province holds, as the monthly store refresh last applied them
# (province variable pp_stored_food_months = months + 100; a province never refreshed counts as empty).
pp_location_stored_months = {
	value = 0
	province ?= {
		if = {
			limit = { has_variable = pp_stored_food_months }
			add = var:pp_stored_food_months
			subtract = 100
		}
	}
	min = 0
}

# Province scope: the farm, orchard and other staple food levels worked in the province (sum of modifier:local_pp_farm_levels
# and modifier:local_pp_staple_levels). The Cookshop cap and the Cookshop AI weight read it.
pp_province_farm_levels = {
	value = 0
	every_location_in_province = {
		add = modifier:local_pp_farm_levels
		add = modifier:local_pp_staple_levels
	}
}

# Location scope: the farm, orchard and other staple food levels worked in the location's province.
pp_location_province_farm_levels = {
	value = 0
	province ?= {
		add = pp_province_farm_levels
	}
}

# Province scope: the standing Victualling Yard levels in the province (sum of modifier:local_pp_victualling_yard_levels,
# raw_modifier, 2026-10-04). The Cookshop cap adds half a level per Yard level.
pp_province_victualling_yard_levels = {
	value = 0
	every_location_in_province = {
		add = modifier:local_pp_victualling_yard_levels
	}
}

# Location scope: the standing Victualling Yard levels in the location's province.
pp_location_province_victualling_yard_levels = {
	value = 0
	province ?= {
		add = pp_province_victualling_yard_levels
	}
}

# Province scope: the food the province gains per month (its locations' food production plus their food consumption,
# which the engine reports as negative). Above 0 the store grows. Read by the Grange's AI weight.
pp_province_food_balance = {
	value = 0
	every_location_in_province = {
		add = food_production
		add = food_consumption
	}
}

# Location scope: the food the location's province gains per month.
pp_location_province_food_balance = {
	value = 0
	province ?= {
		add = pp_province_food_balance
	}
}

# Location scope: the AI's food urgency (2026-10-04, Jan): every building that makes Province Food in the province
# (crop farms and their tiers, the other Provisioning food buildings, Tavern, Cookshop, Public Kitchen;
# tests/test_capital_food_weight.py) adds it to its ai_construct_weight after the income divide, so it is flat.
# Low food = under 12 stored months, critical under 8: the weight rises linearly as the store falls,
# 7.5 per month below 12 (8 months 30, 5 months 52.5, empty 90). The engine's own food value is 0 wherever the
# province makes more food than it eats, however low its store is (save 1454: Constantinople +16.3 a month at
# 5.4 months, Tavern utility -13.29 with Food Utility 0.0000), so this term carries the AI's answer to low stores.
pp_food_urgency_weight = {
	value = 12
	subtract = pp_location_stored_months
	min = 0
	multiply = 7.5
}

# Location scope (a province capital): the Granary's AI weight (2026-10-06, Jan). The engine values food capacity at
# about 0.3 against a queue median of 20-30, so without this the AI builds almost no Granaries. Flat (no income
# divide, so rich countries build too) and positive only (a negative weight drops the whole market from the queue
# pre-screen). Size: 20 + 0.4 per 1,000 people in the location, at most 60 (rural ~26, town ~36, city 60; farms
# score 22-34, Taverns in hungry provinces ~72). Only where a level would fill: the store is full (fading in from
# 85 to 90 % of capacity) and holds under 24 months (fading out from 23 to 24).
pp_granary_ai_weight = {
	value = population
	multiply = 0.4
	add = 20
	max = 60
	multiply = {
		value = province.province_food_percentage
		subtract = 0.85
		divide = 0.05
		min = 0
		max = 1
	}
	multiply = {
		value = 24
		subtract = pp_location_stored_months
		min = 0
		max = 1
	}
}
