# Prosper or Perish - values the food buildings' caps and AI weights read (2026-10-03).

# Location scope: the months of food the province holds, as the monthly store refresh last applied them
# (province variable pp_stored_food_months = months + 100; a province never refreshed counts as empty).
pp_location_stored_months = {
	value = 0
	province ?= {
		if = {
			limit = { has_variable = pp_stored_food_months }
			add = var:pp_stored_food_months
			subtract = 100
		}
	}
	min = 0
}

# Province scope: the farm, orchard and other staple food levels worked in the province (sum of modifier:local_pp_farm_levels
# and modifier:local_pp_staple_levels). The Cookshop cap and the Cookshop AI weight read it.
pp_province_farm_levels = {
	value = 0
	every_location_in_province = {
		add = modifier:local_pp_farm_levels
		add = modifier:local_pp_staple_levels
	}
}

# Location scope: the farm, orchard and other staple food levels worked in the location's province.
pp_location_province_farm_levels = {
	value = 0
	province ?= {
		add = pp_province_farm_levels
	}
}

# Province scope: the standing Victualling Yard levels in the province (sum of modifier:local_pp_victualling_yard_levels,
# raw_modifier, 2026-10-04). The Cookshop cap adds half a level per Yard level.
pp_province_victualling_yard_levels = {
	value = 0
	every_location_in_province = {
		add = modifier:local_pp_victualling_yard_levels
	}
}

# Location scope: the standing Victualling Yard levels in the location's province.
pp_location_province_victualling_yard_levels = {
	value = 0
	province ?= {
		add = pp_province_victualling_yard_levels
	}
}

# Province scope: the food the province gains per month (its locations' food production plus their food consumption,
# which the engine reports as negative). Above 0 the store grows. Read by the Grange's AI weight.
pp_province_food_balance = {
	value = 0
	every_location_in_province = {
		add = food_production
		add = food_consumption
	}
}

# Location scope: the food the location's province gains per month.
pp_location_province_food_balance = {
	value = 0
	province ?= {
		add = pp_province_food_balance
	}
}

# Location scope: the AI's extra wish for food buildings in its country's capital province (2026-10-04, Jan: the AI
# should focus on building up food in the capital). Added flat, after the income divide, to the ai_construct_weight of
# every building that makes Province Food in the province (crop farms and their tiers, the other Provisioning food
# buildings, Tavern, Cookshop, Public Kitchen; tests/test_capital_food_weight.py). Not divided by income: the engine
# values food and profit far less in rich empires (save 1592: farm scores 1.8 at income 300+ vs 7.9 under 20), and the
# capitals belong to the richest. Full up to 6 stored months, fading out to 0 at 14 (the farms' store_full_months).
# Profit gates, caps and the Tavern's store veto still decide; the gold buffer still stops a country's whole build walk.
pp_capital_food_weight = {
	value = 0
	if = {
		limit = {
			exists = owner
			exists = owner.capital
			province_definition = owner.capital.province_definition
		}
		add = { value = 14 subtract = pp_location_stored_months divide = 8 min = 0 max = 1 multiply = 100 }
	}
}
