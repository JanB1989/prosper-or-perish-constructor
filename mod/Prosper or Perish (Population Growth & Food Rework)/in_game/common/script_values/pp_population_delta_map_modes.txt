# Population delta vs snapshotted baselines for debug map modes.
# Metric: (current / baseline) - 1  →  0 = unchanged, -1 = -100%, +2 = +200%.
# Population is stored in thousands; display helpers multiply by 1000 for headcounts.
# Baselines are set by pp_snapshot_population_baselines (event pp_population_baseline_debug.1).

# --- Percent-change values used for map coloring (clamped externally) ---

pp_pop_delta_location_pct = {
	if = {
		limit = {
			has_variable = pp_pop_baseline
			var:pp_pop_baseline > 0
		}
		value = population
		divide = var:pp_pop_baseline
		subtract = 1
	}
	else = {
		value = 0
	}
}
