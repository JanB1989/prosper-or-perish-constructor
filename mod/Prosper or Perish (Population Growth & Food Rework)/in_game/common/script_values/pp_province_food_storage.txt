# Province food storage. EU5 1.4 dropped the stored-food modifier that carried the months of storage, so the
# stored-food tiers (pp_stored_food_tiers.txt), the stored-food map mode and the location view read these values.
# The province-scope values are the source; the location-scope ones (the location's province) delegate to them.

# Province scope: monthly food the province's pops eat, the sum of its locations' consumption. The location value
# food_consumption is negative (it is the pops' draw on the food balance), hence the sign flip.
pp_stored_food_province_consumption = {
	value = 0
	every_location_in_province = {
		add = food_consumption
	}
	multiply = -1
}

# Province scope: months of the province's consumption in storage, capped where the stored-food effects stop rising
# (NEconomy.GROWTH_FROM_FOOD_MULTIPLIER_MAX years = 24 months, the last stored-food tier).
pp_stored_food_province_months = {
	value = 0
	if = {
		limit = { pp_stored_food_province_consumption > 0 }
		value = province_food
		divide = pp_stored_food_province_consumption
		min = 0
		max = {
			value = define:NEconomy|GROWTH_FROM_FOOD_MULTIPLIER_MAX
			multiply = 12
		}
	}
}

# Location scope: the province's monthly food consumption (positive).
pp_province_monthly_food_consumption = {
	value = 0
	province ?= {
		add = pp_stored_food_province_consumption
	}
}

# Location scope: the province's months of stored food (0 .. the cap).
pp_province_food_storage_months = {
	value = 0
	province ?= {
		add = pp_stored_food_province_months
	}
}

# Growth from stored food is the local_population_growth of the province's stored-food tier modifier
# (pp_stored_food_tiers.txt; the engine's own storage term is off, NPop.FOOD_STORAGE_POP_GROWTH = 0). The
# display value pp_province_food_storage_growth (that tier's growth) is generated with the tiers
# (script_values/pp_stored_food_tiers.txt).

# Local population growth as the engine sums it, without the country-wide part: the local_population_growth
# modifier, which includes the stored-food tier's growth (read by the population growth map mode).
pp_population_growth_map_value = {
	value = modifier:local_population_growth
}
