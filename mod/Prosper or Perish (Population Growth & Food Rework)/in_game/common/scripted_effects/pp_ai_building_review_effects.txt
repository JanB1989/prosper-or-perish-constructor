# Prosper or Perish - yearly AI building review: one pass over the country's locations

# Location scope, root = the owning country. Removes one closed building level; import markets are kept because they
# close by design while food is plentiful.
pp_cull_one_closed_building = {
	random_buildings_in_location = {
		limit = {
			is_opened = no
			owner = root
			building_can_be_destroyed_by = root
			NOT = { building_type = building_type:tavern }
		}
		# Debug flag: scripted cull, counted per location (save) and logged. Both run before the level change: removing
		# the last level destroys the building, and with it this scope. Log fields: kind;date;building owner tag;location
		# id;building;level before the cull. The log text reaches the building through THIS (verified in game 2026-09-30;
		# ROOT and saved scopes resolve to nothing there).
		location = { if = { limit = { has_variable = pp_dbg_script_cull } change_variable = { name = pp_dbg_script_cull add = 1 } } else = { set_variable = { name = pp_dbg_script_cull value = 1 } } }
		error_log = "PPBLD;review_cull_closed;[GetDateString];[THIS.GetBuilding.GetOwner.GetTag];[THIS.GetBuilding.GetLocation.GetKey];[THIS.GetBuilding.GetType.GetNameWithNoTooltip];[THIS.GetBuilding.GetLevel]"
		change_building_level = -1
	}
}

# Country scope. Culls one closed level per location. The review builds nothing: the AI builds taverns on its own
# (the scripted tavern builds were removed 2026-09-29).
pp_ai_building_review_effect = {
	every_owned_location = {
		pp_cull_one_closed_building = yes
	}
}
