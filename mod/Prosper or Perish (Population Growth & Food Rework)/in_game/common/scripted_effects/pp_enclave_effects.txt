# PP (2026-10-13, docs/peace_deal_rulebook.md "Enclaves"), country scope (the nation, scope:actor of the enclave
# actions): notes the capital's piece (temporary list pp_connected_to_capital: land or strait, every_connected_location)
# and the capital's region (scope:pp_home_region) for pp_enclave_kept_location.
pp_enclave_mark_capital_piece = {
	capital ?= {
		add_to_temporary_list = pp_connected_to_capital
		every_connected_location = { add_to_temporary_list = pp_connected_to_capital }
		region = { save_scope_as = pp_home_region }
	}
}

# PP (2026-10-13), country scope (the nation, scope:actor): lists the provinces that hold land of a piece the nation
# lets go (pp_enclave_kept_location) in the interaction's source list.
pp_enclave_list_provinces = {
	if = {
		limit = { exists = capital }
		pp_enclave_mark_capital_piece = yes
		every_province = {
			limit = {
				any_location_in_province = {
					owner = scope:actor
					NOT = { pp_enclave_kept_location = yes }
				}
			}
			add_to_list = source
		}
	}
}

# PP (2026-10-13), province scope (the target province), scope:actor = the nation: notes one piece the actor lets go in
# this province (scope:pp_enclave_seat and the temporary list pp_enclave_piece: the seat and all land connected to it).
# Nothing is saved when the province holds no such land (another release of the same tick took it).
pp_enclave_pick_piece = {
	scope:actor = { pp_enclave_mark_capital_piece = yes }
	random_location_in_province = {
		limit = {
			owner = scope:actor
			NOT = { pp_enclave_kept_location = yes }
		}
		save_scope_as = pp_enclave_seat
		add_to_temporary_list = pp_enclave_piece
		every_connected_location = { add_to_temporary_list = pp_enclave_piece }
	}
}

# PP (2026-10-13), province scope (the target province), scope:actor = the nation, scope:pp_enclave_overlord = the
# country the released land answers to (the nation itself, or its overlord when the nation is a subject): lets one
# piece go. A piece that borders a subject of the overlord joins that subject; any other becomes a new subject of the
# overlord from its seat (independent when scope:pp_enclave_overlord is not set: a union junior). Its land gets the
# new country's cores, and the nation and the new country a truce.
pp_enclave_release_piece = {
	pp_enclave_pick_piece = yes
	if = {
		limit = {
			exists = scope:pp_enclave_seat
			exists = scope:pp_enclave_overlord
			any_in_list = {
				list = pp_enclave_piece
				any_neighbor_location = { owner ?= { is_subject_of = scope:pp_enclave_overlord NOT = { this = scope:actor } } }
			}
		}
		random_in_list = {
			list = pp_enclave_piece
			limit = { any_neighbor_location = { owner ?= { is_subject_of = scope:pp_enclave_overlord NOT = { this = scope:actor } } } }
			random_neighbor_location = {
				limit = { owner ?= { is_subject_of = scope:pp_enclave_overlord NOT = { this = scope:actor } } }
				owner = { save_scope_as = pp_enclave_heir }
			}
		}
		every_in_list = {
			list = pp_enclave_piece
			limit = { owner = scope:actor }
			change_location_owner = scope:pp_enclave_heir
		}
	}
	else_if = {
		limit = { exists = scope:pp_enclave_seat }
		scope:pp_enclave_seat = {
			create_country_from_location = {
				save_scope_as = pp_enclave_country
				set_capital = scope:pp_enclave_seat
			}
		}
		if = {
			limit = { exists = scope:pp_enclave_country }
			# a tributary where the overlord's government allows one, else a vassal (a samanta in India, where
			# samanta_advance hides the vassal); without an overlord (a union junior) the new country is independent
			scope:pp_enclave_country = {
				if = {
					limit = { NOT = { exists = scope:pp_enclave_overlord } }
				}
				else_if = {
					limit = { scope:pp_enclave_overlord = { pp_can_hold_tributary = yes } }
					make_subject_of = {
						target = scope:pp_enclave_overlord
						type = subject_type:tributary
					}
				}
				else_if = {
					limit = { scope:pp_enclave_overlord = { has_advance = samanta_advance } }
					make_subject_of = {
						target = scope:pp_enclave_overlord
						type = subject_type:samanta
					}
				}
				else = {
					make_subject_of = {
						target = scope:pp_enclave_overlord
						type = subject_type:vassal
					}
				}
			}
			every_in_list = {
				list = pp_enclave_piece
				limit = { owner = scope:actor }
				change_location_owner = scope:pp_enclave_country
			}
			every_in_list = {
				list = pp_enclave_piece
				limit = { owner = scope:pp_enclave_country }
				add_core = scope:pp_enclave_country
			}
			scope:pp_enclave_country = {
				add_truce_with = {
					target = scope:actor
					mutual = yes
				}
			}
		}
	}
}
