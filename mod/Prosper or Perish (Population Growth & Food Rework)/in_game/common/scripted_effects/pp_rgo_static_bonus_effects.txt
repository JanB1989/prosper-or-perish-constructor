pp_refresh_fruit_orchard_eligibility = {
	set_variable = { name = pp_fruit_orchard_eligible value = 0 }
	if = {
		limit = { pp_fruit_orchard_location_potential = yes }
		set_variable = { name = pp_fruit_orchard_eligible value = 1 }
	}
}
