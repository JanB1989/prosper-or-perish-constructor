# PP trade expeditions (docs/trade_expeditions.md): shared effects. Root = the expedition's country.
# $KIND$ is one of africa, india, far_east.

# Trace for tests and saves: the year of each stage in the country variable pp_te_year_<stage>_<kind>, and the country
# in the global list pp_te_log_<stage>_<kind> (error.log keeps one line per script location, useless as a trace).
pp_te_log_effect = {
	add_to_global_variable_list = {
		name = pp_te_log_$STAGE$_$KIND$
		target = this
	}
	set_variable = {
		name = pp_te_year_$STAGE$_$KIND$
		value = current_year
	}
}

# Fitting-out cost, paid at departure.
pp_te_pay_effect = {
	add_gold = {
		value = $COST$
		multiply = -1
	}
}

# Departure: the most populous owned port is the first waypoint; the leader leaves any command he holds.
pp_te_depart_effect = {
	ordered_owned_location = {
		limit = { is_port = yes }
		order_by = population
		max = 1
		save_scope_as = pp_te_departure
	}
	scope:expedition = {
		if = {
			limit = { exists = expedition_leader }
			remove_commander = expedition_leader
		}
		if = {
			limit = { exists = scope:pp_te_departure }
			add_new_waypoint = scope:pp_te_departure
		}
		else = {
			add_new_waypoint = root.capital
		}
	}
}

# Pick the target among the free candidate sites (pp_te_site_score: free coast first, then weak hosts, then rich
# ports, next to own stations), save it as scope:pp_te_target. Called after pp_te_build_list_$KIND$.
pp_te_pick_target_effect = {
	ordered_in_list = {
		list = pp_te_candidates
		limit = { pp_te_site_free = yes }
		order_by = pp_te_site_score
		max = 1
		save_scope_as = pp_te_target
	}
}

# Start: pay, leave port, choose the site, reserve it, and route the fleet (round the Cape for the Indian Ocean).
pp_te_start_effect = {
	pp_te_log_effect = { STAGE = start KIND = $KIND$ }
	pp_te_pay_effect = { COST = pp_te_cost_$KIND$ }
	pp_te_depart_effect = yes
	pp_te_build_list_$KIND$ = yes
	pp_te_pick_target_effect = yes
	if = {
		limit = { exists = scope:pp_te_target }
		set_variable = {
			name = pp_te_target_$KIND$
			value = scope:pp_te_target
		}
		scope:pp_te_target = {
			set_variable = {
				name = pp_te_reserved
				value = root
			}
		}
		scope:expedition = {
			if = {
				limit = {
					scope:pp_te_target = {
						NOT = { region = region:guinea_region }
						NOT = { region = region:sahel_region }
						NOT = { region = region:kongo_region }
					}
					NOT = { root.capital ?= { continent = continent:asia } }
				}
				add_new_waypoint = location:good_hope_cape
			}
			add_new_waypoint = scope:pp_te_target
		}
	}
	else = {
		# No free site at departure (another fleet reserved the last one): sail for the Cape and look again on arrival.
		# A waypoint at the inland capital lost Mecklenburg's Far East fleet without a trace (1712 test save).
		scope:expedition = { add_new_waypoint = location:good_hope_cape }
	}
}

# Location scope: free the reservation of a site.
pp_te_unreserve_effect = {
	if = {
		limit = { has_variable = pp_te_reserved }
		remove_variable = pp_te_reserved
	}
}

# Arrival. Root = the expedition country. Takes the target the expedition stored at its start; when that harbour was
# taken meanwhile (another country's foothold, or its owner is now at war with us), the fleet sails on to the best free
# site of the same expedition. Then the host decides:
#   no owner                  -> the expedition settles the coast and builds the Castle and Factory
#   the host cedes the site   -> bought for gold, a 20-year treaty (truce), Castle and Factory, garrison
#   a far weaker host refuses -> landing and war for the town (some of the time)
#   the host licenses trade   -> a Factory in its port and trade access; in India and the Far East a trade company
#   the host refuses          -> the fleet trades its cargo and comes home with nothing
# CEDE = chance factor for a sale, WAR = chance of a landing against a weak host that will not sell, LICENCE = chance a
# host grants a factory when it will not sell.
pp_te_arrive_effect = {
	if = {
		limit = { has_variable = pp_te_last_site }
		remove_variable = pp_te_last_site
	}
	if = {
		limit = { has_variable = pp_te_last_host }
		remove_variable = pp_te_last_host
	}
	if = {
		limit = { has_variable = pp_te_target_$KIND$ }
		var:pp_te_target_$KIND$ = {
			pp_te_unreserve_effect = yes
			save_scope_as = pp_te_target
		}
		remove_variable = pp_te_target_$KIND$
	}
	if = {
		limit = {
			OR = {
				NOT = { exists = scope:pp_te_target }
				scope:pp_te_target = { NOT = { pp_te_site_free = yes } }
			}
		}
		pp_te_build_list_$KIND$ = yes
		pp_te_pick_target_effect = yes
	}
	set_variable = {
		name = pp_te_last_$KIND$
		value = current_year
	}
	set_variable = {
		name = pp_te_outcome
		value = 5
	}
	if = {
		limit = {
			exists = scope:pp_te_target
			scope:pp_te_target = { pp_te_site_free = yes }
		}
		set_variable = {
			name = pp_te_last_site
			value = scope:pp_te_target
		}
		scope:pp_te_target = {
			if = {
				limit = { has_owner = yes }
				owner = { save_scope_as = pp_te_host }
			}
		}
		if = {
			limit = { NOT = { exists = scope:pp_te_host } }
			set_variable = { name = pp_te_outcome value = 1 }
			scope:pp_te_target = { pp_te_settle_effect = yes }
		}
		else = {
			set_variable = {
				name = pp_te_last_host
				value = scope:pp_te_host
			}
			random = {
				chance = {
					value = pp_te_cede_chance
					multiply = $CEDE$
					max = 90
				}
				set_variable = { name = pp_te_outcome value = 2 }
			}
			if = {
				limit = {
					var:pp_te_outcome > 4.5
					scope:pp_te_host = { pp_te_weak_host = yes }
					pp_te_free_to_sail = yes
					NOT = { has_truce_with = scope:pp_te_host }
					NOT = { is_at_war_with = scope:pp_te_host }
				}
				random = {
					chance = $WAR$
					set_variable = { name = pp_te_outcome value = 3 }
				}
			}
			if = {
				limit = { var:pp_te_outcome > 4.5 }
				random = {
					chance = {
						value = $LICENCE$
						if = {
							limit = { scope:pp_te_host = { pp_te_strong_host = no } }
							add = 15
						}
						max = 95
					}
					set_variable = { name = pp_te_outcome value = 4 }
				}
			}
			if = {
				limit = {
					var:pp_te_outcome > 1.5
					var:pp_te_outcome < 2.5
				}
				pp_te_cede_effect = yes
			}
			else_if = {
				limit = {
					var:pp_te_outcome > 2.5
					var:pp_te_outcome < 3.5
				}
				pp_te_war_effect = yes
			}
			else_if = {
				limit = {
					var:pp_te_outcome > 3.5
					var:pp_te_outcome < 4.5
				}
				pp_te_licence_effect = yes
			}
		}
	}
	if = {
		limit = {
			var:pp_te_outcome > 2.5
			var:pp_te_outcome < 3.5
		}
		pp_te_log_effect = { STAGE = war KIND = $KIND$ }
	}
	else_if = {
		limit = {
			var:pp_te_outcome > 0.5
			var:pp_te_outcome < 4.5
		}
		pp_te_log_effect = { STAGE = gain KIND = $KIND$ }
		if = {
			limit = { has_variable = pp_te_n_$KIND$ }
			change_variable = {
				name = pp_te_n_$KIND$
				add = 1
			}
		}
		else = {
			set_variable = {
				name = pp_te_n_$KIND$
				value = 1
			}
		}
	}
	else = {
		pp_te_log_effect = { STAGE = refused KIND = $KIND$ }
	}
	if = {
		limit = { var:pp_te_outcome < 1.5 }
		trigger_event_non_silently = pp_trade_expedition.1
	}
	else_if = {
		limit = { var:pp_te_outcome < 2.5 }
		trigger_event_non_silently = pp_trade_expedition.2
	}
	else_if = {
		limit = { var:pp_te_outcome < 3.5 }
		trigger_event_non_silently = pp_trade_expedition.3
	}
	else_if = {
		limit = { var:pp_te_outcome < 4.5 }
		trigger_event_non_silently = pp_trade_expedition.4
	}
	else = {
		trigger_event_non_silently = pp_trade_expedition.5
	}
}

# Location scope (the target). Root takes an unclaimed coast.
pp_te_settle_effect = {
	change_location_owner = root
	root = { save_scope_as = pp_te_holder }
	pp_te_fortify_effect = yes
}

# Root buys the site from scope:pp_te_host: the price goes to the host, a 20-year treaty of peace (truce) binds both,
# and the Castle and Factory goes up.
pp_te_cede_effect = {
	save_scope_as = pp_te_buyer
	set_variable = {
		name = pp_te_price
		value = pp_te_land_price
	}
	add_gold = {
		value = var:pp_te_price
		multiply = -1
	}
	scope:pp_te_host = {
		add_gold = root.var:pp_te_price
	}
	add_truce_with = {
		target = scope:pp_te_host
		years = 20
		mutual = yes
	}
	root = { save_scope_as = pp_te_holder }
	scope:pp_te_target = {
		change_location_owner = root
		pp_te_fortify_effect = yes
	}
}

# Location scope: Castle and Factory, the garrison the fleet brought, and the trace. The location's owner is
# scope:pp_te_holder (the expedition's country, or the winner of a landing war).
pp_te_fortify_effect = {
	construct_building = {
		building_type = building_type:pp_te_factory
		instant = yes
		cost_multiplier = 0
		cost_multiplier_reason = "game_concept_event"
	}
	set_variable = {
		name = pp_te_foothold_of
		value = scope:pp_te_holder
	}
	set_variable = {
		name = pp_te_foothold_year
		value = current_year
	}
	add_to_global_variable_list = {
		name = pp_te_all_footholds
		target = this
	}
	save_scope_as = pp_te_fort_site
	scope:pp_te_holder = {
		add_to_variable_list = {
			name = pp_te_footholds
			target = scope:pp_te_fort_site
		}
	}
	if = {
		limit = { scope:pp_te_holder = { current_age_or_later = { age = age_6_revolutions } } }
		add_location_modifier = { modifier = pp_te_works_fortress days = -1 mode = replace }
	}
	else_if = {
		limit = { scope:pp_te_holder = { current_age_or_later = { age = age_5_absolutism } } }
		add_location_modifier = { modifier = pp_te_works_star_fort days = -1 mode = replace }
	}
	else_if = {
		limit = { scope:pp_te_holder = { current_age_or_later = { age = age_4_reformation } } }
		add_location_modifier = { modifier = pp_te_works_bastion days = -1 mode = replace }
	}
	pp_te_garrison_effect = { COUNT = pp_te_garrison_regiments }
}

# Location scope: $COUNT$ regiments of scope:pp_te_holder put ashore here at full strength (heavy infantry and shot of the
# age).
pp_te_garrison_effect = {
	scope:pp_te_holder.capital = { save_scope_as = pp_te_home }
	while = {
		count = $COUNT$
		if = {
			limit = { scope:pp_te_holder = { current_age_or_later = { age = age_4_reformation } } }
			create_sub_unit_with_owner = {
				type = a_pikemen
				owner = scope:pp_te_holder
				origin = scope:pp_te_home
			}
			create_sub_unit_with_owner = {
				type = a_arquebusiers
				owner = scope:pp_te_holder
				origin = scope:pp_te_home
			}
		}
		else = {
			create_sub_unit_with_owner = {
				type = a_halberdiers
				owner = scope:pp_te_holder
				origin = scope:pp_te_home
			}
			create_sub_unit_with_owner = {
				type = a_early_arquebusiers
				owner = scope:pp_te_holder
				origin = scope:pp_te_home
			}
		}
	}
	# New regiments start without men (old expeditions branch, run pp_exp4); the fleet brings them at full strength.
	every_unit_in_location = {
		limit = { owner ?= scope:pp_te_holder }
		every_sub_unit = {
			add_subunit_strength_percentage = 1
		}
	}
}

# Landing (Goa 1510): a casus belli on the host for the target's province, war, a conquest preference for the region
# (so the AI peace deal may take a town away from its own land: peace_treaties/pp_cede_preferred_location.txt), and
# twice the garrison put ashore at the target.
pp_te_war_effect = {
	add_casus_belli = {
		target = scope:pp_te_host
		type = casus_belli:cb_conquer_province
		province = scope:pp_te_target.province
		years = 20
	}
	declare_war_with_cb = {
		target = scope:pp_te_host
		type = casus_belli:cb_conquer_province
		target_province = scope:pp_te_target.province
	}
	pp_te_conquest_preference_effect = yes
	root = { save_scope_as = pp_te_holder }
	scope:pp_te_target = {
		set_variable = {
			name = pp_te_war_of
			value = root
		}
		pp_te_garrison_effect = { COUNT = pp_te_garrison_regiments }
		pp_te_garrison_effect = { COUNT = pp_te_garrison_regiments }
	}
	add_to_global_variable_list = {
		name = pp_te_all_war_sites
		target = scope:pp_te_target
	}
}

# Licence (Surat 1612, Canton, Hirado): gifts to the host, trade access and a Factory in its port, kept by the crown.
# No trade company subject: script-made trade companies (building subjects, as vanilla create_building_subject) died
# within three months in Indian regions in every test, taking the Factory with them, while crown Factories there stood
# (1712 test save of run cf982ee6, side by side; docs/trade_expeditions.md).
pp_te_licence_effect = {
	set_variable = {
		name = pp_te_price
		value = pp_te_licence_price
	}
	add_gold = {
		value = var:pp_te_price
		multiply = -1
	}
	scope:pp_te_host = {
		add_gold = root.var:pp_te_price
	}
	create_relation = {
		first = scope:pp_te_host
		second = root
		type = relation_type:trade_access
	}
	scope:pp_te_target = {
		construct_building = {
			building_type = building_type:pp_te_lodge
			owner = root
			instant = yes
			cost_multiplier = 0
			cost_multiplier_reason = "game_concept_event"
		}
		set_variable = {
			name = pp_te_foothold_of
			value = root
		}
		set_variable = {
			name = pp_te_foothold_year
			value = current_year
		}
		add_to_global_variable_list = {
			name = pp_te_all_footholds
			target = this
		}
	}
	add_to_variable_list = {
		name = pp_te_footholds
		target = scope:pp_te_target
	}
}

# A conquest preference for the target's region (area_preferences/pp_trade_expedition_preferences.txt), once.
pp_te_conquest_preference_effect = {
	scope:pp_te_target = {
		if = {
			limit = { region = region:sahel_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_sahel } }
		}
		else_if = {
			limit = { region = region:guinea_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_guinea } }
		}
		else_if = {
			limit = { region = region:kongo_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_kongo } }
		}
		else_if = {
			limit = { region = region:swahili_coast_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_swahili } }
		}
		else_if = {
			limit = { region = region:zimbabwe_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_zimbabwe } }
		}
		else_if = {
			limit = { region = region:deccan_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_deccan } }
		}
		else_if = {
			limit = { region = region:western_india_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_western_india } }
		}
		else_if = {
			limit = { region = region:bengal_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_bengal } }
		}
		else_if = {
			limit = { region = region:indochina_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_indochina } }
		}
		else_if = {
			limit = { region = region:indonesia_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_indonesia } }
		}
		else_if = {
			limit = { region = region:south_china_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_south_china } }
		}
		else_if = {
			limit = { region = region:east_china_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_east_china } }
		}
		else_if = {
			limit = { region = region:japan_region }
			root = { pp_add_area_preference_once = { preference = pp_te_hold_japan } }
		}
	}
}

# Lost at sea: the reservation is freed, the cooldown runs from now.

# Console trace (run file pp_te_trace.txt; location scope = one foothold): two error.log lines with the same index, one
# from the founder (year founded, castle standing, founder still holds it: 2 = yes, 1 = no) and one from the current
# owner of the location.
pp_te_trace_effect = {

	set_global_variable = { name = pp_te_t_fy value = var:pp_te_foothold_year }

	if = {
		limit = { has_variable = pp_te_changed_year }
		set_global_variable = { name = pp_te_t_chg value = var:pp_te_changed_year }
	}
	set_global_variable = { name = pp_te_t_fort value = 1 }
	if = {
		limit = { any_buildings_in_location = { building_type = building_type:pp_te_factory } }

	}

	owner ?= {
		error_log = "PPTETO;i=[GetGlobalVariable('pp_te_t_i').GetValue|0];owner=[THIS.GetCountry.GetTag]"
	}
}

# on_location_changed_owner (root = the location, scope:winner = new owner, scope:loser = old owner;
# on_action/pp_trade_expedition_on_actions.txt). A town fought for by a landing war gets the Castle and Factory when the
# attacker takes it; a station that changes hands records when and to whom (staying power in tests and saves).
pp_te_owner_change_effect = {
	if = {
		limit = {
			has_variable = pp_te_war_of
			exists = scope:winner
			scope:winner = var:pp_te_war_of
		}
		remove_variable = pp_te_war_of
		scope:winner = { save_scope_as = pp_te_holder }
		if = {
			limit = {
				NOT = { any_buildings_in_location = { building_type = building_type:pp_te_factory } }
			}
			pp_te_fortify_effect = yes
		}
		scope:winner = {
			pp_te_log_effect = { STAGE = won KIND = war }
		}
	}
	else_if = {
		limit = { has_variable = pp_te_foothold_of }
		set_variable = {
			name = pp_te_changed_year
			value = current_year
		}
		if = {
			limit = { exists = scope:winner }
			set_variable = {
				name = pp_te_changed_to
				value = scope:winner
			}
		}
		add_to_global_variable_list = {
			name = pp_te_changed_hands
			target = this
		}
	}
}
