# The victuals export market sells its food as a negative victuals input: efficiency-immune revenue and a lower
# market demand figure, but a negative input never puts goods into the market. Each month this adds the
# matching victuals to every market with staffed exports, read from the tally good instead of a building scan.
pp_deliver_victuals_exports = {
	every_market_in_world = {
		limit = { pp_export_tally_in_market > 0 }
		add_goods_supply = {
			goods = goods:victuals
			amount = pp_victuals_export_delivery_amount
		}
	}
}
