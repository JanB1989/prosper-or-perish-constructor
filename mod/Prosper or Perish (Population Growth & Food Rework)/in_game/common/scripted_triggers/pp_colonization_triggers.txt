# Prosper or Perish - the mod's economy leaves countries with a smaller tax base than vanilla, so the AI
# reached vanilla's absolute exploration (30) tax-base gate decades late.
# Vanilla definition with only that threshold lowered; keep the rest in sync with vanilla.
# The colonization gate (vanilla 100, PP 60) is the define NAI.AI_COLONIZE_TAX_BASE_THRESHOLD since EU5 1.4
# (loading_screen/common/defines/pp_defines_adjustments.txt); 1.4 removed the scripted ai_country_should_colonize.

REPLACE:country_can_start_exploration_trigger = {
	modifier:may_explore = yes
	OR = {
		is_ai = no
		AND = {
			country_tax_base > 20
			OR = {
				has_game_rule = all_countries_explore
				trigger_if = {
					limit = {
						has_game_rule = western_exploration_only
					}
					this != c:PAP
					not = { has_preference_tag = "warfare" }
					OR = {
						capital.continent = continent:europe
						and = {
							is_subject = yes
							overlord ?= {
								capital.continent = continent:europe
							}
						}
					}
				}
			}
		}
	}
}
