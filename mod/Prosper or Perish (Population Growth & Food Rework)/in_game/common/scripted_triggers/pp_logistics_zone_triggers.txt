# Prosper or Perish - logistics zones (location scope): which overland logistics building a location can build.
# Geography only: the triggers sit in location_potential, which the game re-checks on owner change and which deletes
# buildings that fail it. The zones never overlap (tests/test_logistics.py evaluates them over every location); the
# Carrier Inn serves every road outside them. "Dry" climates are the Koeppen B group (desert and steppe) plus the
# dry-summer continental interiors of Anatolia, Iran and Transoxiana.

# Caravanserai: the dry lands of the caravan roads - Anatolia, Persia, the Levant, Arabia, Central Asia with the Tarim
# basin, and the Maghreb north of the Sahara.
pp_logistics_zone_caravan = {
	OR = {
		climate = arid
		climate = cold_arid
		climate = hot_steppe
		climate = cold_steppe
		climate = dry_summer_continental
	}
	OR = {
		sub_continent = sub_continent:middle_east
		AND = {
			sub_continent = sub_continent:central_asia
			NOT = { area = area:desht_kipchak_area }
		}
		region = region:xinjiang_region
		AND = {
			region = region:maghreb_region
			NOT = { area = area:desert_area }
			NOT = { area = area:western_sahara_area }
		}
	}
}

# Oasis Caravan Station: the desert and savanna of Africa south of the Maghreb - the Sahara, the Sahel, Nubia and Wadai.
pp_logistics_zone_oasis = {
	OR = {
		climate = arid
		climate = hot_steppe
		climate = savanna
	}
	OR = {
		region = region:sahel_region
		AND = {
			region = region:nubia_region
			NOT = { area = area:equatoria_area }
		}
		area = area:desert_area
		area = area:western_sahara_area
		area = area:wadai_area
	}
}

# Yam Station: the steppe, the forests and the taiga of northern and eastern Eurasia - Russia, the Urals, the Pontic
# and Kazakh steppe, Siberia and Mongolia (with the Gobi).
pp_logistics_zone_yam = {
	OR = {
		climate = cold_steppe
		climate = continental
		climate = subarctic
		climate = cold_arid
	}
	OR = {
		region = region:russian_region
		region = region:ural_region
		region = region:steppes_region
		region = region:west_siberia_region
		region = region:east_siberia_region
		region = region:mongolia_region
		area = area:desht_kipchak_area
	}
}

# Banjara Tanda: South Asia.
pp_logistics_zone_banjara = {
	sub_continent = sub_continent:south_asia
}

# Llama Caravan Post: the Andean highlands (no llamas in Mesoamerica).
pp_logistics_zone_llama = {
	region = region:andes_region
	OR = {
		topography = mountains
		topography = plateau
		topography = hills
		topography = valleys
	}
}

# Any regional variant: the Carrier Inn builds outside these.
pp_logistics_zone_any = {
	OR = {
		pp_logistics_zone_caravan = yes
		pp_logistics_zone_oasis = yes
		pp_logistics_zone_yam = yes
		pp_logistics_zone_banjara = yes
		pp_logistics_zone_llama = yes
	}
}
