# Prosper or Perish - logistics zones (location scope): which overland logistics building a location can build.
# Scoped by the map hierarchy only (sub-continent, region, area; never climate, owner or culture): whole regions split
# the world into exact, stable zones, and the triggers sit in location_potential, which the game re-checks on owner
# change. The zones never overlap (tests/test_logistics.py evaluates them over every location); the Carrier Inn serves
# every road outside them.

# Caravanserai: the khan and caravanserai roads - Anatolia, the Caucasus, Persia, the Fertile Crescent, Arabia, Egypt,
# the Maghreb north of the Sahara, Khorasan and Transoxiana, and the Tarim basin.
pp_logistics_zone_caravan = {
	OR = {
		region = region:anatolia_region
		region = region:caucasus_region
		region = region:persia_region
		region = region:crescent_region
		region = region:arabia_region
		region = region:egypt_region
		region = region:xinjiang_region
		AND = {
			region = region:maghreb_region
			NOT = { area = area:desert_area }
			NOT = { area = area:western_sahara_area }
		}
		AND = {
			region = region:khorasan_region
			NOT = { area = area:desht_kipchak_area }
		}
	}
}

# Oasis Caravan Station: the Sahara, the Sahel, Nubia and the Somali coast.
pp_logistics_zone_oasis = {
	OR = {
		region = region:sahel_region
		region = region:nubia_region
		region = region:somalia_region
		area = area:desert_area   # the Sahara part of the Maghreb region
		area = area:western_sahara_area
	}
}

# Yam Station: Russia, the Urals, the Pontic steppe, Siberia, Mongolia and the Kazakh steppe.
pp_logistics_zone_yam = {
	OR = {
		region = region:russian_region
		region = region:ural_region
		region = region:steppes_region
		region = region:west_siberia_region
		region = region:east_siberia_region
		region = region:mongolia_region
		area = area:desht_kipchak_area
	}
}

# Banjara Tanda: South Asia.
pp_logistics_zone_banjara = {
	sub_continent = sub_continent:south_asia
}

# Llama Caravan Post: the Andes (no llamas in Mesoamerica).
pp_logistics_zone_llama = {
	region = region:andes_region
}

# Any regional variant: the Carrier Inn builds outside these.
pp_logistics_zone_any = {
	OR = {
		pp_logistics_zone_caravan = yes
		pp_logistics_zone_oasis = yes
		pp_logistics_zone_yam = yes
		pp_logistics_zone_banjara = yes
		pp_logistics_zone_llama = yes
	}
}
