	OR = {
		raw_material = goods:fiber_crops
		AND = {
			NOT = { climate = arctic }
			NOT = { OR = { climate = cold_arid climate = ha1300_climate_cold_steppe } }
			OR = {
				OR = { topography = flatland topography = ha1300_topo_rolling topography = ha1300_topo_valleys topography = ha1300_topo_floodplains topography = ha1300_topo_deltas }
				topography = hills
				topography = plateau

				vegetation = farmland
