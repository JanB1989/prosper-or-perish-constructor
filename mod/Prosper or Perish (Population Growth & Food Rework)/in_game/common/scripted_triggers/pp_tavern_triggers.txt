# Prosper or Perish - when the AI adds a victuals import level (yearly AI building review)

# Location scope, root = the owning country. A province capital whose province is very short of food and whose
# import market is either missing or already fully at work and paying (the victuals price enters through that profit).
# The province must not be exporting food at the same time.
pp_tavern_wanted = {
	is_province_capital = yes
	modifier:pp_province_food_storage_months < pp_tavern_low_storage_months
	# ...and the store is really short: the engine drops the storage modifier (and this marker) when the province's
	# consumption is zero or negative, e.g. purely tribal land whose pops eat nothing, so a full surplus province reads 0 months above.
	# province_monthly_food_production is the net monthly change (verified 2026-09-25: tribal Taoer +315, Uppland -169);
	# starving provinces whose balance already turned positive stay eligible.
	province = {
		OR = {
			is_starving = yes
			province_monthly_food_production < 0
		}
	}
	OR = {
		# no market and none queued: a queued first level already exists as a building under construction
		AND = {
			NOT = { any_buildings_in_location = { building_type = building_type:tavern } }
			can_build_building = building_type:tavern
		}
		any_buildings_in_location = {
			building_type = building_type:tavern
			building_can_be_upgraded_by = root
			is_at_max_level = no
			building_levels_under_construction = 0
			is_full_capacity = yes
			is_lacking_goods = no
			building_profit > 0
		}
	}
	province = {
		NOT = {
			any_location_in_province = {
				any_buildings_in_location = {
					building_type = building_type:victualling_yard
					is_opened = yes
				}
			}
		}
	}
}
