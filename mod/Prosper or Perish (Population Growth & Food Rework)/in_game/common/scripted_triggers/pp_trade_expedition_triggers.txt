# PP trade expeditions (docs/trade_expeditions.md): shared gates. Country scope unless noted; root = the expedition's
# country.

# A colonial power: a sovereign European country with colonial charters that faces the ocean: a harbour on the Atlantic,
# the North Sea or the Baltic coasts of the west (Iberia, France, the British Isles, the Low Countries and North
# Germany, Scandinavia, the Atlantic islands), a sea-going realm (150+ locations or 8+ ports on its capital's continent
# that face open sea, the test of the AI colonial charter gate, auto_modifiers/pp_colonial_charter_gate.txt) and a tax
# base of 75+ (run cf982ee6: the colonizers of 1652 ran 110-1,893; the one-province colonizers 9-26). In the 1612 save
# of that run the first draft without the Europe and ocean test let the Ottomans, Ming, Korea, Vijayanagara and the
# Papacy in.
pp_te_colonial_power = {
	is_subject = no
	has_advance = colonies
	has_ports = yes
	exists = capital
	capital = { continent = continent:europe }
	country_tax_base >= 75
	any_owned_location = {
		is_port = yes
		OR = {
			region = region:iberia_region
			region = region:france_region
			region = region:great_britain_region
			region = region:ireland_region
			region = region:north_german_region
			region = region:scandinavian_region
			region = region:north_atlantic_islands_region
		}
	}
	OR = {
		num_locations >= 150
		any_port_in_country = {
			count >= 8
			continent = root.capital.continent
			any_neighbor_location = {
				OR = {
					topography = ocean
					topography = deep_ocean
					topography = coastal_ocean
					topography = narrows
					topography = inland_sea
				}
				NOT = { area = area:black_sea_area }
				NOT = { area = area:caspian_sea_area }
			}
		}
	}
}

# The fleet sails only when the treasury is sound and no great power is at war with the country (a fleet and a
# garrison cannot be spared then). Applies to the AI and the player alike.
pp_te_free_to_sail = {
	NOT = {
		any_country_at_war_with = { is_great_power = yes }
	}
}

# The expedition's leader: an adult of the court who holds no other post at sea.
pp_te_leader = {
	is_adult = yes
	is_ruler = no
	is_expedition_leader = no
	has_exploration = no
}

# Host (country scope) far weaker than the buyer (root): not a great power, under half its tax base.
pp_te_weak_host = {
	is_great_power = no
	country_tax_base < pp_te_root_half_tax_base
}

# Host (country scope) strong enough to grant only a licence: a great power, or richer than the buyer.
pp_te_strong_host = {
	OR = {
		is_great_power = yes
		country_tax_base > root.country_tax_base
	}
}

# Location scope: a harbour a trade expedition may still go for: a port, not held by root or its subjects, not held by
# a European country or one of its subjects (Europeans do not sell each other their stations), and no trade-expedition
# foothold of any country there yet.
pp_te_site_free = {
	is_port = yes
	NOT = { has_variable = pp_te_foothold_of }
	NOT = { has_variable = pp_te_reserved }
	NOT = { owner ?= root }
	NOT = {
		owner ?= { is_subject_of = root }
	}
	NOT = {
		owner ?= {
			OR = {
				capital ?= { continent = continent:europe }
				overlord ?= { capital ?= { continent = continent:europe } }
			}
		}
	}
	NOT = {
		owner ?= { is_at_war_with = root }
	}
}

# The Cape route is open to the country: the Cape of Good Hope discovered, or the Age of Reformation.
pp_te_cape_route_known = {
	OR = {
		has_discovered = location:good_hope_cape
		current_age_or_later = { age = age_4_reformation }
	}
}

# Location scope: a foothold of a trade expedition that its founder still holds (owner, or the owner of the factory).
pp_te_foothold_held = {
	has_variable = pp_te_foothold_of
	OR = {
		owner ?= var:pp_te_foothold_of
		any_buildings_in_location = {
			OR = {
				building_type = building_type:pp_te_factory
				building_type = building_type:pp_te_lodge
			}
		}
	}
}
