# Welcome: starts at game start for the human players and never ends (the player closes it when done reading).
# Its panel (gui/panels/situation/pp_mod_welcome_situation.gui) opens the Prosper or Perish section of the Europedia.
pp_mod_welcome_situation = {
	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		has_global_variable = pp_mod_welcome_situation_pending
	}

	can_end = {
		always = no
	}

	visible = {
		is_human = yes
	}

	on_start = {
		remove_global_variable = pp_mod_welcome_situation_pending
	}

	map_color = {
		if = {
			limit = { always = yes }
			value = define:NMapColors|DEFAULT_COLOR
		}
	}
}
