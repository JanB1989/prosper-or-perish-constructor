# Coastal Raid (2026-10-23, Jan): a navy in a war of the PP Sea Raid casus belli (pp_naval_raid_cb.txt) raids one enemy
# coastal location next to its sea zone. Its peasants lose a share and a smaller share arrives as slaves in the
# raider's capital (vanilla's wokou raid ratios, wokou_events.2); the location is Recently Razed for a year and cannot
# be raided again until then. Modelled on raid_clan_holdings. Not tested in game.

pp_coastal_raid = {
	toggle = no
	navy_only = yes
	map = yes
	duration = 30
	cancel_on_combat = yes

	hidden = {
		owner = {
			NOT = { in_war_of_casus_belli = casus_belli:cb_pp_naval_raid }
		}
	}

	allow = {
		is_navy = yes
		in_combat = no
		is_moving = no
		is_exiled = no
		custom_tooltip = {
			text = pp_coastal_raid_target_tt
			unit_location = {
				any_neighbor_location = {
					is_land = yes
					is_coastal = yes
					owner ?= {
						"war_with_country(root.owner)" ?= {
							has_casus_belli = yes
							casus_belli ?= casus_belli:cb_pp_naval_raid
						}
					}
				}
			}
		}
		custom_tooltip = {
			text = pp_coastal_raid_not_razed_tt
			unit_location = {
				any_neighbor_location = {
					is_land = yes
					is_coastal = yes
					owner ?= {
						"war_with_country(root.owner)" ?= {
							has_casus_belli = yes
							casus_belli ?= casus_belli:cb_pp_naval_raid
						}
					}
					NOT = { has_location_modifier = recently_razed }
					any_pop = { pop_type = pop_type:peasants }
				}
			}
		}
	}

	finished_when = {
		OR = {
			in_combat = yes
			is_moving = yes
			is_exiled = yes
		}
	}

	modifier = {
		naval_morale_modifier = -0.25
	}

	start_effect = {
		unit_location = {
			random_neighbor_location = {
				limit = {
					is_land = yes
					is_coastal = yes
					owner ?= {
						"war_with_country(root.owner)" ?= {
							has_casus_belli = yes
							casus_belli ?= casus_belli:cb_pp_naval_raid
						}
					}
					NOT = { has_location_modifier = recently_razed }
					any_pop = { pop_type = pop_type:peasants }
				}
				save_scope_as = pp_raided_location
			}
		}
		scope:pp_raided_location ?= {
			every_pop = {
				limit = {
					pop_type = pop_type:peasants
				}
				save_temporary_scope_as = pp_raided_pop
				root.owner.capital = {
					add_pop = {
						culture = scope:pp_raided_pop.culture
						religion = scope:pp_raided_pop.religion
						size = {
							value = scope:pp_raided_pop.pop_size
							multiply = 0.004
						}
						type = pop_type:slaves
					}
				}
				add_pop_size = {
					value = pop_size
					multiply = -0.01
				}
			}
			add_location_modifier = {
				modifier = recently_razed
				years = 1
				mode = add_and_extend
			}
		}
	}

	ai_will_do = {
		add = 25
	}
}
