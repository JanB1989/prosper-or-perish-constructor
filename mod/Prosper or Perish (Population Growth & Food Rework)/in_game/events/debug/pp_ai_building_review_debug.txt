namespace = pp_ai_building_review_debug

# All option effects are hidden: the engine never builds tooltips for the list, variable and construction effects.
pp_ai_building_review_debug.1 = {
	type = country_event
	title = pp_ai_building_review_debug.1.title
	desc = pp_ai_building_review_debug.1.desc
	outcome = neutral
	orphan = yes

	# step 1: evaluate the rule in my country and count, nothing is built or culled
	option = {
		name = pp_ai_building_review_debug.1.a
		hidden_effect = {
			set_variable = { name = pp_debug_imports_wanted value = 0 }
			set_variable = { name = pp_debug_spare_gold value = pp_ai_spare_construction_gold }
			every_owned_location = {
				limit = { pp_victuals_import_wanted = yes }
				root = { change_variable = { name = pp_debug_imports_wanted add = 1 } }
			}
			trigger_event_non_silently = pp_ai_building_review_debug.2
		}
	}

	# step 2: queue the imports in my country, nothing is culled
	option = {
		name = pp_ai_building_review_debug.1.b
		hidden_effect = {
			every_owned_location = {
				limit = { pp_victuals_import_wanted = yes }
				add_to_temporary_list = pp_imports_wanted
			}
			pp_build_wanted_victuals_imports = yes
		}
	}

	# step 3: the real yearly review for every AI country, cull included
	option = {
		name = pp_ai_building_review_debug.1.c
		hidden_effect = {
			every_country = {
				limit = { is_ai = yes }
				pp_ai_building_review_effect = yes
			}
		}
	}
}
