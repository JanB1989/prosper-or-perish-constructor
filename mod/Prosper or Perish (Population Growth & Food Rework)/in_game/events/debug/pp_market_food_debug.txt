namespace = pp_market_food_debug

pp_market_food_debug.1 = {
	type = location_event
	title = pp_market_food_debug.1.title
	desc = pp_market_food_debug.1.desc
	outcome = neutral
	orphan = yes

	immediate = {
		remove_variable = pp_kaifeng_market_food_before
		remove_variable = pp_kaifeng_market_food_after
		remove_variable = pp_kaifeng_market_food_percentage_before
		remove_variable = pp_kaifeng_market_food_percentage_after
		remove_variable = pp_kaifeng_market_food_debug_missing

		location:kaifeng.market ?= {
			save_scope_as = pp_kaifeng_market_food_debug_target
		}

		if = {
			limit = { exists = scope:pp_kaifeng_market_food_debug_target }

			set_variable = {
				name = pp_kaifeng_market_food_before
				value = { add = scope:pp_kaifeng_market_food_debug_target.market_food }
			}
			set_variable = {
				name = pp_kaifeng_market_food_percentage_before
				value = { add = scope:pp_kaifeng_market_food_debug_target.market_food_percentage }
			}
		}

		pp_monthly_market_food_stockpile_topup = yes

		if = {
			limit = { exists = scope:pp_kaifeng_market_food_debug_target }
			set_variable = {
				name = pp_kaifeng_market_food_after
				value = { add = scope:pp_kaifeng_market_food_debug_target.market_food }
			}
			set_variable = {
				name = pp_kaifeng_market_food_percentage_after
				value = { add = scope:pp_kaifeng_market_food_debug_target.market_food_percentage }
			}
		}
	}

	option = {
		name = pp_market_food_debug.1.a
	}
}
