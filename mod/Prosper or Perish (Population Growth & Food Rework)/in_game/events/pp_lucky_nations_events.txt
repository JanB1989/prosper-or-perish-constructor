namespace = pp_lucky_nations

# Lucky nations (docs/country_finetunes.md): Portugal looks across the strait to Morocco from 1400 (Ceuta 1415).
pp_lucky_nations.1 = {
	type = country_event outcome = neutral title = empty_text desc = empty_text hidden = yes fire_only_once = yes
	dynamic_historical_event = { tag = POR from = 1400.1.1 to = 1600.1.1 monthly_chance = 10 }
	trigger = {
		NOT = { has_area_preference = pp_portugal_morocco }
	}
	immediate = {
		add_area_preference = pp_portugal_morocco
	}
}

# From 1450 Castile/Spain may take Portuguese land again (vanilla's -0.9 preference stays).
pp_lucky_nations.2 = {
	type = country_event outcome = neutral title = empty_text desc = empty_text hidden = yes fire_only_once = yes
	dynamic_historical_event = { tag = CAS tag = SPA from = 1450.1.1 to = 1836.12.31 monthly_chance = 20 }
	trigger = {
		has_area_preference = pp_castile_spare_portugal
	}
	immediate = {
		remove_area_preference = pp_castile_spare_portugal
		if = {
			limit = { NOT = { has_area_preference = castile_no_portugal_conquest } }
			add_area_preference = castile_no_portugal_conquest
		}
	}
}
