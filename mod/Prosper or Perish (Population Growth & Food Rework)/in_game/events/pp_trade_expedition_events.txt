namespace = pp_trade_expedition

# PP trade expeditions (docs/trade_expeditions.md): outcome events. Root = the expedition's country; the site and its
# former host come from the country variables pp_te_last_site / pp_te_last_host set on arrival
# (scripted_effects/pp_trade_expedition_effects.txt). The outcome itself is already applied when the event fires.

# 1. Outcome: settle.
pp_trade_expedition.1 = {
	type = country_event
	title = pp_trade_expedition.1.title
	desc = pp_trade_expedition.1.desc
	outcome = positive
	illustration_tags = { 10 = regular }

	immediate = {
		if = {
			limit = { has_variable = pp_te_last_site }
			var:pp_te_last_site = { save_scope_as = pp_te_site }
		}
	}

	option = {
		name = pp_trade_expedition.1.a
		add_prestige = prestige_mild_bonus
		ai_chance = { base = 100 }
	}
}
