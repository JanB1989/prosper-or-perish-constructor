# Map mode: total building levels per location (traffic-light bucketed heatmap).
# Low buckets retain the current savegame distribution, with extra high-end
# bands so dense late-game locations remain distinguishable.

@pp_building_levels_very_low = 3
@pp_building_levels_low = 8
@pp_building_levels_medium = 22
@pp_building_levels_high = 40
@pp_building_levels_strong = 60
@pp_building_levels_very_strong = 80
@pp_building_levels_cap = 100
@pp_supported_building_levels_low = 30
@pp_supported_building_levels_mid = 70
@pp_supported_building_levels_high = 120
@pp_supported_building_levels_very_high = 200
@pp_supported_building_levels_cap = 300
@pp_unsupported_building_levels_low = 1
@pp_unsupported_building_levels_mid = 5
@pp_unsupported_building_levels_high = 15
@pp_unsupported_building_levels_severe = 30
@pp_unsupported_building_levels_cap = 60

@zoom_step_near = 2
@data_gradient_alpha_inside = 1
@data_gradient_alpha_outside = 1
@data_gradient_width = 0.25
@data_gradient_color_mult = 0.9
@data_edge_width = 0
@data_edge_sharpness = 0.01
@data_edge_alpha = 0
@data_edge_color_mult = 0
@data_before_lighting_blend = 0.5
@data_after_lighting_blend = 0.5

pp_building_levels = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = { has_owner = yes }
			if = {
				limit = { total_building_levels < @pp_building_levels_very_low }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MIN
					max_color = define:NMapColors|MAP_COLOR_LOW
					factor = { value = total_building_levels divide = @pp_building_levels_very_low max = 1 min = 0 }
				}
			}
			else_if = {
				limit = { total_building_levels < @pp_building_levels_low }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_LOW
					max_color = define:NMapColors|MAP_COLOR_MID
					factor = { value = total_building_levels subtract = @pp_building_levels_very_low divide = 5 max = 1 min = 0 }
				}
			}
			else_if = {
				limit = { total_building_levels < @pp_building_levels_medium }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MID
					max_color = define:NMapColors|MAP_COLOR_HIGH
					factor = { value = total_building_levels subtract = @pp_building_levels_low divide = 14 max = 1 min = 0 }
				}
			}
			else_if = {
				limit = { total_building_levels < @pp_building_levels_high }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_HIGH
					max_color = define:NMapColors|MAP_COLOR_MAX
					factor = { value = total_building_levels subtract = @pp_building_levels_medium divide = 18 max = 1 min = 0 }
				}
			}
			else_if = {
				limit = { total_building_levels < @pp_building_levels_strong }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MAX
					max_color = define:NMapColors|MAP_COLOR_TOP
					factor = { value = total_building_levels subtract = @pp_building_levels_high divide = 20 max = 1 min = 0 }
				}
			}
			else_if = {
				limit = { total_building_levels < @pp_building_levels_very_strong }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_TOP
					max_color = rgb { 245 245 245 }
					factor = { value = total_building_levels subtract = @pp_building_levels_strong divide = 20 max = 1 min = 0 }
				}
			}
			else_if = {
				limit = { total_building_levels < @pp_building_levels_cap }
				lerp = {
					min_color = rgb { 245 245 245 }
					max_color = rgb { 170 235 255 }
					factor = { value = total_building_levels subtract = @pp_building_levels_very_strong divide = 20 max = 1 min = 0 }
				}
			}
			else = { value = rgb { 70 210 255 } }
		}
		else = {
			value = rgb { 128 128 128 }
		}
	}
	secondary_map_color = {
		if = {
			limit = {
				has_owner = yes
				total_building_levels > modifier:free_building_levels
			}
			value = rgb { 35 35 42 }
		}
	}
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_0_3" color = define:NMapColors|MAP_COLOR_MIN }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_3_8" color = define:NMapColors|MAP_COLOR_LOW }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_8_22" color = define:NMapColors|MAP_COLOR_MID }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_22_40" color = define:NMapColors|MAP_COLOR_HIGH }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_40_60" color = define:NMapColors|MAP_COLOR_MAX }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_60_80" color = define:NMapColors|MAP_COLOR_TOP }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_RANGE_80_100" color = rgb { 245 245 245 } }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_CAPPED" color = rgb { 70 210 255 } }
	legend_key = { desc = "MAPMODE_PP_BUILDING_LEVELS_OVER_SUPPORTED" color = rgb { 35 35 42 } }
	tooltip_key = {
		if = { limit = { is_land = yes } value = MAPMODE_PP_BUILDING_LEVELS_TT_LAND }
		else = { value = MAPMODE_PP_BUILDING_LEVELS_TT_WATER }
	}
	small_map_names = location
	medium_map_names = location
	large_map_names = location
	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location
	fill_in_impassable = yes
	enable_snow = no
	flatmap_behaviour = always
	use_fow = no
	category = economy
	index = 0
	map_markers = { all = no }
	gradient_parameters = {
		zoom_step = @zoom_step_near
		gradient_alpha_inside = @data_gradient_alpha_inside
		gradient_alpha_outside = @data_gradient_alpha_outside
		gradient_width = @data_gradient_width
		gradient_color_mult = @data_gradient_color_mult
		edge_width = @data_edge_width
		edge_sharpness = @data_edge_sharpness
		edge_alpha = @data_edge_alpha
		edge_color_mult = @data_edge_color_mult
		before_lighting_blend = @data_before_lighting_blend
		after_lighting_blend = @data_after_lighting_blend
	}
	color_refresh_counters = { ProductionList LocationDevelopmentChanged }
	color_and_names_refresh_counters = { LocationOwnerChanged CountryStatus }
}

# Map mode: supported building levels per location.
# Buckets are weighted toward the low end: 0-30, 30-70, 70-120, 120-200, 200-300.

pp_supported_building_levels = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}

# Map mode: local building efficiency modifier per location.
# Diverging traffic-light heatmap: red = costly, yellow = near neutral, green = efficient.

pp_building_efficiency = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = { has_owner = yes }
			if = {
				limit = { pp_building_efficiency_map_value < -1 }
				value = define:NMapColors|MAP_COLOR_MIN
			}
			else_if = {
				limit = { pp_building_efficiency_map_value < -0.5 }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MIN
					max_color = define:NMapColors|MAP_COLOR_LOW
					factor = {
						value = pp_building_efficiency_map_value
						add = 1
						divide = 0.5
						max = 1
						min = 0
					}
				}
			}
			else_if = {
				limit = { pp_building_efficiency_map_value < -0.1 }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_LOW
					max_color = define:NMapColors|MAP_COLOR_MID
					factor = {
						value = pp_building_efficiency_map_value
						add = 0.5
						divide = 0.4
						max = 1
						min = 0
					}
				}
			}
			else_if = {
				limit = { pp_building_efficiency_map_value < 0.1 }
				value = define:NMapColors|MAP_COLOR_MID
			}
			else_if = {
				limit = { pp_building_efficiency_map_value < 0.5 }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MID
					max_color = define:NMapColors|MAP_COLOR_HIGH

	legend_key = { desc = "MAPMODE_PP_BUILDING_EFFICIENCY_NEUTRAL" color = define:NMapColors|MAP_COLOR_MID }
	legend_key = { desc = "MAPMODE_PP_BUILDING_EFFICIENCY_GOOD" color = define:NMapColors|MAP_COLOR_HIGH }
	legend_key = { desc = "MAPMODE_PP_BUILDING_EFFICIENCY_STRONG" color = define:NMapColors|MAP_COLOR_MAX }
	tooltip_key = {
		if = { limit = { is_land = yes } value = MAPMODE_PP_BUILDING_EFFICIENCY_TT_LAND }
