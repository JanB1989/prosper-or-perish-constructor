# Population-delta debug map modes.
# Color = percent change vs snapshotted baseline: (current / saved) - 1.
# Continuous traffic-light: -100% (MIN) through 0% (MID) to +200% (MAX).
# Negative band -50%..0% lerps LOW->MID so medium losses stay visible.
# Snapshot baselines with: event pp_population_baseline_debug.1

@pp_pop_delta_floor = -1
@pp_pop_delta_neg_strong = -0.5
@pp_pop_delta_zero = 0
@pp_pop_delta_pos_small = 0.1
@pp_pop_delta_pos_strong = 0.5
@pp_pop_delta_ceiling = 2

@zoom_step_near = 2
@data_gradient_alpha_inside = 1
@data_gradient_alpha_outside = 1
@data_gradient_width = 0.25
@data_gradient_color_mult = 0.9
@data_edge_width = 0
@data_edge_sharpness = 0.01
@data_edge_alpha = 0
@data_edge_color_mult = 0
@data_before_lighting_blend = 0.5
@data_after_lighting_blend = 0.5
pp_pop_delta_location = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_variable = pp_pop_baseline }
					var:pp_pop_baseline <= 0
				}
			}
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = { pp_pop_delta_location_pct < @pp_pop_delta_floor }
			value = define:NMapColors|MAP_COLOR_MIN
		}
		else_if = {
			limit = { pp_pop_delta_location_pct < @pp_pop_delta_neg_strong }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_MIN
				max_color = define:NMapColors|MAP_COLOR_LOW
				factor = {
					value = pp_pop_delta_location_pct
					add = 1
					divide = 0.5
					max = 1
					min = 0
				}
			}
		}
		else_if = {
			limit = { pp_pop_delta_location_pct < @pp_pop_delta_zero }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_MID
				factor = {
					value = pp_pop_delta_location_pct
					add = 0.5
					divide = 0.5
					max = 1
					min = 0
				}
			}
		}
		else_if = {
			limit = { pp_pop_delta_location_pct < @pp_pop_delta_pos_small }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_MID
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = {
					value = pp_pop_delta_location_pct
					divide = 0.1
					max = 1
					min = 0
				}
			}
		}
		else_if = {
			limit = { pp_pop_delta_location_pct < @pp_pop_delta_pos_strong }
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_HIGH
				max_color = define:NMapColors|MAP_COLOR_MAX
				factor = {
					value = pp_pop_delta_location_pct
					subtract = 0.1
					divide = 0.4
					max = 1
					min = 0
				}
			}
		}
		else_if = {
			limit = { pp_pop_delta_location_pct < @pp_pop_delta_ceiling }
			value = define:NMapColors|MAP_COLOR_MAX
		}
		else = {
			value = define:NMapColors|MAP_COLOR_MAX
		}
	}
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_MINUS_100" color = define:NMapColors|MAP_COLOR_MIN }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_MINUS_50" color = define:NMapColors|MAP_COLOR_LOW }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_MINUS_25" color = define:NMapColors|MAP_COLOR_MID }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_MINUS_10" color = define:NMapColors|MAP_COLOR_MID }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_0" color = define:NMapColors|MAP_COLOR_MID }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_PLUS_10" color = define:NMapColors|MAP_COLOR_HIGH }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_PLUS_25" color = define:NMapColors|MAP_COLOR_HIGH }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_PLUS_50" color = define:NMapColors|MAP_COLOR_MAX }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_PLUS_100" color = define:NMapColors|MAP_COLOR_MAX }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_PLUS_200" color = define:NMapColors|MAP_COLOR_MAX }
	legend_key = { desc = "MAPMODE_PP_POP_DELTA_NO_BASELINE" color = rgb { 128 128 128 } }
	tooltip_key = {
		if = { limit = { is_land = yes } value = MAPMODE_PP_POP_DELTA_LOCATION_TT_LAND }
		else = { value = MAPMODE_PP_POP_DELTA_TT_WATER }
	}
	small_map_names = location
	medium_map_names = location
	large_map_names = location
	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location
	fill_in_impassable = yes
	enable_snow = no
	flatmap_behaviour = always
	use_fow = no
	category = debug
	index = 0
	allow_allocate_hotkey = no
	map_markers = { all = no }
	gradient_parameters = {
		zoom_step = @zoom_step_near
		gradient_alpha_inside = @data_gradient_alpha_inside
		gradient_alpha_outside = @data_gradient_alpha_outside
		gradient_width = @data_gradient_width
		gradient_color_mult = @data_gradient_color_mult
		edge_width = @data_edge_width
		edge_sharpness = @data_edge_sharpness
		edge_alpha = @data_edge_alpha
		edge_color_mult = @data_edge_color_mult
		before_lighting_blend = @data_before_lighting_blend
		after_lighting_blend = @data_after_lighting_blend
	}
	color_refresh_counters = { LocationPopulationChanged }
	color_and_names_refresh_counters = { LocationOwnerChanged CountryStatus }
}
