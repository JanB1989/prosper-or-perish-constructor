NMarket = {
	# Order: navigable river > coastal sea > open sea > small native river > land. Navigable rivers are sea zones,
	# so their step is SEA x (1 + the pp_navigation road's market_access), set in config/river_navigation.json.
	# Every location starts above full access; distance takes it down and the unsupported building levels
	# (local_market_access per level over the supported ones) eat into it. Logistics buildings add it back.
	MARKET_BASE_ACCESS = 1.3 # 1
	MARKET_BASE_DISTANCE_FACTOR = 0.006 # 0.005; land carries market access poorly
	MARKET_OPEN_SEA_DISTANCE_FACTOR = 0.4 # 1.5
	MARKET_SEA_DISTANCE_FACTOR = 0.2 # 0.75
	MARKET_SEA_TO_LAND_DISTANCE_FACTOR = 0.3 # 0.5
	# Native rivers left after the navigation export are small streams: a mild help only.
	MARKET_DOWNSTREAM_FACTOR = 0.5 # 0.5
	MARKET_UPSTREAM_FACTOR = 0.8 # 0.9
	MARKET_NO_PORT_EXTRA_DISTANCE = 0.1 # 0.1; navigation banks carry river ports
	MARKET_PORT_EXTRA_DISTANCE = 0.02 # 0.02, scaled by 1 - harbor suitability
	# Prices count trade fully on both sides (vanilla: 75 % of trade and 25 % / 10 % of burgher-trade exports / imports).
	# Exports: a neighbour's shortage bought from this market raises its price and reaches its producers.
	# Imports: a market whose need is covered by imports sees it in its price and stops building its own copy of the
	# supply, so production concentrates in the exporting markets (specialisation, long run 2026-10-01).
	TRADE_IMPACT_ON_DEMAND_SCALE = 1.0 # 0.75
	BURGHER_TRADE_IMPACT_ON_DEMAND_SCALE = 1.0 # 0.25
	TRADE_IMPACT_ON_SUPPLY_SCALE = 1.0 # 0.75
	BURGHER_TRADE_IMPACT_ON_SUPPLY_SCALE = 1.0 # 0.1
	MIN_PRICE_IMPACT = -0.5
 	MAX_PRICE_IMPACT = 1.5 # 6
	# Market food (tested 2026-09-24), each month:
	#   FOOD_PROVINCE_LOCAL_RESERVE_RATIO: a province hands its stored food above ratio x its food capacity to the
	#     market. 1000 = never.
	#   FOOD_CAPACITY_FACTOR: scales market food capacity (factor x about the sum of its provinces' food capacity).
	#     0 = markets hold nothing, so no province can buy market food and markets trade none between them.
	#   Pass 1: provinces with a food shortage get their need from the market first. The two defines below never
	#     affect pass 1.
	#   Pass 2: other provinces refill their storage, but only from market stock above
	#     FOOD_PROPORTION_TO_PRESERVE_IN_MARKET_IF_POSSIBLE x market food capacity (vanilla 0.25), and each province
	#     takes at most FOOD_PASS2_MAX_MONTHS_OF_DEFICIT x its monthly deficit (vanilla 3). If that product is 0
	#     (no deficit, or the define is 0) there is no cap.
	#   FOOD_PRICE: an empty market's food price sits exactly at it. The food slider bills
	#     needed food x FOOD_PRICE x slider, even when the market delivers nothing.
	#   Goods with a food value carry market food when traded; from an empty market those trades never run.
	#     All food goods here have food = 0 (goods/pp_goods_adjustments.txt), so they trade as normal goods.
 	FOOD_CAPACITY_FACTOR = 0 # 1.0; markets hold no food: no province buys it, markets trade none
 	FOOD_PROVINCE_LOCAL_RESERVE_RATIO = 1000 # 0.75; provinces never hand food to the market
 	MONTHLY_PRICE_CHANGE = 0.15 # 0.04
 	FOOD_PRICE = 0.10 # 0.001; empty markets sit at max(FOOD_PRICE, 0.01) (0.001 gave 0.0099 in every market, 1339):
 	# 0.10 = Province Food at its floor. The AI values location food at this price; the slider bill stays ~0
 	# (food_purchase_efficiency 10000 in auto_modifiers/pp_country_base_values.txt).
 	FOOD_PRICE_IMPACT_ON_PRICES = 0.00
 	MARKET_FOOD_STOCKPILE_TRESHOLD_MONTHS = 24.0
	MARKET_MIN_FOOD_STOCKPILE_RATIO_FOR_BURGHER_TRADE = 0 # 0.25; EU5 1.4: burghers are never held back from food trade by the (always empty) market food stock
}
