# Gathering of the Russian Lands (docs/country_finetunes.md): every country ruling from the Russian region is a war
# target of Muscovy/Russia, even where it is not a land neighbour.
pp_gathering_russian_lands_targets = {
	attacker_potential = {
		OR = {
			tag = MOS
			tag = RUS
		}
		months_since_war >= 36
	}
	candidate_list = {
		every_country = {
			limit = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= { region = region:russian_region }
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 30
	}
}
