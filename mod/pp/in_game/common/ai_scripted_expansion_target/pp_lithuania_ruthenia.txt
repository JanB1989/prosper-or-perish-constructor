# Lithuania takes Ruthenia and Smolensk (docs/country_finetunes.md): every country ruling from Ruthenia or the Smolensk
# area is a war target of Lithuania, even where it is not a land neighbour, after 36 months of peace.
pp_lithuania_ruthenia_targets = {
	attacker_potential = {
		tag = LIT
		months_since_war >= 36
	}
	candidate_list = {
		every_country = {
			limit = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= {
					OR = {
						region = region:ruthenia_region
						area = area:smolensk_area
					}
				}
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 30
	}
}
