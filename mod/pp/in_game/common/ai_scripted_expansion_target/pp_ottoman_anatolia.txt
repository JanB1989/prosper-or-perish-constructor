# Ottoman aim (docs/country_finetunes.md): every country ruling from Anatolia is a war target, even where it is not a
# land neighbour.
pp_ottoman_anatolia_targets = {
	attacker_potential = {
		tag = TUR
		months_since_war >= 36
	}
	candidate_list = {
		every_country = {
			limit = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= { region = region:anatolia_region }
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 30
	}
}

# Once the power base stands (Constantinople + size, scripted_triggers/pp_ottoman_triggers.txt): every country
# ruling from the Levant or Egypt (the Mamluks) is a war target. Twelve months of peace: Egypt takes several wars.
pp_ottoman_mamluk_targets = {
	attacker_potential = {
		tag = TUR
		months_since_war >= 12
		pp_ottoman_power_base = yes
	}
	candidate_list = {
		every_country = {
			limit = {
				NOT = { this = root }
				NOT = { is_subject_of = root }
				capital ?= {
					OR = {
						region = region:egypt_region
						region = region:crescent_region
					}
				}
			}
			add_to_list = source
		}
	}
	score = {
		add = scope:target.country_tax_base
		multiply = 10
		add = 60
	}
}
