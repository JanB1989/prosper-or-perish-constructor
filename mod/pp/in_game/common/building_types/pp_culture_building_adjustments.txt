# Prosper or Perish - cultural building promotion and maintenance adjustments.

REPLACE:library = {
	audio_tier = 2
	is_foreign = no
	pop_type = clergy
	max_levels = pp_library_max_level # education pyramid 2026-11-27 (Jan): 1, town 2, city 3; two allow a University

	category = cultural_category
	employment_size = cultural_employment
	custom_tags = { pp_education_priority }

	AI_ignore_available_worker_flag = yes

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	forbidden_for_estates = yes

	graphical_tags = { medium_religious }

	build_time = cultural_building_time

	# AI (literacy rework 2026-11-25): the literacy cap of nobles, burghers and clergy (pp_library_ai_weight, more of
	# them weigh more), like the other PP weights divided by the owner's income. Until then the weight read open laborer
	# jobs (pp_promotion_signal x 200): the promotion moved to the Village School.
	ai_construct_weight = {
		value = pp_library_ai_weight
		divide = { value = scope:owner.monthly_income_total add = 10 }
	}

	# No allow: vanilla 1.4's only allow forbids a library in Cordoba next to library_cordoba, a building of the
	# unreleased DLC d009_across_the_pillars. Without the DLC the building does not exist, the link fails at load
	# ("Failed to find 'library_cordoba' ... for link 'building_type'") even inside the has_dlc limit, and the gate can
	# never apply. tests/test_building_type_references.py keeps DLC-only links out of the mod.

	# literacy rework 2026-11-25 (Jan): the cap only for the upper classes (vanilla local_max_literacy 5 for all), monthly
	# literacy 0.1 -> 0.02; the promotion speed went to the Village School, the burghers' room to the Trade School, the
	# laborers' 0.6 and clergy's 0.1 room were dropped.
	modifier = {
		local_cultural_influence = 0.1
		local_nobles_max_literacy = 5
		local_burghers_max_literacy = 5
		local_clergy_max_literacy = 5
		local_monthly_literacy = 0.008 # 0.02 until 2026-11-27 (every literacy speed x0.4)
		local_institution_growth_modifier = 0.25
	}

	possible_production_methods = {
		pp_library_maintenance
	}

	construction_demand = library_construction
	can_close = no # building_shutdown
	ai_forbid_shutdown = yes # building_shutdown
}

REPLACE:university = {
	audio_tier = 3
	is_foreign = no
	pop_type = clergy
	max_levels = 1

	category = cultural_category
	employment_size = religious_building_employment
	custom_tags = { pp_education_priority }

	town = yes
	city = yes
	megalopolis = yes

	AI_ignore_available_worker_flag = yes

	expensive = yes

	build_time = large_cultural_building_time

	# AI: pp_university_ai_weight (script_values/pp_school_values.txt): the promotion term, plus a push once the owner
	# can build the Academy of Sciences.
	ai_construct_weight = {
		value = pp_university_ai_weight
		divide = { value = scope:owner.monthly_income_total add = 10 }
	}

	forbidden_for_estates = yes

	allow = {
		is_core_of = owner
		owner = { modifier:university_construction_blocked = no }
		# education pyramid 2026-11-27 (Jan): a University grows out of a place's Libraries. allow is read when it is
		# built, never for a standing one, so the Universities of the game start stay.
		pp_library_levels >= 2
	}

	modifier = {
		# 2026-11-27 (Jan): the Library sets how high the estates' literacy can go and helps institutions take hold; the
		# University sets how fast literacy rises: its speed is about four Library levels' (the Library's is 0.008 a
		# level), no institution growth (was 0.33), and half the cap it had (burghers and nobles 5, was 10; Jan the same
		# night: it should still give some).
		local_cultural_influence = 0.5
		local_burghers_max_literacy = 5
		local_nobles_max_literacy = 5
		local_monthly_literacy = 0.03
		local_migration_attraction = 0.05
		local_pop_promotion_speed = 0.005
		local_laborers_desired_pop = 1.2
		local_burghers_desired_pop = 0.6
		local_clergy_desired_pop = 0.3
		local_nobles_desired_pop = 0.2
	}

	unique_production_methods = {
		pp_university_maintenance = {
			icon_type = building_type
			icon = university
			paper = 1.0
			books = 0.5
			glass = 0.5
			beeswax = 0.1

			category = building_maintenance
		}
	}

	construction_demand = university_construction
	can_close = no # building_shutdown
	ai_forbid_shutdown = yes # building_shutdown
}
