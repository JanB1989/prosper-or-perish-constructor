# Prosper or Perish - schools (literacy rework 2026-11-25, Jan; hand-written like the Library and University in
# pp_culture_building_adjustments.txt). Literacy is earned in the location: the Village School gives the lower classes
# (peasants, laborers, soldiers) their literacy cap, the Trade School the burghers'. Both are open to the estates whose
# people work in them (an estate pays for buildings of its own pop type: clergy for the Village School, burghers for the
# Trade School; docs/mod_authoring_rules.md "Estates may build every production building").
# The AI values class literacy caps at about zero (private notes literacy_ai_149.md), so each school carries a positive
# ai_construct_weight (script_values/pp_school_values.txt): bigger places first, divided by the owner's income like the
# Library's. Levels: script_values/pp_school_values.txt. Upkeep: production_methods/pp_culture_maintenance_adjustments.txt.
# Icons are vanilla placeholders until the drawn ones are installed (icon = ... here and on the maintenance methods).

pp_village_school = {
	audio_tier = 1
	is_foreign = no
	pop_type = clergy
	max_levels = pp_village_school_max_level

	category = cultural_category
	employment_size = religious_building_employment # 0.05, half the Library's cultural_employment
	custom_tags = { pp_education_priority }

	AI_ignore_available_worker_flag = yes

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	forbidden_for_estates = no

	build_time = cultural_building_time

	ai_construct_weight = {
		value = pp_village_school_ai_weight
		divide = { value = scope:owner.monthly_income_total add = 10 }
	}

	modifier = {
		local_peasants_max_literacy = 10 # 5 until 2026-11-26: commoners get no country-wide literacy any more
		local_laborers_max_literacy = 10
		local_soldiers_max_literacy = 10
		local_monthly_literacy = 0.004 # 0.01 until 2026-11-27 (every literacy speed x0.4)
		local_pop_promotion_speed = 0.002 # moved from the Library (2026-11-25)
		# Spare room for laborers: 0.2 idle laborers per level (2026-11-28, Jan: "it's fine if it gives some
		# laborers"; the line is x 5 because flat room counts a fifth under the job room rule). 0.8 per level and 0.1
		# soldiers from 11-27 parked 11.9 M idle laborers and 1.5 M soldiers by 1837 (run bd7d9616, 14,875 school
		# levels); at 0.2 that is about 3 M. No soldier room. The RGO's spare half place stays the main spare room.
		local_laborers_desired_pop = 1 # 0.2 x 5 (job room: flat room counts a fifth)
	}

	possible_production_methods = {
		pp_village_school_maintenance
	}

	construction_demand = library_construction
	can_close = no # building_shutdown
	ai_forbid_shutdown = yes # building_shutdown
}

pp_trade_school = {
	audio_tier = 1
	is_foreign = no
	pop_type = burghers
	max_levels = pp_trade_school_max_level

	category = cultural_category
	employment_size = small_burgher_employment # 0.05
	custom_tags = { pp_education_priority }

	AI_ignore_available_worker_flag = yes

	town = yes
	city = yes
	megalopolis = yes

	forbidden_for_estates = no

	build_time = cultural_building_time

	# education pyramid 2026-11-27 (Jan): the burghers' school stands on the commoners' school.
	allow = {
		has_building = building_type:pp_village_school
	}

	ai_construct_weight = {
		value = pp_trade_school_ai_weight
		add = pp_spare_room_weight_burghers # spare room (2026-11-27): the town lacks idle burghers its buildings need
		divide = { value = scope:owner.monthly_income_total add = 10 }
	}

	modifier = {
		local_burghers_desired_pop = 2.7 # job_room base 2.5
		local_burghers_max_literacy = 10
		local_monthly_literacy = 0.004 # 0.01 until 2026-11-27
		# spare room (2026-11-27, Jan: at least a building): two guild levels (0.25 each) per school level; 0.3 since
		# 2026-11-25, when it moved here from the Library. A typical town promotes about one burgher a month.
	}

	possible_production_methods = {
		pp_trade_school_maintenance
	}

	construction_demand = library_construction
	can_close = no # building_shutdown
	ai_forbid_shutdown = yes # building_shutdown
}

# Academy of Sciences (education pyramid 2026-11-27, Jan): the top of the pyramid, in the capital. Shown to the player as the
# "Society of Sciences" (Jan: vanilla has two event buildings named Academy of Sciences, Russia's and Bavaria's). One level per three
# Universities of the country (pp_academy_of_sciences_max_level, at most ten); each staffed level gives the whole
# country research speed (capital_country_modifier: applied for the current capital, scaled by level, staffing and
# upkeep bought). Unlocked by the advance pp_academy_of_sciences_advance (Scientific Revolution tree).
# location_potential like vanilla's Chancery: a capital building, shown only in the capital, gone when the capital
# moves (remove_if); the project rule against owner gates in location_potential is about buildings that should survive
# an owner change, this one should not. The engine values research speed in a capital block very highly, so it carries
# no AI weight. No pp_education_priority tag: staffed like any other building.
pp_academy_of_sciences = {
	audio_tier = 3
	is_foreign = no
	pop_type = clergy
	max_levels = pp_academy_of_sciences_max_level

	category = cultural_category
	employment_size = religious_building_employment

	AI_ignore_available_worker_flag = yes

	town = yes
	city = yes
	megalopolis = yes

	expensive = yes

	forbidden_for_estates = yes

	build_time = large_cultural_building_time

	location_potential = {
		counts_as_capital = yes
	}

	allow = {
		is_capital = yes
		country_has_no_other_copy_of_building = { type = pp_academy_of_sciences }
	}

	remove_if = {
		is_no_longer_capital = yes
	}

	modifier = {
		local_cultural_influence = 0.25
	}

	capital_country_modifier = {
		research_speed_modifier = 0.02
		# 2026-11-28 (Jan: a small country-wide bonus, never the country's main source): half a point of maximum
		# literacy for every class per staffed level, five points at the ten levels thirty Universities allow. Run
		# bd7d9616: nine countries built it, six levels at most.
		global_nobles_max_literacy = 0.5
		global_clergy_max_literacy = 0.5
		global_burghers_max_literacy = 0.5
		global_peasants_max_literacy = 0.5
		global_laborers_max_literacy = 0.5
		global_soldiers_max_literacy = 0.5
	}

	possible_production_methods = {
		pp_academy_of_sciences_maintenance
	}

	construction_demand = university_construction
	can_close = no # building_shutdown
	ai_forbid_shutdown = yes # building_shutdown
}
