REPLACE:hanseatic_shipwright_guild = {
# Vanilla building; ppc build fills in raw_modifier local_population_capacity from the footprint class.
# The League's shipwrights in a member port (2026-11-25, Jan: levels from the town's development and the League's
# advances, dear to build, cheap to keep, and the town gains too; AGENTS.md "Hansa Military Buildings"). Vanilla:
# one level (two with hsa_hanseatic_shipwrights), 1,000 laborers a level, the age price, port_building_maintenance.
audio_tier = 3
is_special = yes
is_foreign = yes
need_good_relation = yes

pop_type = laborers
# 1 + one per 20 development of the town + the owner's hanseatic_shipwright_guild_max_level (two advances), 5 at most
max_levels = pp_hanseatic_shipwright_guild_max_level
category = military_category
country_potential = {
	modifier:allows_hanseatic_federation_buildings = yes
}
location_potential = {
	is_port = yes
}
# membership gates new levels only (vanilla had it in location_potential, which the engine checks again when the
# location changes owner and then removes the building): a guild that stands stays while its town is at peace with us
allow = {
	custom_tooltip = {
		text = pp_hansa_member_town_tt
		owner = { is_subject_of = scope:actor }
	}
}
remove_if = {
	OR = {
		location = { has_owner = no }
		location.owner ?= { is_at_war_with = root.owner }
	}
}
# 250 shipwrights a level (vanilla dock_employment 1,000): five levels must not take half a town's laborers
employment_size = 0.25
pop_size_created = 0.25
town = yes
city = yes
megalopolis = yes
build_time = medium_port_building_time

price = pp_hanseatic_shipwright_guild_price
increase_per_level_cost = 1.00 # x [building_scaling] increase_per_level_cost_multiplier 0.75 = 1.0 in the build: each level adds the first level's price again

construction_demand = port_building_construction

# owner: one ship at a time and sailors per level as in vanilla, and the dues on the goods shipped through the port
# (the League paid its wars from a levy on shipped goods); town: harbour and repair as in vanilla, sailors of its
# own and a little development
modifier = {
	local_laborers_desired_pop = 1 # job_room
	max_ships_built_at_same_time = 1
	sailors_to_building_owner = 0.01
	gold_to_building_owner = 0.5
	can_build_ships_in_this_location = yes
	harbor_suitability = 0.1
	local_repair_speed = 0.01
	local_sailors = 0.005
	local_monthly_development = 0.002
}

# The League at home carries its houses abroad (2026-11-25, Jan: the Hansa's own buildings are the vehicle of its
# growth): two guild levels let its owner keep one more merchant house (pp_merchant_house_cap reads the modifier;
# the engine scales this block by level and goods access; 1 per level while the guild had one or two levels).
# AGENTS.md "Merchant Houses".
foreign_country_modifier = {
	pp_merchant_houses = 0.5
}

# each level supports itself, so foreign levels never count as unsupported building levels of the location
raw_modifier = {
	free_building_levels = 1
}

# the engine values the guild by its sailors and its ship slot alone and the League is poor; / (builder income + 10)
# as for the merchant houses (scope:owner is the host here)
ai_construct_weight = {
	value = 100
	divide = { value = scope:builder.monthly_income_total add = 10 }
}

unique_production_methods = {
	pp_hanseatic_shipwright_guild_maintenance = {
		icon_type = building_type
		icon = hanseatic_shipwright_guild
		# a quarter of vanilla's port_building_maintenance per level (2026-11-25, Jan: dear to build, cheap to keep;
		# one vanilla level cost the League 0.5 to 1.4 gold a month in run 5d232536, on an income of 13 to 20; a
		# third until 2026-11-27, Jan: the League's upkeep a bit lower)
		tar = 0.05 # vanilla 0.2
		naval_supplies = 0.1 # 0.4
		tools = 0.025 # 0.1
		lumber = 0.05 # 0.2
		stone = 0.025 # 0.1
		category = building_maintenance
	}
}
	can_close = no # building_shutdown
	ai_forbid_shutdown = yes # building_shutdown
}
