TRY_REPLACE:develop_province = {
	ability = adm
	helps_with = "location|subject"

	allow_multiple = yes

	potential = {
		num_locations > 0
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		ai_interaction_source_list = {
			scope:actor = {
				capital.province = {
					add_to_list = source
				}
			}
		}
		name = "develop_province_select_province"
		none_available_msg_key = "develop_province_no_provinces"
		column = {
			data = name
		}
		column = {
			data = development
		}
		column = {
			data = population
		}
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = develop_province interaction_target = target } }
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = {
					value = development
					divide = 100
				}
			}
		}
	}

	location_modifier = {
		local_monthly_development = 0.00125
	}

	progress = {
		scope:target = {
			value = province_average_development
			divide = 100
		}
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		add = {
			if = {
				limit = {
					scope:actor = {
						is_subject = yes
					}
				}
				value = -1000
			}
		}
	}
}
