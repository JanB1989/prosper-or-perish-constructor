# Prosper or Perish - Stored Food (EU5 1.4).
# EU5 1.4 deleted the engine-scaled stored-food modifier. Its effects per stored year (population growth, devastation
# recovery, migration attraction, prosperity) ride the province static modifier pp_stored_food, applied with size =
# the province's stored years; the store lever (Province Food, staple and Surplus Sales output) rides pp_low_stores
# and pp_full_stores, applied with size = the years the store is below / above 12 months (all generated from
# [stored_food] in constructor.toml). The engine's own storage growth term is off (NPop.FOOD_STORAGE_POP_GROWTH = 0).
#
# Exception to the mod's "no pulsing scripts" rule, approved by Jan on 2026-10-01 after the in-game cost test: no
# engine mechanism follows a province's stored food in 1.4, so each country refreshes its provinces' tiers once a
# month. The vanilla monthly country pulse fires this with a random 0-29 day delay, so the countries spread over the
# month. Measured over one game year with the whole-month tiers this replaced (4,071 provinces): about 700 modifier
# changes a month, 0.56 s per game year (about 12 us per province and month, at most 5 ms for one country). The scaled
# modifier works out the consumption once per province (the tiers did it twice) and re-applies only when the stored
# months moved by more than the deadband. Do not add further pulse scripts on this basis.
# Game start: pp_stored_food_game_start (pp_game_start.txt) sets every province's modifier on day one.

monthly_country_pulse = {
	on_actions = {
		delay = { days = { 0 29 } }
		pp_stored_food_country_refresh
	}
}

# Country scope: every_province = the country's provinces.
pp_stored_food_country_refresh = {
	effect = {
		every_province = {
			pp_refresh_stored_food = yes
		}
	}
}
