# Disband Kontor (2026-11-27, Jan: the Hansa should not always die; AGENTS.md "Hansa Survival"): vanilla's disband_kontor
# copied whole, PP's lines are marked "PP". One demand closes one Kontor and costs 0.75 x the peace cost of its town.
# Any winner may still close every Kontor of the loser (Jan: it must stay possible), but a Kontor standing in a town
# that neither the winner nor one of its subjects owns costs 2 x the town's peace cost, so one lost war no longer
# takes the League's whole network (observer runs 5023bdba, 5d232536, defb89f2). Not tested in game.
# Port it on game updates (vanilla pin, tests/vanilla_replace_pins.csv).
REPLACE:disband_kontor = {

	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = "scope:target.location_peace_cost(scope:loser|scope:winner)"
			# PP: vanilla 0.75 in a town of the winner or of one of its subjects, 2 in any other town
			if = {
				limit = {
					scope:target = {
						owner ?= {
							OR = {
								this = scope:winner
								is_subject_of = scope:winner
							}
						}
					}
				}
				multiply = 0.75
			}
			else = {
				multiply = 2
			}
		}
	}

	base_antagonism = 1
	antagonism_type = antagonism_disband_kontor

	potential = {
		scope:war = {
			or = {
				has_casus_belli = no
				casus_belli = {
					is_trade_cb = no
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = location
		source = recipient
		target_flag = target
        visible = {
			location_building_level = {
				building_type = building_type:hanseatic_kontor
				value > 0
				owner = scope:recipient
			}
		}
    }

	allow = {
	}

	effect = {
		scope:target = {
			if = {
				limit = {
					location_building_level = {
						building_type = building_type:hanseatic_kontor
						value > 0
						owner = scope:recipient
					}
				}
				destroy_building_forcefully = "building(building_type:hanseatic_kontor|scope:recipient)"
			}
		}
	}

	ai_desire = {
		value = 10
	}
}
