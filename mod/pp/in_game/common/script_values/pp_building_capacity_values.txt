# Prosper or Perish - static building capacity baselines
# These values are cached into location variables at game start. Keep only
# static geography/RGO inputs here; live size and building pressure live in
# pp_building_caps.txt.

pp_fish_base_capacity_value = {
	value = 0
	if = {
		limit = {
			is_ownable = yes
			raw_material = goods:fish
		}
		add = 3.00
	}
	if = {
		limit = { pp_is_sea_coast = yes }
		add = 4.50
	}
	if = {
		limit = { is_adjacent_to_lake = yes }
		add = 3.00
	}
	if = {
		limit = { topography = wetlands }
		add = 1.50
	}
}

pp_forest_base_capacity_value = {
	value = 0
	if = {
		limit = {
			is_ownable = yes
			OR = {
				raw_material = goods:lumber
				raw_material = goods:fur
				raw_material = goods:wild_game
			}
		}
		add = 4.4
	}
	if = {
		limit = { OR = { vegetation = forest vegetation = ha1300_veg_coniferous_forest vegetation = ha1300_veg_mixed_forest vegetation = ha1300_veg_swamp } }
		add = 6.6
	}
	else_if = {
		limit = { OR = { vegetation = woods vegetation = ha1300_veg_thicket vegetation = ha1300_veg_dry_forest } }
		add = 4.4
	}
	else_if = {
		limit = { vegetation = jungle }
		add = 3.3
	}
}
