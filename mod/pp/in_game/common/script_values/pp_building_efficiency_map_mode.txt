# Script value for the building efficiency map mode.
pp_building_efficiency_map_value = {
	value = modifier:local_build_buildings_efficiency
}

# The engine compares all levels with an allowance: local free building levels x (1 + the owner's
# global_free_building_levels_modifier), never below 0. Since 2026-11-24 (Jan) every country has +100 % from its base
# values and building_levels takes 0.5 per staffed level, so the allowance is twice the free levels minus the staffed
# levels. In level terms, as the location view shows it: limit = (allowance + staffed) / 2,
# counted = (levels + staffed) / 2; staffed = modifier:local_pp_staffed_building_levels (static modifier building_levels).

# The limit: how many fully staffed levels the location supports.
pp_supported_building_levels_map_value = {
	value = 1
	owner ?= { add = modifier:global_free_building_levels_modifier }
	multiply = modifier:free_building_levels
	min = 0
	add = modifier:local_pp_staffed_building_levels
	divide = 2
}

# The levels counted toward the limit: a fully staffed level as one, a closed or unstaffed level as half.
pp_counted_building_levels_map_value = {
	value = total_building_levels
	add = modifier:local_pp_staffed_building_levels
	divide = 2
}

# Levels over the limit. The engine applies unsupported_building_levels at twice this (a staffed level over counts
# twice at half the penalty), so one staffed level over costs 0.01 market access.
pp_unsupported_building_levels_map_value = {
	value = pp_counted_building_levels_map_value
	subtract = pp_supported_building_levels_map_value
	min = 0
}
