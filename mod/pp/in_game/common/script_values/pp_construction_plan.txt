# Prosper or Perish - Construction Plan values (player-only construction rules). The settings with their defaults
# (pp_cplan_reserve_gold, pp_cplan_access_pct, pp_cplan_improve_pct, pp_cplan_business_years), the food rule's numbers,
# the farm and business profit estimates and the build cost price factors are generated from [construction_plan] into
# pp_construction_plan_generated.txt. Variables carry + 100.

# ---- The month's gold (country scope) ----

# Gold above the reserve.
pp_cplan_free_gold = {
	value = gold
	subtract = pp_cplan_reserve_gold
}

# The gold above the reserve at the pass's start (0 outside a pass); each rule group's part of it is generated
# (pp_cplan_share_<group>: its weight over the sum of the weights, pp_construction_plan_generated.txt).
pp_cplan_pot_n = {
	value = 0
	if = {
		limit = { has_variable = pp_cplan_pot }
		value = var:pp_cplan_pot
		subtract = 100
	}
}
# Gold the pass may have spent by the end of the current rule group (0 outside a pass).
pp_cplan_cap_n = {
	value = 0
	if = {
		limit = { has_variable = pp_cplan_cap }
		value = var:pp_cplan_cap
		subtract = 100
	}
}

# Location scope: the owner's logistics line as a market access fraction.
pp_cplan_access_line = {
	value = 0
	owner ?= { add = pp_cplan_access_pct }
	divide = 100
}

# ---- Baseline rules: the food numbers as a normal year would have them (province scope; Jan 2026-11-08) ----
# Nothing is remembered from pass to pass. Short-lived effects are taken out instead: this year's Variable Harvest
# (its peasant food consumption, pp_cplan_harvest_rate) and spoilage. The rules read the store a year from now.

# Peasants of the province (thousands, the unit of the pops' food consumption).
pp_cplan_province_peasants = {
	value = 0
	every_location_in_province = {
		add = num_pop_type:peasants
	}
}
# Food the peasants eat more this month because of this year's harvest (less in a good year).
pp_cplan_harvest_food = {
	value = 0
	province_capital ?= {
		add = pp_cplan_harvest_rate
	}
	multiply = pp_cplan_province_peasants
}
# Food the province eats a month in a normal year.
pp_cplan_normal_consumption = {
	value = pp_stored_food_province_consumption
	subtract = pp_cplan_harvest_food
	min = 0.001
}
# Share of the store that spoils a month (EU5 1.4: the local and the owner's decay over 1 + the preservation of every
# location of the province, at least 1).
pp_cplan_spoil_divisor = {
	value = 1
	every_location_in_province = {
		add = modifier:local_food_preservation_efficiency_modifier
	}
	min = 1
}
pp_cplan_spoil_rate = {
	value = 0
	province_capital ?= {
		add = modifier:local_food_decay
		owner ?= { add = modifier:global_food_decay }
	}
	divide = pp_cplan_spoil_divisor
	min = 0
}
# Months of a normal year's consumption the store holds now (not capped).
pp_cplan_store_months_now = {
	value = province_food
	divide = pp_cplan_normal_consumption
}
# What the store gains a month in a normal year, in months of consumption: the food made minus the food eaten
# (pp_province_food_balance) with this year's harvest taken out, minus what spoils of the store it holds now.
pp_cplan_normal_gain = {
	value = pp_province_food_balance
	add = pp_cplan_harvest_food
	subtract = {
		value = province_food
		multiply = pp_cplan_spoil_rate
	}
	divide = pp_cplan_normal_consumption
}
# The store a year from now in a normal year, in months (the hunger, famine rules read it), never above what the store
# can hold.
pp_cplan_food_in_a_year = {
	value = pp_cplan_normal_gain
	multiply = 12
	add = pp_cplan_store_months_now
	if = {
		limit = { pp_cplan_capacity_months > 0 }
		max = pp_cplan_capacity_months
	}
}
# Province scope: months of the province's consumption its store can hold (0 while nobody eats).
pp_cplan_capacity_months = {
	value = 0
	if = {
		limit = { pp_stored_food_province_consumption > 0 }
		value = province_max_food
		divide = pp_cplan_normal_consumption
	}
}
# Province scope: how full the store is, in %.
pp_cplan_fill_pct = {
	value = province_food_percentage
	multiply = 100
}
# Province scope: the normal month's gain in % of consumption (the Grange rule).
pp_cplan_normal_gain_pct = {
	value = pp_cplan_normal_gain
	multiply = 100
}
# Location scope: victuals' price in this location's market over their default price (0 without a market).
pp_cplan_victuals_price_ratio = {
	value = 0
	if = {
		limit = { market ?= { always = yes } }
		value = "market.market_price(goods:victuals)"
		divide = "default_price(goods:victuals)"
	}
}

# ---- Build cost and payback (location scope; scope:pp_cplan_btype = the building type, the temporary values
# pp_cplan_ipl / pp_cplan_pf / pp_cplan_ef set by pp_cplan_btype_data or the generated pickers) ----
# The engine's formula, calibrated in game on 158 constructions (2026-11-06, all within 1 %): base price
# x (1 + increase_per_level_cost x standing levels) / max(0.5, 1 + build efficiency) x the construction goods' price
# factor. Efficiency = the location's local term + the owner's global term + one category term (urban or rural by the
# location's city status; forts, ports and foreign buildings their own; mills none) + the location's new-building term
# for a building that does not stand yet.

# 1 + the efficiency every building gets here.
pp_cplan_eff_base = {
	value = 1
	add = modifier:local_build_buildings_efficiency
	owner ?= { add = modifier:global_build_buildings_efficiency }
	if = {
		limit = {
			exists = scope:pp_cplan_btype
			NOT = { has_building = scope:pp_cplan_btype }
		}
		add = modifier:local_build_new_buildings_efficiency
	}
}
pp_cplan_eff_std = {
	value = pp_cplan_eff_base
	if = {
		limit = { is_city = yes }
		owner ?= { add = modifier:global_urban_build_buildings_efficiency }
	}
	else = {
		owner ?= { add = modifier:global_rural_build_buildings_efficiency }
	}
	min = 0.5
}
pp_cplan_eff_fort = {
	value = pp_cplan_eff_base
	owner ?= { add = modifier:global_fort_build_buildings_efficiency }
	min = 0.5
}
pp_cplan_eff_port = {
	value = pp_cplan_eff_base
	add = modifier:local_port_build_buildings_efficiency
	owner ?= { add = modifier:global_port_build_buildings_efficiency }
	min = 0.5
}
pp_cplan_eff_foreign = {
	value = pp_cplan_eff_base
	owner ?= { add = modifier:global_foreign_build_buildings_efficiency }
	min = 0.5
}
pp_cplan_eff_mill = {
	value = pp_cplan_eff_base
	min = 0.5
}

# Standing levels and levels under construction of scope:pp_cplan_btype here.
pp_cplan_bt_levels = {
	value = 0
	every_buildings_in_location = {
		limit = { building_type = scope:pp_cplan_btype }
		add = building_level
	}
}
pp_cplan_bt_queued = {
	value = 0
	every_buildings_in_location = {
		limit = { building_type = scope:pp_cplan_btype }
		add = building_levels_under_construction
	}
}

# Monthly profit per level of the standing scope:pp_cplan_btype here.
pp_cplan_bt_profit = {
	value = 0
	every_buildings_in_location = {
		limit = {
			building_type = scope:pp_cplan_btype
			building_level > 0
		}
		add = {
			value = building_profit
			divide = building_level
		}
	}
}

# The gold the next level of scope:pp_cplan_btype costs here.
pp_cplan_est_cost = {
	value = scope:pp_cplan_btype.building_base_cost_in_gold
	multiply = {
		value = pp_cplan_bt_levels
		multiply = scope:pp_cplan_ipl
		add = 1
	}
	divide = scope:pp_cplan_ef
	multiply = scope:pp_cplan_pf
}

# Years the next level of scope:pp_cplan_btype takes to pay back its cost from the profit per level standing here
# (999 without profit; the Victualling Yard step and the location profiles read it).
pp_cplan_est_payback_years = {
	value = 999
	if = {
		limit = { pp_cplan_bt_profit > 0.001 }
		value = pp_cplan_est_cost
		divide = pp_cplan_bt_profit
		divide = 12
	}
	max = 999
}

# Location scope: idle peasants left for a new building here, in thousands like the pops: the idle peasants minus the
# open jobs of the buildings here (workers they could still take). Peasants are promoted into the other classes, so
# they staff any building (Jan 2026-11-09); pp_cplan_staff reads it as scope:pp_cplan_free_peasants. The idle pops
# of the other classes are generated (pp_cplan_idle_<pop type>, pp_construction_plan_generated.txt).
pp_cplan_free_peasants_here = {
	value = "unemployed_pops_of_pop_type_in_location(pop_type:peasants)"
	every_buildings_in_location = {
		subtract = {
			value = building_employment_size_amount
			subtract = building_employed_amount
			min = 0
		}
	}
	min = 0
}
