# Prosper or Perish - schools (literacy rework 2026-11-25, Jan; hand-written). building_types/pp_school_buildings.txt.
# Location scope, scope:owner = the building's owner, as in vanilla's building_caps.txt.
# Every row of a cap is one labelled number (cap_tooltips.py: a row with steps would show as sub-rows), so each
# condition sits in a helper read through this.

# Village School: 1 level with Village Schools (the advance unlocks the building), +1 with the Printing Press advance,
# +1 with Common Schooling (its country modifier pp_village_school_levels), +1 while the owner's Education of the
# Masses law is a policy that teaches the common people (scripted trigger pp_educates_the_masses). At most 4.
pp_village_school_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	add = {
		desc = "PP_BUILDING_LEVEL_PRINTING_PRESS"
		value = this.pp_village_school_printing_press_level
	}
	add = {
		desc = "BUILDING_LEVEL_FROM_OWNER_MODIFIERS"
		value = scope:owner.modifier:pp_village_school_levels
	}
	add = {
		desc = "PP_BUILDING_LEVEL_EDUCATION_POLICY"
		value = this.pp_village_school_education_level
	}
}

pp_village_school_printing_press_level = {
	value = 0
	if = {
		limit = { scope:owner = { has_advance = printing_press_advance } }
		value = 1
	}
}

pp_village_school_education_level = {
	value = 0
	if = {
		limit = { scope:owner = { pp_educates_the_masses = yes } }
		value = 1
	}
}

# Trade School: 1 level with Reckoning Schools (the advance unlocks the building), +1 where the town holds one of the
# mod's specialization town rights, charter or rights (scripted trigger pp_has_specialization_town_rights): +2 since
# 2026-11-27 (Jan), +1 before. At most 3.
pp_trade_school_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	add = {
		desc = "PP_BUILDING_LEVEL_SPECIALIZATION_RIGHTS"
		value = this.pp_trade_school_specialization_level
	}
}

pp_trade_school_specialization_level = {
	value = 0
	if = {
		limit = { pp_has_specialization_town_rights = yes }
		value = 2
	}
}

# AI (ai_construct_weight, divided there by the owner's monthly income + 10 like the Library's): the engine values the
# schools' class literacy caps at about zero, so this flat weight is what builds them. Location values only.
# Village School: 30 + 2 per 1,000 people of the location, at most 150.
pp_village_school_ai_weight = {
	value = population
	multiply = 2
	add = 30
	max = 150
}

# Trade School: 30 + 10 per 1,000 burghers of the location, at most 150.
pp_trade_school_ai_weight = {
	value = num_pop_type:burghers
	multiply = 10
	add = 30
	max = 150
}

# Library (2026-11-25): it now raises the literacy cap of nobles, burghers and clergy and gives no promotion any more,
# so its weight follows those classes: 20 + 10 per 1,000 nobles, burghers and clergy of the location, at most 150.
pp_library_ai_weight = {
	value = num_pop_type:nobles
	add = num_pop_type:burghers
	add = num_pop_type:clergy
	multiply = 10
	add = 20
	max = 150
}

# Education pyramid (2026-11-27, Jan): Libraries -> University -> Academy of Sciences.
# Library: 1 level anywhere, +1 in a town, +2 in a city or megalopolis.
pp_library_max_level = {
	add = {
		desc = "BUILDING_LEVEL_BASE"
		value = 1
	}
	add = {
		desc = "PP_BUILDING_LEVEL_LOCATION_RANK"
		value = this.pp_library_rank_level
	}
}

pp_library_rank_level = {
	value = 0
	if = {
		limit = { location_rank ?= location_rank:town }
		value = 1
	}
	else_if = {
		limit = {
			OR = {
				location_rank ?= location_rank:city
				location_rank ?= location_rank:megalopolis
			}
		}
		value = 2
	}
}

# Location scope: the Library levels standing here. The University needs two (its allow).
pp_library_levels = {
	value = 0
	if = {
		limit = { has_building = building_type:library }
		add = "location_building_level(building_type:library)"
	}
}

# Academy of Sciences (capital only): one level per three Universities in the owner's locations, at most ten.
# The walk over the owner's locations runs only for the capital (the building's location_potential).
pp_academy_of_sciences_max_level = {
	add = {
		desc = "PP_BUILDING_LEVEL_UNIVERSITIES"
		value = this.pp_academy_university_levels
	}
}

pp_academy_university_levels = {
	value = 0
	scope:owner ?= {
		every_owned_location = {
			limit = { has_building = building_type:university }
			add = 1
		}
	}
	divide = 3
	floor = yes
	max = 10
}

# Spare room (2026-11-27, Jan: "the AI will build these in a sensible manner if that type is missing"). Since
# job_room.py an unfilled job gives little room for its workers' class, so a building that is short of workers
# can only hire from idle people of that class. Location scope, read by the schooling buildings'
# ai_construct_weight (added before their income divide): 100 while the location has an open building of that class that
# is not fully staffed and fewer idle people of the class than the threshold (one guild level's jobs). The cheap test
# comes first; the building walk runs only where idle people are few. Burghers only since 2026-11-28: the laborers'
# spare room is the RGO's spare half place, in every location, and the schools no longer give them any.
pp_spare_room_weight_burghers = {
	value = 0
	if = {
		limit = {
			"unemployed_pops_of_pop_type_in_location(pop_type:burghers)" < 0.25
			any_buildings_in_location = {
				building_pop_type = burghers
				is_opened = yes
				is_full_capacity = no
			}
		}
		value = 100
	}
}

# AI: the University's weight (location scope; divided by the owner's income there). The promotion term it had, plus a
# flat push once the owner can build the Academy of Sciences: the engine values the Academy's research highly but never
# builds a University in order to raise the Academy's cap.
pp_university_ai_weight = {
	value = pp_promotion_signal
	multiply = 150
	if = {
		limit = { scope:owner ?= { has_advance = pp_academy_of_sciences_advance } }
		add = 100
	}
}
