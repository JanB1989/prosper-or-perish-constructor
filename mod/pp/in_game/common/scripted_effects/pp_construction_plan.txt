# Prosper or Perish - Construction Plan (branch player-construction-manager, 2026-11-06, Jan).
# Player-only construction rules, run once a month by on_action/pp_construction_plan.txt for each human country that
# switched the plan on. Two layers: the global rules (Food per province, Logistics per location, Goods per good) and
# per-location own rules, which replace the global rules in that location. Location profiles (branch location-profiles,
# pp_location_profiles.txt) run first: a location with a profile follows only the profile's rules, and the global rules
# and own rules leave it alone (one layer per location). The pickers per building and good, the
# build cost data, rule edits, templates and profiles are generated from [construction_plan] into
# pp_construction_plan_generated.txt (construction_plan.py). Constructions are normal, paid constructions
# (construct_building), so they show in the construction queue like the player's own.
# Variables carry + 100 (a variable at 0 counts as unset; var:x = n compares scopes, so only < / > ranges are used).
#
# Country variables: pp_cplan_on (the plan runs), pp_cplan_food_off / _logistics_off / _goods_off (a rule group is
# off), pp_cplan_reserve / _max / _access / _keep / _payback / _gcap / _gfill / _gsur / _famine / _ypb (settings,
# defaults in [construction_plan]),
# pp_cplan_g_<good> (101 keep stocked, 102 expand within the payback line, 103 never), the pass's pp_cplan_started /
# _spent / _cap / _hit, the lists pp_cplan_log (locations built in by the last pass) and pp_cplan_own (locations with
# own rules), pp_cplan_sel (the location the Here tab shows), templates pp_cplan_t<k>_* and profiles pp_cplan_p<k>_*.
# Province: pp_cplan_focus (+ pp_cplan_lo / _hi for Custom), pp_cplan_status + list pp_cplan_status_types. The
# baseline rules keep nothing between passes (Jan 2026-11-08).
# Location: pp_cplan_mode (101 own rules, 102 hands off, unset = the global rules), the rules pp_cplan_r<n>_b
# (building type) / _lv / _m / _c / _t / _pb / _st / _est, pp_cplan_tpl (the template last applied), pp_cplan_logi (101
# always, 102 never), the list pp_cplan_cands (the Here tab's buildings), pp_cplan_log_why + list pp_cplan_log_types.

# Country scope: the monthly pass.
pp_cplan_run = {
	save_scope_as = pp_cplan_country
	save_scope_as = actor # building conditions read the builder as scope:actor
	every_in_list = {
		variable = pp_cplan_log
		clear_variable_list = pp_cplan_log_types
		remove_variable = pp_cplan_log_why
	}
	clear_variable_list = pp_cplan_log
	set_variable = { name = pp_cplan_started value = 100 }
	set_variable = { name = pp_cplan_spent value = 100 }
	remove_variable = pp_cplan_hit
	# each rule group running gets an equal share of the monthly limit; a share left unused passes on to the next
	set_variable = { name = pp_cplan_cap value = 100 }
	# Location profiles first (pp_location_profiles.txt): a location with a profile follows only its profile's rules.
	pp_lprof_run = yes
	# Location rules next: a location set to Own Rules follows only its own list.
	if = {
		limit = { pp_cplan_has_own_locations = yes }
		change_variable = { name = pp_cplan_cap add = pp_cplan_group_share }
		every_in_list = {
			variable = pp_cplan_own
			limit = {
				owner ?= scope:pp_cplan_country
				has_variable = pp_cplan_mode
				var:pp_cplan_mode < 101.5
			}
			pp_cplan_location_rules = yes
		}
	}
	if = {
		limit = { NOT = { has_variable = pp_cplan_food_off } }
		change_variable = { name = pp_cplan_cap add = pp_cplan_group_share }
		every_province = {
			pp_cplan_food_province = yes
		}
	}
	if = {
		limit = { NOT = { has_variable = pp_cplan_logistics_off } }
		change_variable = { name = pp_cplan_cap add = pp_cplan_group_share }
	}
	if = {
		limit = {
			NOT = { has_variable = pp_cplan_logistics_off }
			pp_cplan_budget_left = yes
		}
		ordered_owned_location = {
			limit = {
				pp_cplan_location_ok = yes
				pp_cplan_logistics_wanted = yes
			}
			order_by = pp_logistics_urgency_weight
			max = 200
			check_range_bounds = no
			if = {
				limit = { pp_cplan_budget_left = yes }
				pp_cplan_try_logistics_here = yes
			}
		}
	}
	# Improvements (Jan 2026-11-08): where people fill the player's share of a location's capacity, the improvement
	# that adds the most capacity per gold; the fullest locations first.
	if = {
		limit = { NOT = { has_variable = pp_cplan_improve_off } }
		change_variable = { name = pp_cplan_cap add = pp_cplan_group_share }
	}
	if = {
		limit = {
			NOT = { has_variable = pp_cplan_improve_off }
			pp_cplan_budget_left = yes
		}
		ordered_owned_location = {
			limit = {
				pp_cplan_location_ok = yes
				location_population_percentage >= scope:pp_cplan_country.pp_cplan_improve_line
			}
			order_by = pp_cplan_fill_here
			max = 200
			check_range_bounds = no
			if = {
				limit = { pp_cplan_budget_left = yes }
				pp_cplan_try_improve_here = yes
			}
		}
	}
	if = {
		limit = { NOT = { has_variable = pp_cplan_goods_off } }
		change_variable = { name = pp_cplan_cap add = pp_cplan_group_share }
	}
	if = {
		limit = {
			NOT = { has_variable = pp_cplan_goods_off }
			pp_cplan_budget_left = yes
		}
		pp_cplan_goods_run = yes
	}
	remove_variable = pp_cplan_hit
	# Never runs: tells the script check that the panel's button scopes (set by the GUI with AddScope) exist, so the
	# log has no "used but never set" lines for them.
	if = {
		limit = { always = no }
		save_scope_as = pp_key
		save_scope_as = pp_dir
		save_scope_as = pp_good
		save_scope_as = pp_mode
		save_scope_as = pp_rule
		save_scope_as = pp_slot
		save_scope_as = pp_btype
		save_scope_as = pp_field # Location Profiles
		save_scope_as = pp_col # Location Profiles
		pp_cplan_declare_flags = yes
		pp_lprof_declare_flags = yes
		if = {
			limit = {
				has_variable = pp_cplan_flag
				has_variable = pp_cplan_sel # read by the panel only
			}
			remove_variable = pp_cplan_flag
		}
	}
}

# Province scope: the food rule, one construction a month, the first of these that builds (baseline rules, Jan
# 2026-11-08). Every check reads the game as it stands; nothing is kept from pass to pass. The hunger checks read the
# store a year from now in a normal year (pp_cplan_food_in_a_year: this year's harvest and spoilage taken out).
#   1. famine (below the player's famine line): a Tavern in the province capital, where its market holds victuals;
#   2. hungry (below the low end of the focus band): a staple food maker (crop farm, orchard, fishery, herd, forest
#      village), the line that pays back fastest, in the province location where it pays back fastest;
#   3. a kitchen whenever the Cookshop budget allows (its successor first);
#   4. hungry and nothing above built: a Tavern, where its market holds victuals (provinces that cannot farm);
#   5. a Granary while the store holds less than the player's months, is nearly full and a normal month adds food;
#   6. a Grange where the store holds 24 months and a normal month adds the player's share on top;
#   7. a Victualling Yard: the next level while it pays back within the player's line, a first one where victuals
#      cost at least their default price.
# Locations with own rules, hands off or a profile are left out. While a farm or Tavern of the province has a
# level under construction, the hunger steps 2 and 4 wait (the store only moves once it stands). The state goes into
# pp_cplan_status for the panel.
pp_cplan_food_province = {
	save_scope_as = pp_cplan_province
	clear_variable_list = pp_cplan_status_types
	scope:pp_cplan_country = { remove_variable = pp_cplan_hit }
	if = {
		limit = { pp_cplan_hands_off_n > 0.5 }
		set_variable = { name = pp_cplan_status value = 105 }
	}
	else_if = {
		limit = { NOT = { has_variable = pp_stored_food_months } }
		set_variable = { name = pp_cplan_status value = 108 }
	}
	else_if = {
		limit = { pp_cplan_budget_left = no }
		set_variable = { name = pp_cplan_status value = 107 }
	}
	else = {
		# 1. famine: a Tavern
		if = {
			limit = { pp_cplan_famine = yes }
			province_capital ?= {
				if = {
					limit = {
						pp_cplan_location_ok = yes
						pp_cplan_victuals_here = yes
					}
					pp_cplan_try_tavern_here = yes
				}
			}
		}
		# 2. hungry: a farm
		if = {
			limit = {
				pp_cplan_may_try = yes
				pp_cplan_hungry = yes
				NOT = { any_location_in_province = { pp_cplan_food_queued = yes } }
			}
			ordered_location_in_province = {
				limit = {
					pp_cplan_location_ok = yes
					pp_cplan_farm_best_roi > 0
				}
				order_by = pp_cplan_farm_best_roi
				max = pp_cplan_farm_sites
				check_range_bounds = no
				if = {
					limit = { pp_cplan_may_try = yes }
					pp_cplan_try_farm_here = yes
				}
			}
		}
		# 3. a kitchen, whenever it can be built
		if = {
			limit = { pp_cplan_may_try = yes }
			province_capital ?= {
				if = {
					limit = { pp_cplan_location_ok = yes }
					pp_cplan_try_kitchen_here = yes
				}
			}
		}
		# 4. hungry and nothing else possible: a Tavern
		if = {
			limit = {
				pp_cplan_may_try = yes
				pp_cplan_hungry = yes
				NOT = { any_location_in_province = { pp_cplan_food_queued = yes } }
			}
			province_capital ?= {
				if = {
					limit = {
						pp_cplan_location_ok = yes
						pp_cplan_victuals_here = yes
					}
					pp_cplan_try_tavern_here = yes
				}
			}
		}
		# 5. a Granary
		if = {
			limit = {
				pp_cplan_may_try = yes
				pp_cplan_granary_due = yes
			}
			province_capital ?= {
				if = {
					limit = { pp_cplan_location_ok = yes }
					pp_cplan_try_granary_here = yes
				}
			}
		}
		# 6. a Grange
		if = {
			limit = {
				pp_cplan_may_try = yes
				pp_cplan_grange_due = yes
			}
			province_capital ?= {
				if = {
					limit = { pp_cplan_location_ok = yes }
					pp_cplan_try_grange_here = yes
				}
			}
		}
		# 7. a Victualling Yard
		if = {
			limit = { pp_cplan_may_try = yes }
			every_location_in_province = {
				limit = {
					pp_cplan_location_ok = yes
					OR = {
						pp_victualling_yard_site = yes
						is_market_center = yes
					}
				}
				if = {
					limit = { pp_cplan_may_try = yes }
					pp_cplan_try_yard_here = yes
				}
			}
		}
		# the state for the panel
		if = {
			limit = { scope:pp_cplan_country = { has_variable = pp_cplan_hit } }
			set_variable = { name = pp_cplan_status value = 101 }
		}
		else_if = {
			limit = { pp_cplan_budget_left = no }
			set_variable = { name = pp_cplan_status value = 107 }
		}
		else_if = {
			limit = {
				pp_cplan_hungry = yes
				any_location_in_province = { pp_cplan_food_queued = yes }
			}
			set_variable = { name = pp_cplan_status value = 110 }
		}
		else_if = {
			limit = { pp_cplan_hungry = yes }
			set_variable = { name = pp_cplan_status value = 102 }
		}
		else_if = {
			limit = { pp_cplan_granary_due = yes }
			set_variable = { name = pp_cplan_status value = 109 }
		}
		else = {
			set_variable = { name = pp_cplan_status value = 103 }
		}
	}
	scope:pp_cplan_country = { remove_variable = pp_cplan_hit }
}

# Location scope: start one level of $type$ here and record it ($why$: 101 food, 102 logistics, 103 goods, 104 a
# location rule).
pp_cplan_construct = {
	building_type:$type$ = { save_scope_as = pp_cplan_btype }
	pp_cplan_construct_scoped = { why = $why$ }
}

# Location scope: the same for the building type scope:pp_cplan_btype (never a second level of it here while one is
# under construction: the rules see a level only once it stands).
pp_cplan_construct_scoped = {
	if = {
		limit = { pp_cplan_bt_queued < 0.5 }
		pp_cplan_construct_now = { why = $why$ }
	}
}
pp_cplan_construct_now = {
	save_scope_as = pp_cplan_site
	scope:pp_cplan_country = {
		set_variable = { name = pp_cplan_gold_before value = { value = gold add = 1000000 } }
	}
	construct_building = { building_type = scope:pp_cplan_btype }
	scope:pp_cplan_country = {
		change_variable = { name = pp_cplan_spent add = { value = var:pp_cplan_gold_before subtract = 1000000 subtract = gold } }
		change_variable = { name = pp_cplan_started add = 1 }
		set_variable = { name = pp_cplan_hit value = yes }
		remove_variable = pp_cplan_gold_before
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = pp_cplan_log target = scope:pp_cplan_site } } }
			add_to_variable_list = { name = pp_cplan_log target = scope:pp_cplan_site }
		}
	}
	add_to_variable_list = { name = pp_cplan_log_types target = scope:pp_cplan_btype }
	set_variable = { name = pp_cplan_log_why value = $why$ }
}

# Location scope: the same for the food rule, which also tells the province what it started.
pp_cplan_construct_food = {
	pp_cplan_construct = { type = $type$ why = 101 }
	scope:pp_cplan_province = {
		add_to_variable_list = { name = pp_cplan_status_types target = building_type:$type$ }
	}
}

# ---- Location rules ----

# Location scope (an own-rules location of scope:pp_cplan_country): its rules in order.
pp_cplan_location_rules = {
	save_scope_as = pp_cplan_loc
	pp_cplan_rule = { n = 1 }
	pp_cplan_rule = { n = 2 }
	pp_cplan_rule = { n = 3 }
	pp_cplan_rule = { n = 4 }
}

# Location scope: rule $n$. Up to its level, one level at a time, while its condition holds and the next level pays
# back within its limit (a building that does not stand yet has no profit to measure: its first level needs only the
# condition). The state goes into pp_cplan_r$n$_st (+ 100: 1 started, 2 waiting for the condition, 3 done, 4 a level
# under construction, 5 payback too long, 6 cannot be built now, 7 held back by the budget, 8 the market lacks its
# construction goods), the levels standing into pp_cplan_r$n$_has and, for a building that stands, the payback of its
# next level into pp_cplan_r$n$_est (years + 100; the panel shows this number).
pp_cplan_rule = {
	if = {
		limit = { has_variable = pp_cplan_r$n$_b }
		var:pp_cplan_r$n$_b = { save_scope_as = pp_cplan_btype }
		pp_cplan_btype_data = yes
		remove_variable = pp_cplan_r$n$_est
		set_variable = { name = pp_cplan_r$n$_has value = { value = pp_cplan_bt_levels add = 100 } }
		if = {
			limit = { pp_cplan_bt_levels > 0.5 }
			set_variable = { name = pp_cplan_r$n$_est value = { value = pp_cplan_est_payback_years add = 100 } }
		}
		if = {
			limit = { pp_cplan_bt_levels >= pp_cplan_r$n$_lv_n }
			set_variable = { name = pp_cplan_r$n$_st value = 103 }
		}
		else_if = {
			limit = { pp_cplan_bt_queued > 0 }
			set_variable = { name = pp_cplan_r$n$_st value = 104 }
		}
		else_if = {
			limit = { NOT = { pp_cplan_rule_cond = { n = $n$ } } }
			set_variable = { name = pp_cplan_r$n$_st value = 102 }
		}
		else_if = {
			# NOT = { a b } means neither a nor b in this script language: cannot be built = either check fails
			limit = {
				OR = {
					NOT = { can_build_building = scope:pp_cplan_btype }
					NOT = { owner = { can_build_building = scope:pp_cplan_btype } }
				}
			}
			set_variable = { name = pp_cplan_r$n$_st value = 106 }
		}
		else = {
			if = {
				limit = { NOT = { pp_cplan_rule_payback_ok = { n = $n$ } } }
				set_variable = { name = pp_cplan_r$n$_st value = 105 }
			}
			else_if = {
				limit = { scope:pp_cplan_gok < 0.5 }
				set_variable = { name = pp_cplan_r$n$_st value = 108 }
			}
			else_if = {
				limit = { pp_cplan_budget_left = no }
				set_variable = { name = pp_cplan_r$n$_st value = 107 }
			}
			else = {
				pp_cplan_construct_scoped = { why = 104 }
				set_variable = { name = pp_cplan_r$n$_st value = 101 }
			}
		}
	}
}

# Location scope: one change to rule $n$ (scope:pp_key = lv / metric / cmp / t / tbig / pb / clear, scope:pp_dir =
# up / down). An edited location no longer counts as following its template.
pp_cplan_rule_edit = {
	switch = {
		trigger = scope:pp_key
		flag:lv = {
			pp_cplan_step_var = { var = pp_cplan_r$n$_lv current = pp_cplan_r$n$_lv_n step = 1 min = 101 max = pp_cplan_max_rule_level100 }
		}
		flag:metric = {
			if = {
				limit = { NOT = { has_variable = pp_cplan_r$n$_m } }
				if = {
					limit = { scope:pp_dir = flag:up }
					set_variable = { name = pp_cplan_r$n$_m value = 101 }
				}
				else = {
					set_variable = { name = pp_cplan_r$n$_m value = pp_cplan_metric_last100 }
				}
			}
			else = {
				if = {
					limit = { scope:pp_dir = flag:up }
					change_variable = { name = pp_cplan_r$n$_m add = 1 }
				}
				else = {
					change_variable = { name = pp_cplan_r$n$_m subtract = 1 }
				}
				if = {
					limit = {
						OR = {
							var:pp_cplan_r$n$_m < 100.5
							var:pp_cplan_r$n$_m > pp_cplan_metric_last100
						}
					}
					remove_variable = pp_cplan_r$n$_m
				}
			}
			# a new condition starts at its value here now
			if = {
				limit = { has_variable = pp_cplan_r$n$_m }
				set_variable = { name = pp_cplan_r$n$_t value = { value = pp_cplan_r$n$_now add = 100000 } }
			}
			else = {
				remove_variable = pp_cplan_r$n$_t
				remove_variable = pp_cplan_r$n$_c
			}
		}
		flag:cmp = {
			if = {
				limit = { has_variable = pp_cplan_r$n$_c }
				remove_variable = pp_cplan_r$n$_c
			}
			else = {
				set_variable = { name = pp_cplan_r$n$_c value = 101 }
			}
		}
		flag:t = {
			if = {
				limit = { NOT = { has_variable = pp_cplan_r$n$_t } }
				set_variable = { name = pp_cplan_r$n$_t value = 100000 }
			}
			if = {
				limit = { scope:pp_dir = flag:up }
				change_variable = { name = pp_cplan_r$n$_t add = pp_cplan_r$n$_step }
			}
			else = {
				change_variable = { name = pp_cplan_r$n$_t subtract = pp_cplan_r$n$_step }
			}
		}
		flag:tbig = {
			if = {
				limit = { NOT = { has_variable = pp_cplan_r$n$_t } }
				set_variable = { name = pp_cplan_r$n$_t value = 100000 }
			}
			if = {
				limit = { scope:pp_dir = flag:up }
				change_variable = { name = pp_cplan_r$n$_t add = { value = pp_cplan_r$n$_step multiply = 10 } }
			}
			else = {
				change_variable = { name = pp_cplan_r$n$_t subtract = { value = pp_cplan_r$n$_step multiply = 10 } }
			}
		}
		flag:pb = {
			pp_cplan_step_var = { var = pp_cplan_r$n$_pb current = pp_cplan_r$n$_pb_n step = 1 min = 100 max = 150 }
			if = {
				limit = { var:pp_cplan_r$n$_pb < 100.5 }
				remove_variable = pp_cplan_r$n$_pb
			}
		}
		flag:clear = {
			remove_variable = pp_cplan_r$n$_b
			remove_variable = pp_cplan_r$n$_lv
			remove_variable = pp_cplan_r$n$_m
			remove_variable = pp_cplan_r$n$_c
			remove_variable = pp_cplan_r$n$_t
			remove_variable = pp_cplan_r$n$_pb
			remove_variable = pp_cplan_r$n$_st
			remove_variable = pp_cplan_r$n$_est
			remove_variable = pp_cplan_r$n$_has
		}
	}
	remove_variable = pp_cplan_tpl
}

# Location scope: puts the building type scope:pp_btype into the first free rule (up to one level more than stands
# here), unless a rule already names it; the location switches to its own rules.
pp_cplan_add_rule = {
	scope:pp_btype = { save_scope_as = pp_cplan_btype }
	if = {
		limit = {
			NOT = {
				OR = {
					AND = { has_variable = pp_cplan_r1_b var:pp_cplan_r1_b = scope:pp_btype }
					AND = { has_variable = pp_cplan_r2_b var:pp_cplan_r2_b = scope:pp_btype }
					AND = { has_variable = pp_cplan_r3_b var:pp_cplan_r3_b = scope:pp_btype }
					AND = { has_variable = pp_cplan_r4_b var:pp_cplan_r4_b = scope:pp_btype }
				}
			}
		}
		if = {
			limit = { NOT = { has_variable = pp_cplan_r1_b } }
			pp_cplan_fill_rule = { n = 1 }
		}
		else_if = {
			limit = { NOT = { has_variable = pp_cplan_r2_b } }
			pp_cplan_fill_rule = { n = 2 }
		}
		else_if = {
			limit = { NOT = { has_variable = pp_cplan_r3_b } }
			pp_cplan_fill_rule = { n = 3 }
		}
		else_if = {
			limit = { NOT = { has_variable = pp_cplan_r4_b } }
			pp_cplan_fill_rule = { n = 4 }
		}
	}
	if = {
		limit = { NOT = { has_variable = pp_cplan_mode } }
		set_variable = { name = pp_cplan_mode value = 101 }
	}
	else_if = {
		limit = { var:pp_cplan_mode > 101.5 }
		set_variable = { name = pp_cplan_mode value = 101 }
	}
	remove_variable = pp_cplan_tpl
	pp_cplan_own_sync = yes
}

# Location scope: rule $n$ builds scope:pp_cplan_btype up to one level more than stands here, always, any payback.
pp_cplan_fill_rule = {
	set_variable = { name = pp_cplan_r$n$_b value = scope:pp_cplan_btype }
	set_variable = { name = pp_cplan_r$n$_has value = { value = pp_cplan_bt_levels add = 100 } }
	set_variable = { name = pp_cplan_r$n$_lv value = { value = pp_cplan_bt_levels add = 101 max = pp_cplan_max_rule_level100 } }
	remove_variable = pp_cplan_r$n$_m
	remove_variable = pp_cplan_r$n$_c
	remove_variable = pp_cplan_r$n$_t
	remove_variable = pp_cplan_r$n$_pb
	remove_variable = pp_cplan_r$n$_st
	remove_variable = pp_cplan_r$n$_est
	# a building that stands gets its payback at once (the panel shows the plan's own number for it)
	if = {
		limit = { pp_cplan_bt_levels > 0.5 }
		save_scope_as = pp_cplan_loc
		owner ?= { save_scope_as = pp_cplan_country }
		pp_cplan_btype_data = yes
		set_variable = { name = pp_cplan_r$n$_est value = { value = pp_cplan_est_payback_years add = 100 } }
	}
}

# Location scope: the next mode, the plan's rules -> own rules -> hands off -> the plan's rules.
pp_cplan_cycle_mode = {
	if = {
		limit = { NOT = { has_variable = pp_cplan_mode } }
		set_variable = { name = pp_cplan_mode value = 101 }
	}
	else_if = {
		limit = { var:pp_cplan_mode < 101.5 }
		set_variable = { name = pp_cplan_mode value = 102 }
	}
	else = {
		remove_variable = pp_cplan_mode
	}
	remove_variable = pp_cplan_tpl
	pp_cplan_own_sync = yes
}

# Location scope: keeps the owner's list of own-rules locations in step with this location's mode.
pp_cplan_own_sync = {
	save_temporary_scope_as = pp_cplan_this
	if = {
		limit = {
			has_variable = pp_cplan_mode
			var:pp_cplan_mode < 101.5
		}
		owner ?= {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = pp_cplan_own target = scope:pp_cplan_this } } }
				add_to_variable_list = { name = pp_cplan_own target = scope:pp_cplan_this }
			}
		}
	}
	else = {
		owner ?= {
			if = {
				limit = { is_target_in_variable_list = { name = pp_cplan_own target = scope:pp_cplan_this } }
				remove_list_variable = { name = pp_cplan_own target = scope:pp_cplan_this }
			}
		}
	}
	# one layer per location: own rules or hands off take the place of a location profile (pp_location_profiles.txt)
	if = {
		limit = {
			has_variable = pp_cplan_mode
			has_variable = pp_lprof
		}
		pp_lprof_unassign = yes
	}
}

# Location scope: the buildings the Here tab offers: the ones standing here (list pp_cplan_cands_built, the buildings
# themselves, so the panel reads their profit per level), then every other type the owner can build that this location
# allows (list pp_cplan_cands, building types).
pp_cplan_fill_cands = {
	save_scope_as = pp_cplan_loc
	owner ?= { save_scope_as = actor } # building conditions read the builder as scope:actor
	clear_variable_list = pp_cplan_cands
	clear_variable_list = pp_cplan_cands_built
	every_buildings_in_location = {
		limit = { building_level > 0 }
		save_temporary_scope_as = pp_cplan_bld
		scope:pp_cplan_loc = {
			add_to_variable_list = { name = pp_cplan_cands_built target = scope:pp_cplan_bld }
		}
	}
	owner ?= {
		every_buildable_building_type = {
			save_temporary_scope_as = pp_cplan_bt
			if = {
				limit = {
					scope:pp_cplan_loc = {
						can_build_building = scope:pp_cplan_bt
						NOT = { has_building = scope:pp_cplan_bt }
					}
				}
				scope:pp_cplan_loc = {
					add_to_variable_list = { name = pp_cplan_cands target = scope:pp_cplan_bt }
				}
			}
		}
	}
}

# Location scope (scope:pp_cplan_loc = this location, scope:pp_cplan_country = its owner): this location's variable
# $from$ -> the country's $to$ (a number, or a scope with _scope).
pp_cplan_put_num = {
	if = {
		limit = { has_variable = $from$ }
		scope:pp_cplan_country = { set_variable = { name = $to$ value = scope:pp_cplan_loc.var:$from$ } }
	}
	else = {
		scope:pp_cplan_country = { remove_variable = $to$ }
	}
}
pp_cplan_put_scope = {
	if = {
		limit = { has_variable = $from$ }
		var:$from$ = { save_temporary_scope_as = pp_cplan_tmp }
		scope:pp_cplan_country = { set_variable = { name = $to$ value = scope:pp_cplan_tmp } }
	}
	else = {
		scope:pp_cplan_country = { remove_variable = $to$ }
	}
}

# Location scope: the country's variable $from$ -> this location's $to$.
pp_cplan_take_num = {
	if = {

		set_variable = { name = $to$ value = scope:pp_cplan_country.var:$from$ }

	else = {
		remove_variable = $to$
	}
}
pp_cplan_take_scope = {

		scope:pp_cplan_country = { var:$from$ = { save_temporary_scope_as = pp_cplan_tmp } }
		set_variable = { name = $to$ value = scope:pp_cplan_tmp }
	}
	else = {
		remove_variable = $to$
	}
}

# ---- Settings (scripted GUIs in scripted_guis/pp_construction_plan.txt) ----

# Any scope: flips the presence of the variable $var$.
pp_cplan_flip = {
	if = {
		limit = { has_variable = $var$ }
		remove_variable = $var$
	}
	else = {
		set_variable = { name = $var$ value = yes }
	}
}

# Any scope: moves the setting $var$ (+ 100) one $step$ in the direction of the flag scope:pp_dir (up / down), kept
# between $min$ and $max$ (both + 100). $current$ is the setting's value with its default.
pp_cplan_step_var = {
	set_variable = { name = $var$ value = { value = $current$ add = 100 } }
	if = {
		limit = { scope:pp_dir = flag:up }
		change_variable = { name = $var$ add = $step$ }
	}
	else = {
		change_variable = { name = $var$ subtract = $step$ }
	}
	if = {
		limit = { var:$var$ < $min$ }
		set_variable = { name = $var$ value = $min$ }
	}
	if = {
		limit = { var:$var$ > $max$ }
		set_variable = { name = $var$ value = $max$ }
	}
}

# Country scope: the goods mode of one good, from the flag scope:pp_mode (generated switch pp_cplan_set_good_mode).
pp_cplan_apply_mode = {
	switch = {
		trigger = scope:pp_mode
		flag:off = { remove_variable = $var$ }
		flag:keep = { set_variable = { name = $var$ value = 101 } }
		flag:expand = { set_variable = { name = $var$ value = 102 } }
		flag:never = { set_variable = { name = $var$ value = 103 } }
	}
}
