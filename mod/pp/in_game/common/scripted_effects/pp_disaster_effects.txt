# PP (2026-11-27, Jan): long disasters, growing pain (AGENTS.md "Long Disasters: Growing Pain"). Jan: "If it's not
# resolved, there's a period of growing bigger pain, but at the end, you can resolve it with lower stability if you
# survived until then"; "I kind of don't even want something like a hard end". The copies of the general disasters
# (`ppc disasters apply`) call these with their own type; no pulse of its own: the steps run from the disaster's
# on_monthly, the cleanup from its on_end. The step modifiers pp_disaster_pain_<type>_1..5 are generated
# (pp_disaster_modifiers.txt); none of them touches what the disaster's end asks for.

# Country scope, the disaster's on_monthly: one step modifier per five years from 25 (the last stays for as long as the
# disaster runs); notice pp_disaster.1 at the first step, pp_disaster.2 at each later one. A save past a step starts at
# the step it is in.
pp_disaster_pain_effect = {
	hidden_effect = {
		if = {
			limit = { had_disaster_for_years = { disaster_type = disaster_type:$type$ years >= 45 } }
			if = {
				limit = { NOT = { has_country_modifier = pp_disaster_pain_$type$_5 } }
				pp_disaster_pain_clear_modifiers_effect = { type = $type$ }
				add_country_modifier = { modifier = pp_disaster_pain_$type$_5 years = -1 }
				trigger_event_silently = pp_disaster.2
			}
		}
		else_if = {
			limit = { had_disaster_for_years = { disaster_type = disaster_type:$type$ years >= 40 } }
			if = {
				limit = { NOT = { has_country_modifier = pp_disaster_pain_$type$_4 } }
				pp_disaster_pain_clear_modifiers_effect = { type = $type$ }
				add_country_modifier = { modifier = pp_disaster_pain_$type$_4 years = -1 }
				trigger_event_silently = pp_disaster.2
			}
		}
		else_if = {
			limit = { had_disaster_for_years = { disaster_type = disaster_type:$type$ years >= 35 } }
			if = {
				limit = { NOT = { has_country_modifier = pp_disaster_pain_$type$_3 } }
				pp_disaster_pain_clear_modifiers_effect = { type = $type$ }
				add_country_modifier = { modifier = pp_disaster_pain_$type$_3 years = -1 }
				trigger_event_silently = pp_disaster.2
			}
		}
		else_if = {
			limit = { had_disaster_for_years = { disaster_type = disaster_type:$type$ years >= 30 } }
			if = {
				limit = { NOT = { has_country_modifier = pp_disaster_pain_$type$_2 } }
				pp_disaster_pain_clear_modifiers_effect = { type = $type$ }
				add_country_modifier = { modifier = pp_disaster_pain_$type$_2 years = -1 }
				trigger_event_silently = pp_disaster.2
			}
		}
		else_if = {
			limit = { had_disaster_for_years = { disaster_type = disaster_type:$type$ years >= 25 } }
			if = {
				limit = { NOT = { has_country_modifier = pp_disaster_pain_$type$_1 } }
				pp_disaster_pain_clear_modifiers_effect = { type = $type$ }
				add_country_modifier = { modifier = pp_disaster_pain_$type$_1 years = -1 }
				trigger_event_silently = pp_disaster.1
			}
		}
	}
}

# Country scope, the disaster's on_monthly (only disasters with a relaxed end): once the relaxed end is open, tell the
# country (pp_disaster.3, once per disaster).
pp_disaster_relief_notice_effect = {
	hidden_effect = {
		if = {
			limit = {
				pp_disaster_relief_due_trigger = { type = $type$ }
				NOT = { has_variable = pp_disaster_relief_$type$ }
			}
			set_variable = pp_disaster_relief_$type$
			trigger_event_silently = pp_disaster.3
		}
	}
}

pp_disaster_pain_clear_modifiers_effect = {
	if = {
		limit = { has_country_modifier = pp_disaster_pain_$type$_1 }
		remove_country_modifier = pp_disaster_pain_$type$_1
	}
	if = {
		limit = { has_country_modifier = pp_disaster_pain_$type$_2 }
		remove_country_modifier = pp_disaster_pain_$type$_2
	}
	if = {
		limit = { has_country_modifier = pp_disaster_pain_$type$_3 }
		remove_country_modifier = pp_disaster_pain_$type$_3
	}
	if = {
		limit = { has_country_modifier = pp_disaster_pain_$type$_4 }
		remove_country_modifier = pp_disaster_pain_$type$_4
	}
	if = {
		limit = { has_country_modifier = pp_disaster_pain_$type$_5 }
		remove_country_modifier = pp_disaster_pain_$type$_5
	}
}

# Country scope, the disaster's on_end (every ending, after vanilla's on_end): the pain is over.
pp_disaster_pain_clear_effect = {
	pp_disaster_pain_clear_modifiers_effect = { type = $type$ }
	if = {
		limit = { has_variable = pp_disaster_relief_$type$ }
		remove_variable = pp_disaster_relief_$type$
	}
}

# Country scope, the disaster's on_end when the relaxed branch ended it: the same disaster may not start again for 25
# years (vanilla's block variable, read by pp_disaster_restart_allowed_trigger in the copy's can_start;
# religious_turmoil's spawn chance is very high and its start asks for stability below 20).
pp_disaster_restart_block_effect = {
	block_disaster_temporarily_effect = { type = $type$ years = 25 years_in_days = 9125 }
}
