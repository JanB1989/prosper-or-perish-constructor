# Prosper or Perish - Location plan (2026-11-07, Jan; plans as town rights since 2026-11-19, branch town-plans): the
# player decides what a location is for (gui/pp_location_plan.gui, the location window's Plans row, map mode
# pp_location_plan). A plan is a town right without conditions and without a bonus (pp_plan_<plan>,
# common/town_rights), set and removed only by these buttons; the Construction Plan builds toward the held plans. The
# toggle switch, Clear and the declarations are generated into pp_location_plan_generated.txt (location_plan.py).
# Location variables: pp_lplan_st_g_<plan> (the plan's state after the last pass + 100) and pp_lplan_est_g_<plan> (the
# shortest payback found, years + 100). Country: pp_lplan_sel (the location the window shows).

# Location scope: sets the plan $g$, or takes it off when the location holds it. Strictly one if / else on the held
# right: a script grant has no checks (the same right could be held twice) and a revoke takes off one copy. A plan
# taken off takes its state along. The game sends its own town rights message for both.
pp_tplan_flip = {
	if = {
		limit = { has_town_rights = town_rights_type:pp_plan_$g$ }
		revoke_town_rights_of_type = town_rights_type:pp_plan_$g$
		remove_variable = pp_lplan_st_g_$g$
		remove_variable = pp_lplan_est_g_$g$
		# what the rule closed here for its workers opens again once no plan it was for is left (pp_plan_closer.txt)
		pp_pclose_plan_off = yes
	}
	else = {
		grant_town_rights = town_rights_type:pp_plan_$g$
	}
	refresh_map_colors = yes
}

# Location scope (root, from the scripted GUI): every plan taken off.
pp_tplan_clear_do = {
	pp_tplan_clear_all = yes
	refresh_map_colors = yes
}

# Location scope (root, from the scripted GUI): the window shows this location.
pp_lplan_select = {
	owner ?= { set_variable = { name = pp_lplan_sel value = root } }
	# Never runs: declares the window's flag scope and the flags it passes, and uses the variable only the window reads.
	if = {
		limit = { always = no }
		save_scope_as = pp_key
		pp_tplan_declare_flags = yes
		if = {
			limit = { has_variable = pp_lplan_sel }
			remove_variable = pp_lplan_sel
		}
	}
}
