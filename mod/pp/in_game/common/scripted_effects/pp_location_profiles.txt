# Prosper or Perish - Location Profiles (branch location-profiles, 2026-11-07, Jan; docs/location_profiles.md).
# The player defines profiles (an icon, a colour, ordered build rules) and assigns them to locations on the map. In a
# location with a profile the global Construction Plan rules stand back (pp_cplan_location_ok) and the profile's rules
# build there: the first rule group of the plan's monthly pass (pp_cplan_run), with its own share of the monthly limit.
# Player-only like the plan: the pass runs for human countries that switched the plan on, the edits are scripted GUIs.
# The per-target buildings, the dispatch, the per-slot edits and the flag declarations are generated into
# pp_location_profiles_generated.txt (location_profiles.py). Variables carry + 100 (a variable at 0 counts as unset;
# var:x = n compares scopes, so numbers use < / > ranges).
#
# Country: pp_lprof_<k>_on (slot k in use), pp_lprof_<k>_ig / _ic (icon: a goods scope or a category flag),
# pp_lprof_<k>_col (colour), pp_lprof_<k>_r<r>_g / _c (rule target: goods scope or category flag), pp_lprof_<k>_r<r>_pb
# (payback limit in years, unset = any), the list pp_lprof_locs (locations with a profile).
# Location: pp_lprof (slot k), pp_lprof_r<r>_st (rule state after the last pass: 1 started, 2 nothing to build, 3 a
# level under construction, 5 payback too long, 6 cannot be built here, 7 budget, 8 the market lacks the construction
# goods, 9 waiting for an earlier rule's construction this month). Pass temporaries: pp_lprof_st, pp_lprof_fail,
# pp_lprof_best, pp_lprof_pick (location), the list pp_lprof_gone (country).

# ---- The monthly pass (country scope, inside pp_cplan_run: scope:pp_cplan_country and scope:actor saved) ----

pp_lprof_run = {
	pp_lprof_cleanup = yes
	if = {
		limit = { pp_lprof_has_locations = yes }
		change_variable = { name = pp_cplan_cap add = pp_cplan_group_share }
		every_in_list = {
			variable = pp_lprof_locs
			limit = {
				owner ?= scope:pp_cplan_country
				has_variable = pp_lprof
			}
			pp_lprof_location_rules = yes
		}
	}
	remove_variable = pp_cplan_hit
}

# Country scope: the list loses entries whose profile is gone (cleared by another owner), and an own location keeps no
# profile the country did not give it (an earlier owner's). A location lost and won back keeps its own profile.
pp_lprof_cleanup = {
	every_in_list = {
		variable = pp_lprof_locs
		limit = { NOT = { has_variable = pp_lprof } }
		pp_lprof_mark_gone = yes
	}
	pp_lprof_drop_gone = yes
	every_owned_location = {
		limit = { has_variable = pp_lprof }
		save_temporary_scope_as = pp_lprof_this
		if = {
			limit = {
				scope:pp_cplan_country = {
					NOT = { is_target_in_variable_list = { name = pp_lprof_locs target = scope:pp_lprof_this } }
				}
			}
			remove_variable = pp_lprof
			pp_lprof_clear_states = yes
			refresh_map_colors = yes
		}
	}
}

# Location scope: goes on root's (the country's) list pp_lprof_gone.
pp_lprof_mark_gone = {
	save_temporary_scope_as = pp_lprof_this
	root = { add_to_variable_list = { name = pp_lprof_gone target = scope:pp_lprof_this } }
}

# Country scope (root): every location on pp_lprof_gone leaves pp_lprof_locs; one the country owns loses its profile.
pp_lprof_drop_gone = {
	every_in_list = {
		variable = pp_lprof_gone
		save_temporary_scope_as = pp_lprof_this
		if = {
			limit = { owner ?= root }
			remove_variable = pp_lprof
			pp_lprof_clear_states = yes
		}
		root = { remove_list_variable = { name = pp_lprof_locs target = scope:pp_lprof_this } }
	}
	clear_variable_list = pp_lprof_gone
}

# Location scope: rule $r$ of profile $k$ (pp_lprof_location_rules runs them in order). A rule without a target has no
# state. After a construction started here this month by an earlier rule: 9. Else its target decides (generated
# pp_lprof_t_<target>), within its payback limit (scope:pp_lprof_pb, years + 100; no limit = any).
pp_lprof_rule = {
	remove_variable = pp_lprof_st
	remove_variable = pp_lprof_fail
	remove_variable = pp_lprof_best
	if = {
		limit = {
			scope:pp_cplan_country = {
				OR = {
					has_variable = pp_lprof_$k$_r$r$_g
					has_variable = pp_lprof_$k$_r$r$_c
				}
			}
		}
		if = {
			limit = { scope:pp_cplan_country = { has_variable = pp_cplan_hit } }
			set_variable = { name = pp_lprof_st value = 109 }
		}
		else = {
			save_temporary_scope_value_as = { name = pp_lprof_pb value = 1000000 }
			if = {
				limit = { scope:pp_cplan_country = { has_variable = pp_lprof_$k$_r$r$_pb } }
				save_temporary_scope_value_as = { name = pp_lprof_pb value = scope:pp_cplan_country.var:pp_lprof_$k$_r$r$_pb }
			}
			if = {
				limit = { scope:pp_cplan_country = { has_variable = pp_lprof_$k$_r$r$_g } }
				pp_lprof_goods_target = { v = pp_lprof_$k$_r$r$_g }
			}
			else = {
				pp_lprof_category_target = { v = pp_lprof_$k$_r$r$_c }
			}
		}
		pp_lprof_finish = { r = $r$ }
	}
	else = {
		remove_variable = pp_lprof_r$r$_st
		remove_variable = pp_lprof_r$r$_est
	}
}

# Location scope: writes rule $r$'s state: what the target set (1 started, 3 a level under construction, 9 waiting),
# else the best reason among its candidates (pp_lprof_fail: 105 budget -> 7, 104 goods -> 8, 103 payback -> 5,
# 102 cannot be built -> 6, 101 at its highest level -> 2), else no candidate at all (6).
pp_lprof_finish = {
	if = {
		limit = { has_variable = pp_lprof_st }
		set_variable = { name = pp_lprof_r$r$_st value = var:pp_lprof_st }
	}
	else_if = {
		limit = { NOT = { has_variable = pp_lprof_fail } }
		set_variable = { name = pp_lprof_r$r$_st value = 106 }
	}
	else_if = {
		limit = { var:pp_lprof_fail > 104.5 }
		set_variable = { name = pp_lprof_r$r$_st value = 107 }
	}
	else_if = {
		limit = { var:pp_lprof_fail > 103.5 }
		set_variable = { name = pp_lprof_r$r$_st value = 108 }
	}
	else_if = {
		limit = { var:pp_lprof_fail > 102.5 }
		set_variable = { name = pp_lprof_r$r$_st value = 105 }
	}
	else_if = {
		limit = { var:pp_lprof_fail > 101.5 }
		set_variable = { name = pp_lprof_r$r$_st value = 106 }
	}
	else = {
		set_variable = { name = pp_lprof_r$r$_st value = 102 }
	}
	# the shortest payback of a standing candidate's next level (years + 100; 1099 = no profit), shown beside the state
	if = {
		limit = { has_variable = pp_lprof_best }
		set_variable = { name = pp_lprof_r$r$_est value = var:pp_lprof_best }
	}
	else = {
		remove_variable = pp_lprof_r$r$_est
	}
	remove_variable = pp_lprof_st
	remove_variable = pp_lprof_fail
	remove_variable = pp_lprof_best
}

# Location scope (scope:pp_cplan_btype and the cost values set): keeps the shortest payback seen for this rule.
pp_lprof_note_payback = {
	if = {
		limit = {
			OR = {
				NOT = { has_variable = pp_lprof_best }
				var:pp_lprof_best > pp_lprof_payback100
			}
		}
		set_variable = { name = pp_lprof_best value = pp_lprof_payback100 }
	}
}

# Location scope: records why a candidate failed ($r$ = 101..105), keeping the reason closest to a construction.
pp_lprof_fail = {
	if = {
		limit = { NOT = { has_variable = pp_lprof_fail } }
		set_variable = { name = pp_lprof_fail value = $r$ }
	}
	else_if = {
		limit = { var:pp_lprof_fail < $r$ }
		set_variable = { name = pp_lprof_fail value = $r$ }
	}
}

# Location scope: one more level of the standing $type$ (cost data: increase per level $ipl$, construction demand
# $demand$, efficiency category $cat$) when it is below its highest level, can be built here, its next level pays back
# within scope:pp_lprof_pb and the market holds its construction goods. Nothing once a construction started.
pp_lprof_expand = {
	if = {
		limit = {
			scope:pp_cplan_country = { NOT = { has_variable = pp_cplan_hit } }
			any_buildings_in_location = { building_type = building_type:$type$ building_level > 0 }
		}
		if = {
			limit = { any_buildings_in_location = { building_type = building_type:$type$ is_at_max_level = yes } }
			pp_lprof_fail = { r = 101 }
		}
		else_if = {
			limit = { NOT = { location_and_owner_can_build = { building_type = $type$ } } }
			pp_lprof_fail = { r = 102 }
		}
		else = {
			building_type:$type$ = { save_scope_as = pp_cplan_btype }
			save_temporary_scope_value_as = { name = pp_cplan_ipl value = $ipl$ }
			save_temporary_scope_value_as = { name = pp_cplan_pf value = pp_cplan_cd_$demand$ }
			save_temporary_scope_value_as = { name = pp_cplan_ef value = pp_cplan_eff_$cat$ }
			pp_lprof_note_payback = yes
			if = {
				limit = { pp_lprof_payback100 > scope:pp_lprof_pb }
				pp_lprof_fail = { r = 103 }
			}
			else_if = {
				limit = { NOT = { pp_cplan_goods_here_$demand$ = yes } }
				pp_lprof_fail = { r = 104 }
			}
			else = {
				pp_lprof_build = yes
			}
		}
	}
}

# Location scope (none of the target's buildings stands here): $type$ joins the start candidates when it can be built
# here and the market holds its construction goods; pp_lprof_start_pick starts the cheapest (a first level has no
# profit to measure, so no payback check).
pp_lprof_start = {
	if = {
		limit = { location_and_owner_can_build = { building_type = $type$ } }
		if = {
			limit = { NOT = { pp_cplan_goods_here_$demand$ = yes } }
			pp_lprof_fail = { r = 104 }
		}
		else = {
			building_type:$type$ = { save_scope_as = pp_cplan_btype }
			save_temporary_scope_value_as = { name = pp_cplan_ipl value = $ipl$ }
			save_temporary_scope_value_as = { name = pp_cplan_pf value = pp_cplan_cd_$demand$ }
			save_temporary_scope_value_as = { name = pp_cplan_ef value = pp_cplan_eff_$cat$ }
			pp_lprof_consider = yes
		}
	}
	else = {
		pp_lprof_fail = { r = 102 }
	}
}

# Location scope: scope:pp_cplan_btype becomes the pick when it costs less than the pick so far.
pp_lprof_consider = {
	if = {
		limit = { NOT = { has_variable = pp_lprof_best } }
		pp_lprof_keep = yes
	}
	else_if = {
		limit = { pp_lprof_cost1 < var:pp_lprof_best }
		pp_lprof_keep = yes
	}
}
pp_lprof_keep = {
	set_variable = { name = pp_lprof_best value = pp_lprof_cost1 }
	set_variable = { name = pp_lprof_pick value = scope:pp_cplan_btype }
}

# Location scope: starts the picked building, if any.
pp_lprof_start_pick = {
	if = {
		limit = { has_variable = pp_lprof_pick }
		var:pp_lprof_pick = { save_scope_as = pp_cplan_btype }
		pp_lprof_build = yes
	}
	remove_variable = pp_lprof_pick
	remove_variable = pp_lprof_best
}

# Location scope (market access): $type$ standing grows like pp_lprof_expand; not standing, it starts when it can be
# built here and the market holds its construction goods (the plan's logistics order: the first that can wins).
pp_lprof_try = {
	if = {
		limit = { any_buildings_in_location = { building_type = building_type:$type$ building_level > 0 } }
		pp_lprof_expand = { type = $type$ demand = $demand$ cat = $cat$ ipl = $ipl$ }
	}
	else_if = {
		limit = { scope:pp_cplan_country = { NOT = { has_variable = pp_cplan_hit } } }
		if = {
			limit = { NOT = { location_and_owner_can_build = { building_type = $type$ } } }
			pp_lprof_fail = { r = 102 }
		}
		else_if = {
			limit = { NOT = { pp_cplan_goods_here_$demand$ = yes } }
			pp_lprof_fail = { r = 104 }
		}
		else = {
			building_type:$type$ = { save_scope_as = pp_cplan_btype }
			pp_lprof_build = yes
		}
	}
}

# Location scope: one level of scope:pp_cplan_btype while the month's budget lasts (the plan's construct records it in
# the ledger as a profile's construction, why 105).
pp_lprof_build = {
	if = {
		limit = { pp_cplan_budget_left = yes }
		pp_cplan_construct_scoped = { why = 105 }
		if = {
			limit = { scope:pp_cplan_country = { has_variable = pp_cplan_hit } }
			set_variable = { name = pp_lprof_st value = 101 }
		}
	}
	else = {
		pp_lprof_fail = { r = 105 }
	}
}

# ---- Assignment ----

# Location scope (scope:pp_lprof_country = its owner): profile $n$ (slot + 100). Replaces the location's own rules or
# hands off (its stored own rules stay for later); the old states go.
pp_lprof_assign_to = {
	set_variable = { name = pp_lprof value = $n$ }
	pp_lprof_clear_states = yes
	save_temporary_scope_as = pp_lprof_this
	scope:pp_lprof_country = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = pp_lprof_locs target = scope:pp_lprof_this } } }
			add_to_variable_list = { name = pp_lprof_locs target = scope:pp_lprof_this }
		}
	}
	remove_variable = pp_cplan_mode
	remove_variable = pp_cplan_tpl
	pp_cplan_own_sync = yes
}

# Location scope: no profile here (also called when the location gets own rules or hands off, pp_cplan_own_sync).
pp_lprof_unassign = {
	remove_variable = pp_lprof
	pp_lprof_clear_states = yes
	save_temporary_scope_as = pp_lprof_this
	owner ?= {
		if = {
			limit = { is_target_in_variable_list = { name = pp_lprof_locs target = scope:pp_lprof_this } }
			remove_list_variable = { name = pp_lprof_locs target = scope:pp_lprof_this }
		}
	}
	refresh_map_colors = yes
}

# ---- Slot edits (country scope, root = the player's country) ----

# Slot $k$ on: icon lumber, colour $col$ (+ 100), no rules.
pp_lprof_new_slot = {
	set_variable = { name = pp_lprof_$k$_on value = yes }
	set_variable = { name = pp_lprof_$k$_ig value = goods:lumber }
	remove_variable = pp_lprof_$k$_ic
	set_variable = { name = pp_lprof_$k$_col value = $col$ }
	pp_lprof_clear_rules = { k = $k$ }
}

# The item scope:pp_key into the pair $g$ (goods scope, from scope:pp_lprof_val) / $c$ (category flag); the other one
# goes.
pp_lprof_set_target = {
	if = {
		limit = { exists = scope:pp_lprof_val }
		set_variable = { name = $g$ value = scope:pp_lprof_val }
		remove_variable = $c$
	}
	else_if = {
		limit = { pp_lprof_category_key = yes }
		set_variable = { name = $c$ value = scope:pp_key }
		remove_variable = $g$
	}
}

# Any scope: the goods scope in the variable $from$ -> $to$ (removed when $from$ is unset).
pp_lprof_copy_g = {
	if = {
		limit = { has_variable = $from$ }
		var:$from$ = { save_temporary_scope_as = pp_lprof_tmp }
		set_variable = { name = $to$ value = scope:pp_lprof_tmp }
	}
	else = {
		remove_variable = $to$
	}
}

# Rules $a$ and $b$ of slot $k$ swap places, and so do their states in the slot's locations (var:pp_lprof between $lo$
# and $hi$).
pp_lprof_swap_rules = {
	pp_lprof_copy_g = { from = pp_lprof_$k$_r$a$_g to = pp_lprof_swap_g }
	pp_lprof_copy_c = { from = pp_lprof_$k$_r$a$_c to = pp_lprof_swap_c }
	pp_cplan_copy_num = { from = pp_lprof_$k$_r$a$_pb to = pp_lprof_swap_pb }
	pp_lprof_copy_g = { from = pp_lprof_$k$_r$b$_g to = pp_lprof_$k$_r$a$_g }
	pp_lprof_copy_c = { from = pp_lprof_$k$_r$b$_c to = pp_lprof_$k$_r$a$_c }
	pp_cplan_copy_num = { from = pp_lprof_$k$_r$b$_pb to = pp_lprof_$k$_r$a$_pb }
	pp_lprof_copy_g = { from = pp_lprof_swap_g to = pp_lprof_$k$_r$b$_g }
	pp_lprof_copy_c = { from = pp_lprof_swap_c to = pp_lprof_$k$_r$b$_c }
	pp_cplan_copy_num = { from = pp_lprof_swap_pb to = pp_lprof_$k$_r$b$_pb }
	remove_variable = pp_lprof_swap_g
	remove_variable = pp_lprof_swap_c
	remove_variable = pp_lprof_swap_pb
	every_in_list = {
		variable = pp_lprof_locs
		limit = {
			has_variable = pp_lprof
			var:pp_lprof > $lo$
			var:pp_lprof < $hi$
		}
		pp_cplan_copy_num = { from = pp_lprof_r$a$_st to = pp_lprof_swap_st }
		pp_cplan_copy_num = { from = pp_lprof_r$b$_st to = pp_lprof_r$a$_st }
		pp_cplan_copy_num = { from = pp_lprof_swap_st to = pp_lprof_r$b$_st }
		remove_variable = pp_lprof_swap_st
	}
}

# Slot $k$ cleared and taken off every location of the country's list that carries it (var:pp_lprof between $lo$ and
# $hi$); a listed location another country owns now only leaves the list.
pp_lprof_delete_slot = {
	remove_variable = pp_lprof_$k$_on
	remove_variable = pp_lprof_$k$_ig
	remove_variable = pp_lprof_$k$_ic
	remove_variable = pp_lprof_$k$_col
	pp_lprof_clear_rules = { k = $k$ }
	every_in_list = {
		variable = pp_lprof_locs
		limit = {
			has_variable = pp_lprof
			var:pp_lprof > $lo$
			var:pp_lprof < $hi$
		}
		pp_lprof_mark_gone = yes
	}
	pp_lprof_drop_gone = yes
	refresh_map_colors = yes
}
