# Prosper or Perish - Construction Plan window buttons (gui/pp_construction_plan.gui, Global settings tab).
# Flags passed from the GUI: scope:pp_key (which setting or rule group), scope:pp_dir (up / down), scope:pp_n (w1..w10).

# Country scope: the plan on / off.
pp_cplan_toggle_on = {
	scope = country
	effect = {
		pp_cplan_flip = { var = pp_cplan_on }
	}
}

# Country scope: a rule group's weight (scope:pp_key = profiles / food / logistics / improve / business; scope:pp_n = the
# step of its bar clicked, w1..w10, or scope:pp_dir = up / down from its - and + buttons).
pp_cplan_weight = {
	scope = country
	effect = {
		switch = {
			trigger = scope:pp_key
			flag:profiles = { pp_cplan_set_weight = { var = pp_cplan_w_profiles current = pp_cplan_profiles_w } }
			flag:food = { pp_cplan_set_weight = { var = pp_cplan_w_food current = pp_cplan_food_w } }
			flag:logistics = { pp_cplan_set_weight = { var = pp_cplan_w_logistics current = pp_cplan_logistics_w } }
			flag:improve = { pp_cplan_set_weight = { var = pp_cplan_w_improve current = pp_cplan_improve_w } }
			flag:business = { pp_cplan_set_weight = { var = pp_cplan_w_business current = pp_cplan_business_w } }
		}
	}
}

# Country scope: one step of a line (scope:pp_key = reserve / reserve_big / access_line / improve_line /
# business_line, scope:pp_dir = up / down). The reserve moves 100 gold a click, 1,000 a right-click (reserve_big).
pp_cplan_step = {
	scope = country
	effect = {
		switch = {
			trigger = scope:pp_key
			flag:reserve = { pp_cplan_step_var = { var = pp_cplan_reserve current = pp_cplan_reserve_gold step = 100 min = 100 max = 1000100 } }
			flag:reserve_big = { pp_cplan_step_var = { var = pp_cplan_reserve current = pp_cplan_reserve_gold step = 1000 min = 100 max = 1000100 } }
			flag:access_line = { pp_cplan_step_var = { var = pp_cplan_access current = pp_cplan_access_pct step = 1 min = 150 max = 199 } }
			flag:improve_line = { pp_cplan_step_var = { var = pp_cplan_imp current = pp_cplan_improve_pct step = 10 min = 150 max = 400 } }
			flag:business_line = { pp_cplan_step_var = { var = pp_cplan_biz current = pp_cplan_business_years step = 1 min = 101 max = 150 } }
		}
	}
}
