# Prosper or Perish - Construction Plan panel buttons (gui/pp_construction_plan.gui). Flags passed from the GUI:
# scope:pp_key (which setting), scope:pp_dir (up / down), scope:pp_good + scope:pp_mode (goods modes).

# Country scope: the plan on / off.
pp_cplan_toggle_on = {
	scope = country
	effect = {
		pp_cplan_flip = { var = pp_cplan_on }
	}
}

# Country scope: a rule group on / off (scope:pp_key = food / logistics / goods).
pp_cplan_toggle_module = {
	scope = country
	effect = {
		switch = {
			trigger = scope:pp_key
			flag:food = { pp_cplan_flip = { var = pp_cplan_food_off } }
			flag:logistics = { pp_cplan_flip = { var = pp_cplan_logistics_off } }
			flag:goods = { pp_cplan_flip = { var = pp_cplan_goods_off } }
		}
	}
}

# Country scope: one step of a setting (scope:pp_key = reserve / max / access / keep, scope:pp_dir = up / down).
pp_cplan_step = {
	scope = country
	effect = {
		switch = {
			trigger = scope:pp_key
			flag:reserve = { pp_cplan_step_var = { var = pp_cplan_reserve current = pp_cplan_reserve_months step = 1 min = 100 max = 136 } }
			flag:max = { pp_cplan_step_var = { var = pp_cplan_max current = pp_cplan_max_per_month step = 1 min = 101 max = 120 } }
			flag:access = { pp_cplan_step_var = { var = pp_cplan_access current = pp_cplan_access_pct step = 5 min = 150 max = 195 } }
			flag:keep = { pp_cplan_step_var = { var = pp_cplan_keep current = pp_cplan_keep_months step = 1 min = 101 max = 124 } }
		}
	}
}

# Country scope: a good's mode (scope:pp_good = the good's flag, scope:pp_mode = off / keep / expand / never).
pp_cplan_good_mode = {
	scope = country
	effect = {
		pp_cplan_set_good_mode = yes
	}
}

# Province scope: the next focus (the cycle wraps). Custom starts from the band the province had.
pp_cplan_cycle_focus = {
	scope = province
	effect = {
		if = {
			limit = { NOT = { has_variable = pp_cplan_lo } }
			set_variable = { name = pp_cplan_lo value = pp_cplan_band_lo100 }
		}
		if = {
			limit = { NOT = { has_variable = pp_cplan_hi } }
			set_variable = { name = pp_cplan_hi value = pp_cplan_band_hi100 }
		}
		set_variable = { name = pp_cplan_focus value = { value = pp_cplan_focus_n add = 101 } }
		if = {
			limit = { var:pp_cplan_focus > pp_cplan_focus_last100 }
			set_variable = { name = pp_cplan_focus value = 101 }
		}
	}
}

# Province scope: moves one end of the band (scope:pp_key = lo / hi, scope:pp_dir = up / down); the province becomes
# Custom with the band it had, and the band never turns over.
pp_cplan_step_band = {
	scope = province
	effect = {
		set_variable = { name = pp_cplan_lo value = pp_cplan_band_lo100 }
		set_variable = { name = pp_cplan_hi value = pp_cplan_band_hi100 }
		set_variable = { name = pp_cplan_focus value = pp_cplan_custom_id100 }
		if = {
			limit = { scope:pp_key = flag:lo }
			pp_cplan_step_var = { var = pp_cplan_lo current = pp_cplan_band_lo step = 1 min = 100 max = 124 }
			if = {
				limit = { var:pp_cplan_lo > var:pp_cplan_hi }
				set_variable = { name = pp_cplan_hi value = var:pp_cplan_lo }
			}
		}
		else = {
			pp_cplan_step_var = { var = pp_cplan_hi current = pp_cplan_band_hi step = 1 min = 100 max = 124 }
			if = {
				limit = { var:pp_cplan_hi < var:pp_cplan_lo }
				set_variable = { name = pp_cplan_lo value = var:pp_cplan_hi }
			}
		}
	}
}

# Location scope: the logistics override, Follow the plan -> Always -> Never -> Follow.
pp_cplan_cycle_logi = {
	scope = location
	effect = {
		if = {
			limit = { NOT = { has_variable = pp_cplan_logi } }
			set_variable = { name = pp_cplan_logi value = 101 }
		}
		else_if = {
			limit = { var:pp_cplan_logi < 101.5 }
			set_variable = { name = pp_cplan_logi value = 102 }
		}
		else = {
			remove_variable = pp_cplan_logi
		}
	}
}

# Location scope: Hands off (the plan builds nothing here) on / off.
pp_cplan_toggle_here = {
	scope = location
	effect = {
		pp_cplan_flip = { var = pp_cplan_here_off }
	}
}

# Location scope: a row of the Logistics tab (below full market access, or an override set).
pp_cplan_logistics_row = {
	scope = location
	is_shown = {
		OR = {
			market_access < 0.99
			has_variable = pp_cplan_logi
			has_variable = pp_cplan_here_off
		}
	}
}
