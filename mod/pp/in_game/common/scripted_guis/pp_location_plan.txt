# Prosper or Perish - Location plan buttons (gui/pp_location_plan*.gui, the location window's plan button and its Plans
# row). They set and take off the location's plans (town rights pp_plan_<plan>): what the Construction Plan builds
# toward there. scope:pp_key = the plan's flag. The GUI shows them only to the player who owns the location (observers'
# commands are dropped); the script checks the owner is a player too.

# Location scope: show this location in the Location Plan window.
pp_lplan_select = {
	scope = location
	effect = {
		pp_lplan_select = yes
	}
}

# Location scope, scope:pp_key = flag:<plan>: set the plan, or take it off when held.
pp_tplan_toggle = {
	scope = location
	is_valid = {
		exists = owner
		owner = { is_ai = no }
	}
	effect = {
		pp_tplan_toggle_switch = yes
	}
}

# Location scope: take off every plan.
pp_tplan_clear = {
	scope = location
	is_valid = {
		exists = owner
		owner = { is_ai = no }
	}
	effect = {
		pp_tplan_clear_do = yes
	}
}
