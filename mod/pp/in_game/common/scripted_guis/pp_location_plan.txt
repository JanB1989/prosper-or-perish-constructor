# Prosper or Perish - Location plan buttons (gui/pp_location_plan*.gui, the location window's plan button and its row
# "Also build here"). They set the location's marks: what the Construction Plan also builds there, and what the window
# and the map mode show. scope:pp_key = the item's flag (a good); scope:pp_bt = a building type (a building's mark).

# Location scope: show this location in the Location Plan window.
pp_lplan_select = {
	scope = location
	effect = {
		pp_lplan_select = yes
	}
}

# Location scope: mark or unmark one item.
pp_lplan_toggle = {
	scope = location
	effect = {
		pp_lplan_toggle_switch = yes
	}
}

# Location scope, scope:pp_bt = a building type: mark or unmark that building here (the toggle on a building's entry
# in the location window and the build panel, and the tiles of the row "Also build here").
pp_lplan_bmark = {
	scope = location
	effect = {
		pp_lplan_bflip = yes
	}
}

# Country scope, scope:pp_bt = a building type: Ctrl + click on vanilla's auto expand for all buildings of a type (the
# Production window's rows, the build panel's right click). Marks the type in every own location where it stands; when
# every one of them is marked already, unmarks them all. Script cannot see the GUI's filtered list.
pp_lplan_bmark_all = {
	scope = country
	effect = {
		if = {
			limit = {
				any_owned_location = {
					any_buildings_in_location = { building_type = scope:pp_bt }
					NOT = { is_target_in_variable_list = { name = pp_lplan_b target = scope:pp_bt } }
				}
			}
			every_owned_location = {
				limit = {
					any_buildings_in_location = { building_type = scope:pp_bt }
					NOT = { is_target_in_variable_list = { name = pp_lplan_b target = scope:pp_bt } }
				}
				pp_lplan_bflip = yes
			}
		}
		else = {
			every_owned_location = {
				limit = {
					any_buildings_in_location = { building_type = scope:pp_bt }
					is_target_in_variable_list = { name = pp_lplan_b target = scope:pp_bt }
				}
				pp_lplan_bflip = yes
			}
		}
	}
}

# Location scope, scope:pp_bt = a building type: shown while that building is marked here (the GUI asks it with
# IsShown: it cannot look into a variable list itself). No effect.
pp_lplan_bmarked = {
	scope = location
	is_shown = {
		exists = scope:pp_bt
		is_target_in_variable_list = { name = pp_lplan_b target = scope:pp_bt }
	}
}

# Location scope, scope:pp_bt = a marked building type: shown while a level of it, or of a later tier of its line,
# stands here (the map icon's green tile; generated trigger pp_lplan_b_done). No effect.
pp_lplan_bdone = {
	scope = location
	is_shown = {
		exists = scope:pp_bt
		pp_lplan_b_done = yes
	}
}

# Location scope: remove every mark.
pp_lplan_clear = {
	scope = location
	effect = {
		pp_lplan_clear_do = yes
	}
}
