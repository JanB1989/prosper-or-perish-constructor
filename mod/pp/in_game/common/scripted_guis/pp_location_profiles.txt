# Prosper or Perish - Location Profiles buttons (docs/location_profiles.md): the profile editor (the Construction Plan
# window's Location profiles tab), the Location Plan window's profile row, the location window's row, the map's
# right-click menu and the paint mode call these. Parameters
# are flag scopes the GUI adds with AddScope: scope:pp_slot (s1..sN; e = the profile in the editor's buffer; none for
# assign; next = the first free slot for new), scope:pp_field (icon, r1..rN), scope:pp_key (an item: a good, logistics =
# market access, a building category, or a symbol for the icon), scope:pp_dir (up / down), scope:pp_col (c1..cN).
# Country-scope buttons run with root = the player's country (GuiScope.SetRoot(GetPlayer.MakeScope)). The effects are
# in scripted_effects/pp_location_profiles*.txt.

# Location scope (root = the location): its profile scope:pp_slot (a slot that is on), or none.
pp_lprof_assign = {
	scope = location
	effect = {
		pp_lprof_assign_do = yes
	}
}

# Country scope: a new profile in slot scope:pp_slot (next = the first free one: icon lumber, its colour, no rules);
# the editor shows it.
pp_lprof_new = {
	scope = country
	effect = {
		pp_lprof_new_do = yes
	}
}

# Country scope: the editor shows slot scope:pp_slot (its data goes into the edit buffer).
pp_lprof_select = {
	scope = country
	effect = {
		pp_lprof_select_do = yes
	}
}

# Country scope: the icon (scope:pp_field = icon) or rule r's target (r1..rN) of slot scope:pp_slot = the item scope:pp_key.
pp_lprof_set = {
	scope = country
	effect = {
		pp_lprof_set_do = yes
	}
}

# Country scope: the colour scope:pp_col of slot scope:pp_slot.
pp_lprof_color = {
	scope = country
	effect = {
		pp_lprof_color_do = yes
	}
}

# Country scope: a global rule group (scope:pp_field: global_food / global_logistics / global_improve /
# global_business) of slot scope:pp_slot off in its locations, or on again.
pp_lprof_global = {
	scope = country
	effect = {
		pp_lprof_global_do = yes
	}
}

# Country scope: the example profiles ([[location_profiles.examples]]), each in the next free slot.
pp_lprof_examples = {
	scope = country
	effect = {
		pp_lprof_examples_do = yes
	}
}

# Country scope: one step (scope:pp_dir) of the payback limit of rule scope:pp_field of slot scope:pp_slot.
pp_lprof_payback = {
	scope = country
	effect = {
		pp_lprof_payback_do = yes
	}
}

# Country scope: empties rule scope:pp_field of slot scope:pp_slot.
pp_lprof_rule_clear = {
	scope = country
	effect = {
		pp_lprof_rule_clear_do = yes
	}
}

# Country scope: swaps rule scope:pp_field of slot scope:pp_slot with its neighbour in the direction scope:pp_dir.
pp_lprof_rule_move = {
	scope = country
	effect = {
		pp_lprof_rule_move_do = yes
	}
}

# Country scope: clears slot scope:pp_slot and takes it off every location.
pp_lprof_delete = {
	scope = country
	effect = {
		pp_lprof_delete_do = yes
	}
}
