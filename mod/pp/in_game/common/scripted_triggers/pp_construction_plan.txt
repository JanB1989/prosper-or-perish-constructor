# Prosper or Perish - Construction Plan triggers (player-only construction rules; see on_action/pp_construction_plan.txt).
# Every trigger here runs inside pp_cplan_run, where scope:pp_cplan_country is the player country, unless noted.

# Location scope: the global rules may build here (the player's own location).
# A location profile takes from the global rules only what it sets (Jan 2026-11-09): the targets of its rules and the
# global groups it switches off (pp_lprof_x*); everything else the global groups still build there, so a profile need
# not repeat them. The checks per group follow.
pp_cplan_location_ok = {
	owner ?= scope:pp_cplan_country
}

# Location scope: the food rule may build here (the province capital's Tavern, kitchen, Granary).
pp_cplan_food_ok = {
	pp_cplan_location_ok = yes
	NOT = { has_variable = pp_lprof_xf }
}

# Location scope: the food rule's farm step may build here (a profile with a staple food rule decides the food here).
pp_cplan_farm_ok = {
	pp_cplan_food_ok = yes
	NOT = { pp_lprof_food_target = yes }
}

# Location scope: the food rule may build a victuals maker here (Grange, Victualling Yard).
pp_cplan_victuals_ok = {
	pp_cplan_food_ok = yes
	NOT = { pp_lprof_has_good = { g = victuals } }
}

# Location scope: the logistics rule may build here (a profile with a market access rule decides it here).
pp_cplan_logistics_ok = {
	pp_cplan_location_ok = yes
	NOT = { has_variable = pp_lprof_xl }
	NOT = { pp_lprof_has_good = { g = logistics } }
}

# Location scope: the improvements rule may build here.
pp_cplan_improve_ok = {
	pp_cplan_location_ok = yes
	NOT = { has_variable = pp_lprof_xi }
}

# Location scope: a construction can start here: none is under way, whoever started it (a rule, the game's own
# building automation, the player). The engine builds a location's constructions one after another, so a second one
# paid now only waits (test world 2026-11-09: in Ocana a Cookshop moved 344 days in a year while the Granary, Grange
# and Carrier Inn the plan had started beside it moved 0). The ONE place for this check: every rule group asks here.
# A location has as many construction slots as its modifier max_constructions_at_same_time (in game 2026-11-10:
# rural 1, town 1, London 2; London's two guild levels moved together while a third waited, Cambridge built the
# same pair one after the other). num_civil_constructions counts everything started there, queued and the
# estates' constructions too, so it can be above the slots.
pp_cplan_site_free = {
	num_civil_constructions < pp_cplan_site_slots
}

# Location scope: the business rule looks for a business here: the player's location, held by the player, with a
# market, no construction under way (pp_cplan_site_free), and no profile here that keeps the businesses out. A
# profile's own goods and categories are left out per building (generated pp_cplan_business_pick).
pp_cplan_business_site = {
	pp_cplan_location_ok = yes
	NOT = { has_variable = pp_lprof_xg }
	controller ?= scope:pp_cplan_country
	pp_cplan_site_free = yes
	market ?= { always = yes }
}

# Location scope, scope:pp_cplan_free_peasants saved (pp_cplan_free_peasants_here): a level of a building that employs
# $type$ finds its $size$ workers here (thousands; Jan 2026-11-09: no business where nobody can work in it). Idle
# peasants left after the open jobs here staff any building (they are promoted), idle pops of its own class too
# (generated pp_cplan_idle_<pop type>, 0 for peasants: their count is the first line).
# The class is read through a value per pop type with the type written out, the form vanilla and the mod's other
# files use. With the argument inside the quoted call ("unemployed_pops_of_pop_type_in_location(pop_type:$type$)")
# the trigger logged "Script system error! ... trigger [ pop_type ]" on every check (about 2,000 lines a session, in
# game 2026-11-10); no vanilla file puts an argument inside quotes, so most likely the game does not fill it in there.
pp_cplan_staff = {
	OR = {
		scope:pp_cplan_free_peasants >= $size$
		pp_cplan_idle_$type$ >= $size$
	}
}

# Any scope: the rule group $why$ saves: this pass it already met something it wants and cannot pay for from its
# purse (pp_cplan_note_wait), so it builds nothing more in the pass, even things that would fit (Jan 2026-11-10: the
# plan must not favour cheap buildings because they are cheap).
pp_cplan_saving = {
	scope:pp_cplan_country = { has_variable = pp_cplan_save_cost_$why$ }
}

# Location scope, after pp_cplan_btype_data: the next level of scope:pp_cplan_btype here (its cost by the engine's
# formula) fits in the purse of the rule group $why$, and in the gold above the reserve (the purses never hold more
# than that at the pass's start; the second line keeps the reserve whatever happened since).
pp_cplan_affordable = {
	pp_cplan_est_cost <= scope:pp_cplan_country.pp_cplan_purse_n_$why$
	pp_cplan_est_cost <= scope:pp_cplan_country.pp_cplan_free_gold
}

# Location scope: the location's market has room for one more level that makes $out$ of the good $g$ a month (the
# market limit, Jan 2026-11-10): what is under construction of the good in the market (pp_cplan_uc_$g$ on the
# market's centre location, counted by pp_cplan_market_load, + 100) plus this level stays within the limit's share of
# the market's monthly demand for it (pp_cplan_mroom_<good>), or nothing of it is under construction there: one level
# is always allowed. A location without a market has no limit.
pp_cplan_market_room = {
	OR = {
		NOT = { market ?= { always = yes } }
		market.location ?= {
			OR = {
				NOT = { has_variable = pp_cplan_uc_$g$ }
				var:pp_cplan_uc_$g$ < 100.001
				pp_cplan_mroom_$g$ >= $out$
			}
		}
	}
}

# Any scope: the current rule has not built yet. The gold is checked by the construction itself
# (pp_cplan_construct_scoped), so a step the gold does not reach is still counted for the window.
pp_cplan_may_try = {
	scope:pp_cplan_country = { NOT = { has_variable = pp_cplan_hit } }
}

# Location scope: the logistics rule wants a level here: market access without event modifiers below the line while
# the owner controls the location. The building's own gates (access, building levels, one level at a time) still
# apply. No market-switch wait (Jan: no history): in run af7dce96, 98.7 % of the owned locations below 97 % at a save
# were below it five years earlier too.
pp_cplan_logistics_wanted = {
	controller ?= owner
	pp_cplan_access_normal < pp_cplan_access_line
}

# Province scope, in pp_cplan_food_province (scope:pp_cplan_fy = pp_cplan_food_in_a_year, read once a province): the
# hunger rule: a normal year leaves less than pp_cplan_hungry_months in the store a year from now (a bad harvest alone
# does not count, a province that refills by itself within the year neither).
pp_cplan_hungry = {
	has_variable = pp_stored_food_months
	scope:pp_cplan_fy < pp_cplan_hungry_months
}

# Province scope, in pp_cplan_food_province: the famine rule: the same, below the famine line.
pp_cplan_famine = {
	has_variable = pp_stored_food_months
	scope:pp_cplan_fy < pp_cplan_famine_months
}

# Province scope: the Granary rule: the store can hold less than the line's months of consumption, it is at least the
# line's share full, and a normal month still adds food (otherwise the surplus, not the room, keeps the store low and
# a Granary adds nothing).
pp_cplan_granary_due = {
	has_variable = pp_stored_food_months
	pp_cplan_capacity_months > 0
	pp_cplan_capacity_months < pp_cplan_granary_cap
	pp_cplan_fill_pct >= pp_cplan_granary_fill
	pp_cplan_normal_gain > 0
}

# Province scope: the Grange rule: the store holds 24 months (the stored-food effects stop rising there) and a normal
# month still adds the line's share of consumption on top. A working Grange packs the store below 24 months, so the
# next level waits until it is full again.
pp_cplan_grange_due = {
	has_variable = pp_stored_food_months
	var:pp_stored_food_months > 123.5
	pp_cplan_normal_gain_pct >= pp_cplan_grange_surplus
}

# Location scope: the location's market holds the victuals a Tavern would serve.
pp_cplan_victuals_here = {
	market ?= {
		"stockpile_in_market(goods:victuals)" >= pp_cplan_tavern_victuals
	}
}
