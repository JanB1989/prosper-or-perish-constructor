# PP (2026-11-27, Jan): long disasters, growing pain (AGENTS.md "Long Disasters: Growing Pain"). The copies of the
# general disasters in common/disasters (written by `ppc disasters apply`) call these with their own type; the relaxed
# end triggers pp_disaster_relief_<type>_trigger are generated next to this file (pp_disaster_relief_triggers.txt).

# Country scope: the disaster has lasted long enough for the relaxed end (vanilla's end with less stability).
pp_disaster_relief_due_trigger = {
	had_disaster_for_years = {
		disaster_type = disaster_type:$type$
		years >= 50
	}
}

# Country scope: no restart while the block a relieved ending set runs (block_disaster_temporarily_effect, vanilla's
# variable blocked_disaster_<type>, which only revolution_disaster reads in vanilla).
pp_disaster_restart_allowed_trigger = {
	NOT = { has_blocked_disaster_trigger = { type = $type$ } }
}
