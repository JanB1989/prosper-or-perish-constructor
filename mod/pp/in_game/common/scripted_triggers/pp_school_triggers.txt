# Country scope: the Education of the Masses law (education_masses_law) is set to a policy that teaches the common
# people, i.e. one whose vanilla effect included a literacy cap for peasants or for everyone: basic religious,
# theocratic, Scottish presbyterian, personal religious (printing press), secular (Enlightenment) and Vedic education.
# Not: uneducated masses, apprenticeships (burghers only), oral tradition and the two epic traditions (no literacy).
pp_educates_the_masses = {
	OR = {
		has_policy = basic_religious_education
		has_policy = theocratic_education
		has_policy = scottish_presbyterian_education
		has_policy = personal_religious_education
		has_policy = secular_education
		has_policy = vedic_education
	}
}
