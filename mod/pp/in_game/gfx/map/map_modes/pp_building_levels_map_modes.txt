# Map mode: total building levels per location (traffic-light bucketed heatmap).
# Clean manual/geometric-style buckets keep dense locations readable without
# leaving the existing traffic-light map palette.

@pp_building_levels_very_low = 10
@pp_building_levels_low = 25
@pp_building_levels_medium = 50
@pp_building_levels_high = 75
@pp_building_levels_strong = 100
@pp_building_levels_very_strong = 150
@pp_building_levels_dense = 200

@pp_building_levels_cap = 500
@pp_supported_building_levels_low = 30
@pp_supported_building_levels_mid = 70

@data_edge_sharpness = 0.01

@data_edge_color_mult = 0
@data_before_lighting_blend = 0.5
@data_after_lighting_blend = 0.5

pp_building_levels = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = { has_owner = yes }
			if = {
				limit = { total_building_levels < @pp_building_levels_very_low }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MIN
					max_color = define:NMapColors|MAP_COLOR_LOW
					factor = { value = total_building_levels divide = @pp_building_levels_very_low max = 1 min = 0 }
				}

				limit = { total_building_levels < @pp_building_levels_low }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_LOW

# Map mode: local building efficiency modifier per location.
# Diverging traffic-light heatmap: red = costly, yellow = near neutral, green = efficient.

pp_building_efficiency = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = { has_owner = yes }
			if = {
				limit = { pp_building_efficiency_map_value < -1 }
				value = define:NMapColors|MAP_COLOR_MIN
			}
			else_if = {
				limit = { pp_building_efficiency_map_value < -0.5 }
				lerp = {
					min_color = define:NMapColors|MAP_COLOR_MIN
					max_color = define:NMapColors|MAP_COLOR_LOW

						value = pp_building_efficiency_map_value

					min_color = define:NMapColors|MAP_COLOR_LOW
					max_color = define:NMapColors|MAP_COLOR_MID
