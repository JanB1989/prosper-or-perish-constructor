# Prosper or Perish - Location Plan map mode (branch location-plan, 2026-11-07, Jan): the planned icons over the player's
# locations (marked in the Location Plan window, gui/pp_location_plan.gui). Only visual.
#
# The icons are vanilla's raw goods marker (map_markers.gui, patched by location_plan.py: in this map mode it draws the
# plan's icons on a green tile once done, red while still to do). The engine creates raw goods markers only with the
# raw_material color mode (tested 2026-11-07: map_markers = { raw_goods_marker = yes } alone shows none, also not in
# vanilla's market_food_balance; raw_material map names do not create them either), so this mode takes that color mode
# and fades it almost out: a neutral map under the icons. A map_color block would be ignored with a color_mode.

@zoom_step_near = 			2	# closest zoom
@data_gradient_alpha_inside = 			0.12	# the raw material colors faded almost out
@data_gradient_alpha_outside = 			0.12
@data_gradient_width = 					0.25
@data_gradient_color_mult = 			0.9
@data_edge_width = 						0
@data_edge_sharpness = 					0.01
@data_edge_alpha = 						0
@data_edge_color_mult = 				0
@data_before_lighting_blend = 			0.5
@data_after_lighting_blend = 			0.5

pp_location_plan = {
	color_mode = raw_material

	legend_key = { desc = "MAPMODE_PP_LOCATION_PLAN_DONE" color = rgb { 64 179 77 } }
	legend_key = { desc = "MAPMODE_PP_LOCATION_PLAN_OPEN" color = rgb { 191 64 51 } }
	# no tooltip_key: ignored with a color_mode (error.log); each icon has its own tooltip

	small_map_names = location
	medium_map_names = location
	large_map_names = location

	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location

	fill_in_impassable = no
	enable_snow = no
	enable_rivers = yes

	flatmap_behaviour = Always
	use_fow = no

	category = economy
	index = 0	# the row in the category's menu (hud_bot.gui shows economy rows 0-3 only; 40 hid the mode): first row

	# vanilla's raw goods marker, which draws the plan's icons in this map mode (map_markers.gui, location_plan.py)
	map_markers = {
		all = no
		raw_goods_marker = yes
	}

	gradient_parameters = {
		zoom_step = @zoom_step_near
		gradient_alpha_inside = 	@data_gradient_alpha_inside
		gradient_alpha_outside = 	@data_gradient_alpha_outside
		gradient_width = 			@data_gradient_width
		gradient_color_mult =		@data_gradient_color_mult
		edge_width = 				@data_edge_width
		edge_sharpness = 			@data_edge_sharpness
		edge_alpha = 				@data_edge_alpha
		edge_color_mult = 			@data_edge_color_mult
		before_lighting_blend =		@data_before_lighting_blend
		after_lighting_blend =		@data_after_lighting_blend
	}

	refresh_colors_on_selection_change = yes
	color_refresh_counters = { Month ProductionList }
	color_and_names_refresh_counters = { LocationOwnerChanged CountryStatus }
}
